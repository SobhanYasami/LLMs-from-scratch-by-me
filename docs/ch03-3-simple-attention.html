3.3: A simple self-attention mechanism

Book section 3.3 · code in Python, Go, Rust, C, Zig

Aim. See, with no trainable weights, how every token gets a context vector that blends in the other tokens.

Take the sentence "Your journey starts with one step." already embedded as six 3-dimensional vectors (chapter 2 made these). Row 1 is "Your", row 2 is "journey", and so on. For each token we want a context vector: its own embedding, enriched with information from all the other tokens.

x1 Your     [0.43, 0.15, 0.89]
x2 journey  [0.55, 0.87, 0.66]
x3 starts   [0.57, 0.85, 0.64]
x4 with     [0.22, 0.58, 0.33]
x5 one      [0.77, 0.25, 0.10]
x6 step     [0.05, 0.80, 0.55]

Three steps, for one query (x2, "journey")

  1. Scores. Dot product of the query with every token. A dot product is a similarity measure: the more two vectors point the same way, the bigger it is. 
    scores = [0.9544, 1.4950, 1.4754, 0.8434, 0.7070, 1.0865]
  2. Weights. Normalise the scores so they are positive and sum to 1. Plain division by the sum works, but softmax is standard: it handles extreme values better and has friendlier gradients. 
    divide by sum: [0.1455, 0.2278, 0.2249, 0.1285, 0.1077, 0.1656]
softmax      : [0.1385, 0.2379, 0.2333, 0.1240, 0.1082, 0.1581]
  3. Context vector. Weighted sum of all token vectors, using those weights. 
    z2 = 0.1385*x1 + 0.2379*x2 + ... + 0.1581*x6 = [0.4419, 0.6515, 0.5683]
Softmax in code subtracts the largest score first. The result is identical, but exp can no longer overflow on big numbers.

All tokens at once: three matrix operations

Stack the loop over all queries and it becomes matrix algebra. With X the 6x3 input matrix:

scores  = X x X^T                  (6 x 6)  every token against every token
weights = softmax(scores, row by row)          each row sums to 1
context = weights x X                (6 x 3)  one context vector per token
weights (row 2 is the one we computed by hand):
  [0.2098, 0.2006, 0.1981, 0.1242, 0.1220, 0.1452]
  [0.1385, 0.2379, 0.2333, 0.1240, 0.1082, 0.1581]   <- journey
  ...
context vectors:
  [0.4421, 0.5931, 0.5790]
  [0.4419, 0.6515, 0.5683]   <- z2, same as before

Row 2 of the matrix result equals the vector we built step by step. That is the whole check. The matrix form is just the loop, done faster.

What is missing

Nothing here learns. The weights come straight from raw embedding similarity. The next page adds three trainable matrices so the model can learn what to look for.

"""Step 6 (book 3.3) - simplified self-attention, no trainable weights.

For every token, build a *context vector*: an enriched embedding that mixes
in information from all the other tokens, weighted by how relevant each is.

    scores  = X @ X.T                  how similar is token i to token j?   [T, T]
    weights = softmax(scores, rows)    turn each row into weights summing to 1
    context = weights @ X              weighted sum of all token vectors   [T, d]

Tip: a dot product measures similarity. Similar vectors -> big score -> big weight.
There is nothing to learn here yet; trainable weights arrive in 3.4.
"""

import torch


def simple_attention(inputs: torch.Tensor):
    """``inputs`` is [T, d]. Returns (scores, weights, context)."""
    scores = inputs @ inputs.T
    weights = torch.softmax(scores, dim=-1)
    context = weights @ inputs
    return scores, weights, context

// Package attention holds the attention building blocks (book chapter 3).
//
// Files, in the order you learn them:
//   - matrix.go:         the few matrix operations attention needs.
//   - rng.go:            a seeded random generator shared with the other languages.
//   - simple.go:         simplified self-attention, no trainable weights (3.3).
//   - self_attention.go: scaled dot-product attention with W_query/key/value (3.4).
//   - causal.go:         hide future tokens with a mask, optional dropout (3.5).
//   - multihead.go:      several heads in parallel plus an output projection (3.6).
//
// Tip: there is no tensor library here, so Matrix is a row-major []float32.
// Everything attention does is these operations: matmul, transpose, softmax.
package attention

import "math"

// Matrix is a row-major rows x cols matrix of float32.
type Matrix struct {
	Rows, Cols int
	Data       []float32
}

func NewMatrix(rows, cols int) Matrix {
	return Matrix{rows, cols, make([]float32, rows*cols)}
}

// FromRows builds a matrix from a slice of equal-length rows.
func FromRows(rows [][]float32) Matrix {
	m := NewMatrix(len(rows), len(rows[0]))
	for i, r := range rows {
		copy(m.Data[i*m.Cols:], r)
	}
	return m
}

func (m Matrix) At(i, j int) float32     { return m.Data[i*m.Cols+j] }
func (m Matrix) Set(i, j int, v float32) { m.Data[i*m.Cols+j] = v }
func (m Matrix) Row(i int) []float32     { return m.Data[i*m.Cols : (i+1)*m.Cols] }

// MatMul returns a x b ([n,k] x [k,m] -> [n,m]).
func MatMul(a, b Matrix) Matrix {
	out := NewMatrix(a.Rows, b.Cols)
	for i := 0; i < a.Rows; i++ {
		for j := 0; j < b.Cols; j++ {
			var sum float32
			for k := 0; k < a.Cols; k++ {
				sum += a.At(i, k) * b.At(k, j)
			}
			out.Set(i, j, sum)
		}
	}
	return out
}

func Transpose(m Matrix) Matrix {
	out := NewMatrix(m.Cols, m.Rows)
	for i := 0; i < m.Rows; i++ {
		for j := 0; j < m.Cols; j++ {
			out.Set(j, i, m.At(i, j))
		}
	}
	return out
}

// SoftmaxRows turns every row into weights that are positive and sum to 1.
// Tip: subtracting the row maximum first does not change the result but keeps
// exp() from overflowing. A -Inf entry becomes exactly 0.
func SoftmaxRows(m Matrix) Matrix {
	out := NewMatrix(m.Rows, m.Cols)
	for i := 0; i < m.Rows; i++ {
		row := m.Row(i)
		max := row[0]
		for _, v := range row {
			if v > max {
				max = v
			}
		}
		var sum float32
		for j, v := range row {
			e := float32(math.Exp(float64(v - max)))
			out.Set(i, j, e)
			sum += e
		}
		for j := 0; j < m.Cols; j++ {
			out.Set(i, j, out.At(i, j)/sum)
		}
	}
	return out
}

// Scale multiplies every element by s.
func Scale(m Matrix, s float32) Matrix {
	out := NewMatrix(m.Rows, m.Cols)
	for i, v := range m.Data {
		out.Data[i] = v * s
	}
	return out
}

// MaskFuture sets every element above the diagonal to -Inf (the "future" tokens).
func MaskFuture(m Matrix) Matrix {
	out := NewMatrix(m.Rows, m.Cols)
	copy(out.Data, m.Data)
	for i := 0; i < m.Rows; i++ {
		for j := i + 1; j < m.Cols; j++ {
			out.Set(i, j, float32(math.Inf(-1)))
		}
	}
	return out
}

// SliceCols returns columns [start, end) as a new matrix.
func SliceCols(m Matrix, start, end int) Matrix {
	out := NewMatrix(m.Rows, end-start)
	for i := 0; i < m.Rows; i++ {
		copy(out.Row(i), m.Row(i)[start:end])
	}
	return out
}

// Step 6 (book 3.3) - simplified self-attention, no trainable weights.
//
// For every token, build a context vector: an enriched embedding that mixes in
// information from all the other tokens, weighted by how relevant each is.
//
//	scores  = X x X^T                  how similar is token i to token j?   [T, T]
//	weights = softmax(scores, rows)    turn each row into weights summing to 1
//	context = weights x X              weighted sum of all token vectors   [T, d]
//
// Tip: a dot product measures similarity. Similar vectors -> big score -> big weight.
// There is nothing to learn here yet; trainable weights arrive in 3.4.

// SimpleAttention takes inputs [T, d] and returns (scores, weights, context).
func SimpleAttention(inputs Matrix) (scores, weights, context Matrix) {
	scores = MatMul(inputs, Transpose(inputs))
	weights = SoftmaxRows(scores)
	context = MatMul(weights, inputs)
	return
}
rust-version/src/attention/matrix.rs
//! The few matrix operations attention needs.
//!
//! Tip: there is no tensor library here, so `Matrix` is a row-major `Vec<f32>`.
//! Everything attention does is these operations: matmul, transpose, softmax.

use crate::tokenization::rng::Rng;

/// A row-major rows x cols matrix of f32.
#[derive(Clone, Debug)]
pub struct Matrix {
    pub rows: usize,
    pub cols: usize,
    pub data: Vec<f32>,
}

impl Matrix {
    pub fn new(rows: usize, cols: usize) -> Self {
        Self {
            rows,
            cols,
            data: vec![0.0; rows * cols],
        }
    }

    /// Build from rows of equal length.
    pub fn from_rows(rows: &[&[f32]]) -> Self {
        let mut m = Self::new(rows.len(), rows[0].len());
        for (i, r) in rows.iter().enumerate() {
            m.data[i * m.cols..(i + 1) * m.cols].copy_from_slice(r);
        }
        m
    }

    /// A rows x cols matrix, uniform in [0, 1), like `torch.rand`.
    pub fn rand(rows: usize, cols: usize, rng: &mut Rng) -> Self {
        let data = (0..rows * cols).map(|_| rng.next_f64() as f32).collect();
        Self { rows, cols, data }
    }

    pub fn at(&self, i: usize, j: usize) -> f32 {
        self.data[i * self.cols + j]
    }

    pub fn row(&self, i: usize) -> &[f32] {
        &self.data[i * self.cols..(i + 1) * self.cols]
    }

    /// a x b: [n,k] x [k,m] -> [n,m].
    pub fn matmul(&self, b: &Matrix) -> Matrix {
        let mut out = Matrix::new(self.rows, b.cols);
        for i in 0..self.rows {
            for j in 0..b.cols {
                let mut sum = 0.0;
                for k in 0..self.cols {
                    sum += self.at(i, k) * b.at(k, j);
                }
                out.data[i * b.cols + j] = sum;
            }
        }
        out
    }

    pub fn transpose(&self) -> Matrix {
        let mut out = Matrix::new(self.cols, self.rows);
        for i in 0..self.rows {
            for j in 0..self.cols {
                out.data[j * self.rows + i] = self.at(i, j);
            }
        }
        out
    }

    /// Turn every row into weights that are positive and sum to 1.
    ///
    /// Tip: subtracting the row maximum first does not change the result but keeps
    /// `exp` from overflowing. A `-inf` entry becomes exactly 0.
    pub fn softmax_rows(&self) -> Matrix {
        let mut out = Matrix::new(self.rows, self.cols);
        for i in 0..self.rows {
            let row = self.row(i);
            let max = row.iter().cloned().fold(f32::NEG_INFINITY, f32::max);
            let mut sum = 0.0;
            for (j, &v) in row.iter().enumerate() {
                let e = ((v - max) as f64).exp() as f32;
                out.data[i * self.cols + j] = e;
                sum += e;
            }
            for j in 0..self.cols {
                out.data[i * self.cols + j] /= sum;
            }
        }
        out
    }

    /// Multiply every element by `s`.
    pub fn scale(&self, s: f32) -> Matrix {
        Matrix {
            rows: self.rows,
            cols: self.cols,
            data: self.data.iter().map(|v| v * s).collect(),
        }
    }

    /// Set every element above the diagonal to `-inf` (the "future" tokens).
    pub fn mask_future(&self) -> Matrix {
        let mut out = self.clone();
        for i in 0..self.rows {
            for j in i + 1..self.cols {
                out.data[i * self.cols + j] = f32::NEG_INFINITY;
            }
        }
        out
    }

    /// Columns `[start, end)` as a new matrix.
    pub fn slice_cols(&self, start: usize, end: usize) -> Matrix {
        let mut out = Matrix::new(self.rows, end - start);
        for i in 0..self.rows {
            out.data[i * out.cols..(i + 1) * out.cols].copy_from_slice(&self.row(i)[start..end]);
        }
        out
    }

    /// Zero each element with probability `p` and scale the rest by 1/(1-p).
    /// The scaling keeps the average unchanged, so nothing needs rescaling at
    /// inference time (when dropout is simply switched off).
    pub fn dropout(&self, p: f32, rng: &mut Rng) -> Matrix {
        let data = self
            .data
            .iter()
            .map(|&v| {
                if rng.next_f64() >= p as f64 {
                    v / (1.0 - p)
                } else {
                    0.0
                }
            })
            .collect();
        Matrix {
            rows: self.rows,
            cols: self.cols,
            data,
        }
    }
}

//! Step 6 (book 3.3) - simplified self-attention, no trainable weights.
//!
//! For every token, build a *context vector*: an enriched embedding that mixes
//! in information from all the other tokens, weighted by how relevant each is.
//!
//! ```text
//! scores  = X x X^T                  how similar is token i to token j?   [T, T]
//! weights = softmax(scores, rows)    turn each row into weights summing to 1
//! context = weights x X              weighted sum of all token vectors   [T, d]
//! ```
//!
//! Tip: a dot product measures similarity. Similar vectors -> big score -> big weight.
//! There is nothing to learn here yet; trainable weights arrive in 3.4.

use super::Matrix;

/// `inputs` is [T, d]. Returns (scores, weights, context).
pub fn simple_attention(inputs: &Matrix) -> (Matrix, Matrix, Matrix) {
    let scores = inputs.matmul(&inputs.transpose());
    let weights = scores.softmax_rows();
    let context = weights.matmul(inputs);
    (scores, weights, context)
}

/* The few matrix operations attention needs (book chapter 3).
 *
 * Files in attention/, in the order you learn them:
 *   matrix.[ch]          this file: matmul, transpose, softmax, ...
 *   simple.[ch]          simplified self-attention, no trainable weights (3.3)
 *   self_attention.[ch]  scaled dot-product attention with W_query/key/value (3.4)
 *   causal.[ch]          hide future tokens with a mask, optional dropout (3.5)
 *   multihead.[ch]       several heads in parallel plus an output projection (3.6)
 * The seeded random generator is shared with chapter 2: tokenization/rng.h.
 *
 * Tip: there is no tensor library, so a Matrix is a row-major float array.
 * Memory: every operation returns a NEW matrix allocated from an Arena, a
 * bump allocator. You free the whole arena once; no free() per matrix.
 */
#ifndef MATRIX_H
#define MATRIX_H

#include <stddef.h>

#include "../tokenization/rng.h"

typedef struct {
    unsigned char *base;
    size_t used;
    size_t capacity;
} Arena;

Arena arena_new(size_t capacity);
void arena_free(Arena *a);

typedef struct {
    size_t rows, cols;
    float *data;
} Matrix;

Matrix matrix_new(Arena *a, size_t rows, size_t cols);                      /* zero-filled */
Matrix matrix_from(Arena *a, size_t rows, size_t cols, const float *values); /* copy of values */
Matrix matrix_rand(Arena *a, size_t rows, size_t cols, Rng *rng);           /* uniform [0, 1) */

static inline float *matrix_row(Matrix m, size_t i) { return m.data + i * m.cols; }
static inline float matrix_at(Matrix m, size_t i, size_t j) { return m.data[i * m.cols + j]; }

Matrix matmul(Arena *a, Matrix x, Matrix y);       /* [n,k] x [k,m] -> [n,m] */
Matrix transpose(Arena *a, Matrix m);
Matrix softmax_rows(Arena *a, Matrix m);           /* each row: positive, sums to 1 */
Matrix scale(Arena *a, Matrix m, float s);
Matrix mask_future(Arena *a, Matrix m);            /* above the diagonal -> -inf */
Matrix slice_cols(Arena *a, Matrix m, size_t start, size_t end); /* columns [start, end) */
Matrix dropout(Arena *a, Matrix m, float p, Rng *rng);           /* zero w.p. p, scale rest by 1/(1-p) */

#endif
C-version/attention/matrix.c
#include "matrix.h"

#include <math.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#include "../tokenization/util.h"

Arena arena_new(size_t capacity) { return (Arena){xmalloc(capacity), 0, capacity}; }
void arena_free(Arena *a) {
    free(a->base);
    *a = (Arena){0};
}

static void *arena_alloc(Arena *a, size_t bytes) {
    bytes = (bytes + 15) & ~(size_t)15; /* keep every block 16-byte aligned */
    if (a->used + bytes > a->capacity) {
        fputs("arena out of memory: raise the capacity passed to arena_new\n", stderr);
        exit(1);
    }
    void *p = a->base + a->used;
    a->used += bytes;
    return p;
}

Matrix matrix_new(Arena *a, size_t rows, size_t cols) {
    Matrix m = {rows, cols, arena_alloc(a, rows * cols * sizeof(float))};
    memset(m.data, 0, rows * cols * sizeof(float));
    return m;
}

Matrix matrix_from(Arena *a, size_t rows, size_t cols, const float *values) {
    Matrix m = matrix_new(a, rows, cols);
    memcpy(m.data, values, rows * cols * sizeof(float));
    return m;
}

Matrix matrix_rand(Arena *a, size_t rows, size_t cols, Rng *rng) {
    Matrix m = matrix_new(a, rows, cols);
    for (size_t i = 0; i < rows * cols; i++) m.data[i] = (float)rng_f64(rng);
    return m;
}

Matrix matmul(Arena *a, Matrix x, Matrix y) {
    Matrix out = matrix_new(a, x.rows, y.cols);
    for (size_t i = 0; i < x.rows; i++)
        for (size_t j = 0; j < y.cols; j++) {
            float sum = 0;
            for (size_t k = 0; k < x.cols; k++) sum += matrix_at(x, i, k) * matrix_at(y, k, j);
            out.data[i * out.cols + j] = sum;
        }
    return out;
}

Matrix transpose(Arena *a, Matrix m) {
    Matrix out = matrix_new(a, m.cols, m.rows);
    for (size_t i = 0; i < m.rows; i++)
        for (size_t j = 0; j < m.cols; j++) out.data[j * out.cols + i] = matrix_at(m, i, j);
    return out;
}

/* Tip: subtracting the row maximum first does not change the result but keeps
 * exp() from overflowing. A -inf entry becomes exactly 0. */
Matrix softmax_rows(Arena *a, Matrix m) {
    Matrix out = matrix_new(a, m.rows, m.cols);
    for (size_t i = 0; i < m.rows; i++) {
        const float *row = matrix_row(m, i);
        float max = row[0], sum = 0;
        for (size_t j = 1; j < m.cols; j++)
            if (row[j] > max) max = row[j];
        for (size_t j = 0; j < m.cols; j++) {
            float e = (float)exp((double)(row[j] - max));
            out.data[i * m.cols + j] = e;
            sum += e;
        }
        for (size_t j = 0; j < m.cols; j++) out.data[i * m.cols + j] /= sum;
    }
    return out;
}

Matrix scale(Arena *a, Matrix m, float s) {
    Matrix out = matrix_new(a, m.rows, m.cols);
    for (size_t i = 0; i < m.rows * m.cols; i++) out.data[i] = m.data[i] * s;
    return out;
}

Matrix mask_future(Arena *a, Matrix m) {
    Matrix out = matrix_from(a, m.rows, m.cols, m.data);
    for (size_t i = 0; i < m.rows; i++)
        for (size_t j = i + 1; j < m.cols; j++) out.data[i * m.cols + j] = -INFINITY;
    return out;
}

Matrix slice_cols(Arena *a, Matrix m, size_t start, size_t end) {
    Matrix out = matrix_new(a, m.rows, end - start);
    for (size_t i = 0; i < m.rows; i++) memcpy(matrix_row(out, i), matrix_row(m, i) + start, (end - start) * sizeof(float));
    return out;
}

Matrix dropout(Arena *a, Matrix m, float p, Rng *rng) {
    Matrix out = matrix_new(a, m.rows, m.cols);
    for (size_t i = 0; i < m.rows * m.cols; i++)
        if (rng_f64(rng) >= (double)p) out.data[i] = m.data[i] / (1.0f - p);
    return out;
}

Zig-version/src/attention/matrix.zig
//! The few matrix operations attention needs.
//!
//! Tip: there is no tensor library, so a `Matrix` is a row-major `[]f32`.
//! Everything attention does is these operations: matmul, transpose, softmax.
//! Every operation allocates its result from the allocator you pass in; the
//! demos use an arena, so nothing needs freeing one by one.

const std = @import("std");
const Rng = @import("../tokenization/rng.zig").Rng;

pub const Matrix = struct {
    rows: usize,
    cols: usize,
    data: []f32,

    /// Zero-filled.
    pub fn init(gpa: std.mem.Allocator, rows: usize, cols: usize) !Matrix {
        const data = try gpa.alloc(f32, rows * cols);
        @memset(data, 0);
        return .{ .rows = rows, .cols = cols, .data = data };
    }

    /// A copy of `values`, which holds rows * cols numbers.
    pub fn fromSlice(gpa: std.mem.Allocator, rows: usize, cols: usize, values: []const f32) !Matrix {
        const m = try init(gpa, rows, cols);
        @memcpy(m.data, values);
        return m;
    }

    /// Uniform in [0, 1), like `torch.rand`.
    pub fn rand(gpa: std.mem.Allocator, rows: usize, cols: usize, rng: *Rng) !Matrix {
        const m = try init(gpa, rows, cols);
        for (m.data) |*v| v.* = @floatCast(rng.nextF64());
        return m;
    }

    pub fn at(self: Matrix, i: usize, j: usize) f32 {
        return self.data[i * self.cols + j];
    }

    pub fn row(self: Matrix, i: usize) []f32 {
        return self.data[i * self.cols ..][0..self.cols];
    }

    /// self x other: [n,k] x [k,m] -> [n,m].
    pub fn matmul(self: Matrix, gpa: std.mem.Allocator, other: Matrix) !Matrix {
        const out = try init(gpa, self.rows, other.cols);
        for (0..self.rows) |i| {
            for (0..other.cols) |j| {
                var sum: f32 = 0;
                for (0..self.cols) |k| sum += self.at(i, k) * other.at(k, j);
                out.data[i * out.cols + j] = sum;
            }
        }
        return out;
    }

    pub fn transpose(self: Matrix, gpa: std.mem.Allocator) !Matrix {
        const out = try init(gpa, self.cols, self.rows);
        for (0..self.rows) |i| {
            for (0..self.cols) |j| out.data[j * out.cols + i] = self.at(i, j);
        }
        return out;
    }

    /// Turn every row into weights that are positive and sum to 1.
    ///
    /// Tip: subtracting the row maximum first does not change the result but keeps
    /// `exp` from overflowing. A `-inf` entry becomes exactly 0.
    pub fn softmaxRows(self: Matrix, gpa: std.mem.Allocator) !Matrix {
        const out = try init(gpa, self.rows, self.cols);
        for (0..self.rows) |i| {
            const r = self.row(i);
            var max = r[0];
            for (r[1..]) |v| max = @max(max, v);
            var sum: f32 = 0;
            for (r, 0..) |v, j| {
                const e: f32 = @floatCast(@exp(@as(f64, v - max)));
                out.data[i * out.cols + j] = e;
                sum += e;
            }
            for (0..self.cols) |j| out.data[i * out.cols + j] /= sum;
        }
        return out;
    }

    /// Multiply every element by `s`.
    pub fn scale(self: Matrix, gpa: std.mem.Allocator, s: f32) !Matrix {
        const out = try init(gpa, self.rows, self.cols);
        for (self.data, out.data) |v, *o| o.* = v * s;
        return out;
    }

    /// Set every element above the diagonal to `-inf` (the "future" tokens).
    pub fn maskFuture(self: Matrix, gpa: std.mem.Allocator) !Matrix {
        const out = try fromSlice(gpa, self.rows, self.cols, self.data);
        for (0..self.rows) |i| {
            for (i + 1..self.cols) |j| out.data[i * out.cols + j] = -std.math.inf(f32);
        }
        return out;
    }

    /// Columns `[start, end)` as a new matrix.
    pub fn sliceCols(self: Matrix, gpa: std.mem.Allocator, start: usize, end: usize) !Matrix {
        const out = try init(gpa, self.rows, end - start);
        for (0..self.rows) |i| @memcpy(out.row(i), self.row(i)[start..end]);
        return out;
    }

    /// Zero each element with probability `p` and scale the rest by 1/(1-p).
    /// The scaling keeps the average unchanged, so nothing needs rescaling at
    /// inference time (when dropout is simply switched off).
    pub fn dropout(self: Matrix, gpa: std.mem.Allocator, p: f32, rng: *Rng) !Matrix {
        const out = try init(gpa, self.rows, self.cols);
        for (self.data, out.data) |v, *o| {
            if (rng.nextF64() >= p) o.* = v / (1.0 - p);
        }
        return out;
    }
};

test "softmax rows sum to 1 and -inf becomes 0" {
    const gpa = std.testing.allocator;
    const m = try Matrix.fromSlice(gpa, 1, 3, &.{ 1, 2, -std.math.inf(f32) });
    defer gpa.free(m.data);
    const s = try m.softmaxRows(gpa);
    defer gpa.free(s.data);
    try std.testing.expectApproxEqAbs(@as(f32, 1.0), s.data[0] + s.data[1] + s.data[2], 1e-6);
    try std.testing.expectEqual(@as(f32, 0), s.data[2]);
}

test "matmul and transpose" {

    defer gpa.free(a.data);

//! Step 6 (book 3.3) - simplified self-attention, no trainable weights.
//!
//! For every token, build a *context vector*: an enriched embedding that mixes
//! in information from all the other tokens, weighted by how relevant each is.
//!
//! ```
//! scores  = X x X^T                  how similar is token i to token j?   [T, T]
//! weights = softmax(scores, rows)    turn each row into weights summing to 1
//! context = weights x X              weighted sum of all token vectors   [T, d]
//! ```
//!
//! Tip: a dot product measures similarity. Similar vectors -> big score -> big weight.
//! There is nothing to learn here yet; trainable weights arrive in 3.4.

const std = @import("std");
const Matrix = @import("matrix.zig").Matrix;

pub const Result = struct { scores: Matrix, weights: Matrix, context: Matrix };

/// `inputs` is [T, d].
pub fn simpleAttention(gpa: std.mem.Allocator, inputs: Matrix) !Result {
    const scores = try inputs.matmul(gpa, try inputs.transpose(gpa));
    const weights = try scores.softmaxRows(gpa);
    const context = try weights.matmul(gpa, inputs);
    return .{ .scores = scores, .weights = weights, .context = context };
}
Python uses torch, so there is no matrix file: inputs @ inputs.T is the whole thing. The other four languages show the matrix operations (matmul, transpose, softmax) that torch hides, in a few lines each.
