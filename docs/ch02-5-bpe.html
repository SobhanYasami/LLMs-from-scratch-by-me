2.5: Byte pair encoding

Book section 2.5 · code in Python, Go, Rust, C, Zig

Aim. Replace the word-level tokenizer with the one GPT-2, GPT-3 and the original ChatGPT use: sub-word pieces, so no <|unk|> is ever needed.

We do not implement BPE training here. Python, Go and Rust call tiktoken (OpenAI, written in Rust) with the GPT-2 vocabulary of 50,257 tokens. <|endoftext|> gets the last ID, 50256.

[15496, 11, 466, 345, 588, 8887, 30, 220, 50256, 554, 262, 4252, 18250, 8812, 2114, 286, 262, 20562, 13]
Hello, do you like tea? <|endoftext|> In the sunlit terraces of the palace.

How BPE avoids unknown words

Training starts with every single character as a token, then repeatedly merges the most frequent neighbouring pairs into new tokens (d + e becomes de, then maybe def, and so on) up to a size limit. At encode time, an unfamiliar word is split into the largest known pieces, down to single characters or bytes if needed. Any string can be encoded.

C and Zig have no tiktoken, so they ship a small from-scratch BPE encoder (bpe.c, bpe.zig, shown in their tabs). It reads the GPT-2 vocabulary from assets/gpt2-ranks.txt and repeatedly merges the neighbouring pair with the lowest rank. It matches tiktoken's IDs, including on emoji, tabs and repeated special tokens. Reading it is the best way to see how BPE merges work.
tiktoken refuses special-token text by default, to prevent injection. Python passes allowed_special to opt in. Go and Rust use r50k_base, the same vocabulary as GPT-2 under its tiktoken name.

Code (the BPE demo is demo_bpe / demoBPE in each runner)

python-version/main.py
"""Demo runner for the educational tokenizers and data sampling.

Run from anywhere:

    python-version/.venv/bin/python python-version/main.py          # all chapters
    python-version/.venv/bin/python python-version/main.py ch03     # one chapter

Pipeline:
    raw text -> split_text -> build_vocab -> SimpleTokenizerV1/V2 -> tiktoken (GPT-2 BPE)
    -> sliding-window dataset

Tip: read ``main()`` first, then jump into ``tokenization/`` following each demo.
See python-version/README.md for a walkthrough of each step.
"""

import sys
from pathlib import Path

import tiktoken
import torch

from tokenization import (
    END_OF_TEXT,
    GPTDatasetV1,
    InputEmbedding,
    SimpleTokenizerV1,
    SimpleTokenizerV2,
    build_vocab,
    create_dataloader_v1,
    split_text,
)

# Resolved from this file, so the script works no matter the current directory.
DATA_PATH = Path(__file__).resolve().parent.parent / "assets" / "the-verdict.txt"

V1_TEXT = '"It\'s the last he painted, you know," Mrs. Gisburn said with pardonable pride.'
V2_TEXT = f"Hello, do you like tea? {END_OF_TEXT} In the sunlit terraces of the palace."


def load_text() -> str:
    return DATA_PATH.read_text(encoding="utf-8")


def load_vocab() -> dict[str, int]:
    """Read the corpus, tokenize it, and build the vocabulary."""
    return build_vocab(split_text(load_text()))


def show_vocab_tail(vocab: dict[str, int], count: int = 5) -> None:
    """Print the last ``count`` vocabulary entries (the highest IDs)."""
    for item in list(vocab.items())[-count:]:
        print(item)


def demo_v1(vocab: dict[str, int]) -> None:
    """V1 works only on words it has seen in the corpus."""
    tokenizer = SimpleTokenizerV1(vocab)
    print("version 1 tokenizer :")
    ids = tokenizer.encode(V1_TEXT)
    print(ids)
    print(tokenizer.decode(ids))
    print("-------------------------")


def demo_v2(vocab: dict[str, int]) -> None:
    """V2 maps unknown words to <|unk|> and handles the <|endoftext|> separator."""
    tokenizer = SimpleTokenizerV2(vocab)
    print("version 2 tokenizer :")
    ids = tokenizer.encode(V2_TEXT)
    print(ids)
    print(tokenizer.decode(ids))
    print("---------------------")


def demo_bpe() -> None:
    """Byte pair encoding with OpenAI's tiktoken (GPT-2 vocabulary, ~50k tokens).

    Unlike the simple tokenizers above, BPE learns sub-word pieces, so it never
    needs an <|unk|> token: any string can be broken into known pieces.
    """
    tokenizer = tiktoken.get_encoding("gpt2")
    print("bpe tokenizer :")
    ids = tokenizer.encode(V2_TEXT, allowed_special={END_OF_TEXT})
    print(ids)
    print(tokenizer.decode(ids))
    print("---------------------")


def demo_sampling() -> None:
    """Next-word prediction pairs: the target is the input shifted by one."""
    tokenizer = tiktoken.get_encoding("gpt2")
    # Drop the first 50 tokens: gives a slightly more interesting passage.
    enc_sample = tokenizer.encode(load_text())[50:]

    context_size = 4
    x = enc_sample[:context_size]
    y = enc_sample[1 : context_size + 1]
    print("data sampling :")
    print(f"x: {x}")
    print(f"y:      {y}")

    for i in range(1, context_size + 1):
        context = enc_sample[:i]
        desired = enc_sample[i]
        print(tokenizer.decode(context), "---->", tokenizer.decode([desired]))

    # The same idea, packaged: every window of the whole text, in batches.
    dataset = GPTDatasetV1(load_text(), tokenizer, max_length=4, stride=4)
    print(f"windows: {len(dataset)}, first pair: {dataset[0]}")
    loader = create_dataloader_v1(load_text(), tokenizer, batch_size=8, max_length=4, stride=4, shuffle=False)
    inputs, targets = next(iter(loader))
    print("inputs:\n", inputs)
    print("targets:\n", targets)
    print("---------------------")


def demo_dataloader() -> None:
    """Batches of windows. stride < max_length overlaps windows; stride == max_length does not."""
    tokenizer = tiktoken.get_encoding("gpt2")
    text = load_text()
    print("data loader :")
    loader = create_dataloader_v1(text, tokenizer, batch_size=1, max_length=4, stride=1, shuffle=False)
    it = iter(loader)
    print(next(it))  # first window
    print(next(it))  # second window: shifted by one token
    loader = create_dataloader_v1(text, tokenizer, batch_size=8, max_length=4, stride=4, shuffle=False)
    inputs, targets = next(iter(loader))
    print("inputs shape:", tuple(inputs.shape))
    print("---------------------")


def demo_embeddings() -> None:
    """Token IDs -> vectors, then add position vectors."""
    torch.manual_seed(123)  # embedding weights start random; fix them for repeatable output
    tokenizer = tiktoken.get_encoding("gpt2")
    max_length, output_dim = 4, 256
    loader = create_dataloader_v1(
        load_text(), tokenizer, batch_size=8, max_length=max_length, stride=max_length, shuffle=False
    )
    inputs, _ = next(iter(loader))

    embedding = InputEmbedding(tokenizer.n_vocab, max_length, output_dim)
    token_embeddings = embedding.token_embedding(inputs)
    pos_embeddings = embedding.pos_embedding(torch.arange(max_length))
    input_embeddings = embedding(inputs)
    print("embeddings :")
    # Shapes are the thing to notice: [batch, tokens] grows a new axis of size D.
    print("token ids shape:     ", tuple(inputs.shape))
    print("token embeddings:    ", tuple(token_embeddings.shape))
    print("position embeddings: ", tuple(pos_embeddings.shape), " <- one vector per position, shared by all rows")
    print("input embeddings:    ", tuple(input_embeddings.shape), " <- token + position")
    print("---------------------")


def run_chapter2() -> None:
    vocab = load_vocab()
    show_vocab_tail(vocab)
    demo_v1(vocab)
    demo_v2(vocab)
    demo_bpe()
    demo_sampling()
    demo_dataloader()
    demo_embeddings()


def main() -> None:
    """``main.py [ch02|ch03]``: run one chapter's demos, or both when no argument is given."""
    chapters = sys.argv[1:] or ["ch02", "ch03"]
    for chapter in chapters:
        if chapter == "ch02":
            run_chapter2()
        elif chapter == "ch03":
            import ch03

            ch03.run()
        else:
            sys.exit(f"unknown chapter {chapter!r}; use ch02 or ch03")


if __name__ == "__main__":
    main()

// Demo runner for the educational tokenizers and data sampling.
//
// Run from go-version/:
//
//	go run ./cmd         # all chapters
//	go run ./cmd ch03    # one chapter
//
// Tip: read main() first, then jump into tokenization/ following each demo.
//
// Pipeline mirrors python-version/main.py:
//
//	raw text -> SplitText -> BuildVocab -> SimpleTokenizerV1/V2 -> tiktoken (GPT-2 BPE)
//	-> sliding-window dataset
package main

import (
	"fmt"
	"os"
	"sort"
	"strings"

	"github.com/pkoukk/tiktoken-go"
	tiktoken_loader "github.com/pkoukk/tiktoken-go-loader"

	"llm-from-scratch/tokenization"
)

// Run from go-version/, like python-version/main.py resolves paths from its own dir.
const dataPath = "../assets/the-verdict.txt"

const v1Text = `"It's the last he painted, you know," Mrs. Gisburn said with pardonable pride.`

var v2Text = "Hello, do you like tea? " + tokenization.EndOfText + " In the sunlit terraces of the palace."

func check(err error) {
	if err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(1)
	}
}

func formatIDs(ids []int) string {
	parts := make([]string, len(ids))
	for i, id := range ids {
		parts[i] = fmt.Sprint(id)
	}
	return "[" + strings.Join(parts, ", ") + "]"
}

func loadText() string {
	raw, err := os.ReadFile(dataPath)
	check(err)
	return string(raw)
}

func loadVocab() map[string]int {
	return tokenization.BuildVocab(tokenization.SplitText(loadText()))
}

// showVocabTail prints the last count vocab entries (the highest IDs).
func showVocabTail(vocab map[string]int, count int) {
	type entry struct {
		token string
		id    int
	}
	entries := make([]entry, 0, len(vocab))
	for t, id := range vocab {
		entries = append(entries, entry{t, id})
	}
	sort.Slice(entries, func(i, j int) bool { return entries[i].id < entries[j].id })
	for _, e := range entries[len(entries)-count:] {
		fmt.Printf("('%s', %d)\n", e.token, e.id)
	}
}

// demoV1: V1 works only on words it has seen in the corpus.
func demoV1(vocab map[string]int) {
	t := tokenization.NewSimpleTokenizerV1(vocab)
	fmt.Println("version 1 tokenizer :")
	ids, err := t.Encode(v1Text)
	check(err)
	fmt.Println(formatIDs(ids))
	decoded, err := t.Decode(ids)
	check(err)
	fmt.Println(decoded)
	fmt.Println("-------------------------")
}

// demoV2: V2 maps unknown words to <|unk|> and handles <|endoftext|>.
func demoV2(vocab map[string]int) {
	t := tokenization.NewSimpleTokenizerV2(vocab)
	fmt.Println("version 2 tokenizer :")
	ids, err := t.Encode(v2Text)
	check(err)
	fmt.Println(formatIDs(ids))
	decoded, err := t.Decode(ids)
	check(err)
	fmt.Println(decoded)
	fmt.Println("---------------------")
}

func newBPE() *tiktoken.Tiktoken {
	// Offline loader: embeds the BPE files instead of downloading them.
	tiktoken.SetBpeLoader(tiktoken_loader.NewOfflineLoader())
	enc, err := tiktoken.GetEncoding("r50k_base") // same vocab as GPT-2
	check(err)
	return enc
}

// demoBPE: byte pair encoding, GPT-2 vocabulary (~50k tokens). No <|unk|> needed.
func demoBPE() {
	enc := newBPE()
	fmt.Println("bpe tokenizer :")
	ids := enc.Encode(v2Text, []string{tokenization.EndOfText}, nil)
	fmt.Println(formatIDs(ids))
	fmt.Println(enc.Decode(ids))
	fmt.Println("---------------------")
}

// demoSampling: next-word prediction pairs, target is the input shifted by one.
func demoSampling() {
	enc := newBPE()
	text := loadText()
	// Drop the first 50 tokens: gives a slightly more interesting passage.
	sample := enc.Encode(text, nil, nil)[50:]

	const contextSize = 4
	fmt.Println("data sampling :")
	fmt.Println("x:", formatIDs(sample[:contextSize]))
	fmt.Println("y:     ", formatIDs(sample[1:contextSize+1]))
	for i := 1; i <= contextSize; i++ {
		fmt.Println(enc.Decode(sample[:i]), "---->", enc.Decode([]int{sample[i]}))
	}

	// The same idea, packaged: every window of the whole text, in batches.
	ds := tokenization.NewGPTDataset(enc.Encode(text, nil, nil), 4, 4)
	in, tg := ds.Get(0)
	fmt.Println("windows:", ds.Len(), "first pair:", formatIDs(in), formatIDs(tg))
	inputs, targets := ds.Batches(8)
	fmt.Println("first batch inputs:")
	for _, row := range inputs[0] {
		fmt.Println(" ", formatIDs(row))
	}
	fmt.Println("first batch targets:")
	for _, row := range targets[0] {
		fmt.Println(" ", formatIDs(row))
	}
	fmt.Println("---------------------")
}

// demoDataLoader: batches of windows. stride < maxLength overlaps windows.
func demoDataLoader() {
	enc := newBPE()
	ids := enc.Encode(loadText(), nil, nil)
	fmt.Println("data loader :")
	b := tokenization.NewDataLoader(tokenization.NewGPTDataset(ids, 4, 1), 1, false, true, 123).Epoch()
	fmt.Println(formatIDs(b[0].Inputs[0]), formatIDs(b[0].Targets[0])) // first window
	fmt.Println(formatIDs(b[1].Inputs[0]), formatIDs(b[1].Targets[0])) // second: shifted by one
	batches := tokenization.NewDataLoader(tokenization.NewGPTDataset(ids, 4, 4), 8, false, true, 123).Epoch()
	fmt.Printf("inputs shape: (%d, %d)\n", len(batches[0].Inputs), len(batches[0].Inputs[0]))
	fmt.Println("---------------------")
}

// demoEmbeddings: token IDs -> vectors, then add position vectors.
func demoEmbeddings() {
	enc := newBPE()
	const maxLength, outputDim = 4, 256
	ds := tokenization.NewGPTDataset(enc.Encode(loadText(), nil, nil), maxLength, maxLength)
	inputs := tokenization.NewDataLoader(ds, 8, false, true, 123).Epoch()[0].Inputs

	emb := tokenization.NewInputEmbedding(50257, maxLength, outputDim, 123)
	tokenEmb := make([][][]float32, len(inputs))
	for i, row := range inputs {
		tokenEmb[i] = emb.Token.Forward(row)
	}
	input := emb.Forward(inputs)
	fmt.Println("embeddings :")
	// Shapes are the thing to notice: [batch, tokens] grows a new axis of size D.
	fmt.Printf("token ids shape:      (%d, %d)\n", len(inputs), len(inputs[0]))
	fmt.Printf("token embeddings:     (%d, %d, %d)\n", len(tokenEmb), len(tokenEmb[0]), len(tokenEmb[0][0]))
	fmt.Printf("position embeddings:  (%d, %d)  <- one vector per position, shared by all rows\n", len(emb.Pos.Weight), len(emb.Pos.Weight[0]))
	fmt.Printf("input embeddings:     (%d, %d, %d)  <- token + position\n", len(input), len(input[0]), len(input[0][0]))
	fmt.Println("---------------------")
}

func runChapter2() {
	vocab := loadVocab()
	showVocabTail(vocab, 5)
	demoV1(vocab)
	demoV2(vocab)
	demoBPE()
	demoSampling()
	demoDataLoader()
	demoEmbeddings()
}

func main() {
	chapters := os.Args[1:]
	if len(chapters) == 0 {
		chapters = []string{"ch02", "ch03"}
	}
	for _, c := range chapters {
		switch c {
		case "ch02":
			runChapter2()
		case "ch03":
			runChapter3()
		default:
			fmt.Fprintf(os.Stderr, "unknown chapter %q; use ch02 or ch03\n", c)

		}
	}

//! Demo runner for the educational tokenizers and data sampling.
//!
//! Run from anywhere: `cargo run --manifest-path rust-version/Cargo.toml`
//!
//! Tip: read `main()` first, then jump into `tokenization/` following each demo.
//!
//! Pipeline mirrors python-version/main.py:
//!   raw text -> split_text -> build_vocab -> SimpleTokenizerV1/V2 -> tiktoken (GPT-2 BPE)
//!   -> sliding-window dataset

use std::collections::HashMap;
use std::fs;
use std::path::Path;

mod ch03;

use llm_from_scratch::tokenization::{
    DataLoader, END_OF_TEXT, GptDataset, InputEmbedding, SimpleTokenizerV1, SimpleTokenizerV2,
    build_vocab, split_text,
};
use tiktoken_rs::CoreBPE;

const V1_TEXT: &str =
    r#""It's the last he painted, you know," Mrs. Gisburn said with pardonable pride."#;

fn v2_text() -> String {
    format!("Hello, do you like tea? {END_OF_TEXT} In the sunlit terraces of the palace.")
}

/// Resolved from the crate dir at compile time, so the cwd does not matter.
fn load_text() -> String {
    let path = Path::new(env!("CARGO_MANIFEST_DIR")).join("../assets/the-verdict.txt");
    fs::read_to_string(path).expect("read the-verdict.txt")
}

fn load_vocab() -> HashMap<String, usize> {
    build_vocab(&split_text(&load_text()))
}

/// Print the last `count` vocabulary entries (the highest IDs).
fn show_vocab_tail(vocab: &HashMap<String, usize>, count: usize) {
    let mut entries: Vec<_> = vocab.iter().collect();
    entries.sort_by_key(|(_, id)| **id);
    for (token, id) in &entries[entries.len() - count..] {
        println!("('{token}', {id})");
    }
}

/// V1 works only on words it has seen in the corpus.
fn demo_v1(vocab: &HashMap<String, usize>) {
    let tokenizer = SimpleTokenizerV1::new(vocab.clone());
    println!("version 1 tokenizer :");
    let ids = tokenizer.encode(V1_TEXT).unwrap();
    println!("{ids:?}");
    println!("{}", tokenizer.decode(&ids).unwrap());
    println!("-------------------------");
}

/// V2 maps unknown words to <|unk|> and handles the <|endoftext|> separator.
fn demo_v2(vocab: &HashMap<String, usize>) {
    let tokenizer = SimpleTokenizerV2::new(vocab.clone());
    println!("version 2 tokenizer :");
    let ids = tokenizer.encode(&v2_text()).unwrap();
    println!("{ids:?}");
    println!("{}", tokenizer.decode(&ids).unwrap());
    println!("---------------------");
}

/// GPT-2 vocabulary (~50k tokens); tiktoken calls it r50k_base.
fn bpe() -> CoreBPE {
    tiktoken_rs::r50k_base().expect("load r50k_base")
}

/// Byte pair encoding: sub-word pieces, so no <|unk|> is ever needed.
fn demo_bpe() {
    let tokenizer = bpe();
    println!("bpe tokenizer :");
    let ids = tokenizer.encode_with_special_tokens(&v2_text());
    println!("{ids:?}");
    println!("{}", tokenizer.decode(&ids).unwrap());
    println!("---------------------");
}

/// Next-word prediction pairs: the target is the input shifted by one.
fn demo_sampling() {
    let tokenizer = bpe();
    let text = load_text();
    let all_ids = tokenizer.encode_ordinary(&text);
    // Drop the first 50 tokens: gives a slightly more interesting passage.
    let sample = &all_ids[50..];

    let context_size = 4;
    println!("data sampling :");
    println!("x: {:?}", &sample[..context_size]);
    println!("y:      {:?}", &sample[1..=context_size]);
    for i in 1..=context_size {
        let context = tokenizer.decode(&sample[..i]).unwrap();
        let desired = tokenizer.decode(&sample[i..=i]).unwrap();
        println!("{context} ----> {desired}");
    }

    // The same idea, packaged: every window of the whole text, in batches.
    let dataset = GptDataset::new(&all_ids, 4, 4);
    println!(
        "windows: {}, first pair: {:?}",
        dataset.len(),
        dataset.get(0)
    );
    let (inputs, targets) = dataset.batches(8).next().unwrap();
    println!("first batch inputs:");
    for row in inputs {
        println!("  {row:?}");
    }
    println!("first batch targets:");
    for row in targets {
        println!("  {row:?}");
    }
    println!("---------------------");
}

/// Batches of windows. stride < max_length overlaps windows.
fn demo_dataloader() {
    let ids = bpe().encode_ordinary(&load_text());
    println!("data loader :");
    let overlapping = GptDataset::new(&ids, 4, 1);
    let b = DataLoader::new(&overlapping, 1, false, true, 123).epoch();
    println!("{:?} {:?}", b[0].inputs[0], b[0].targets[0]); // first window
    println!("{:?} {:?}", b[1].inputs[0], b[1].targets[0]); // second: shifted by one
    let dataset = GptDataset::new(&ids, 4, 4);
    let batches = DataLoader::new(&dataset, 8, false, true, 123).epoch();
    println!(
        "inputs shape: ({}, {})",
        batches[0].inputs.len(),
        batches[0].inputs[0].len()
    );
    println!("---------------------");
}

/// Token IDs -> vectors, then add position vectors.
fn demo_embeddings() {
    let tokenizer = bpe();
    let (max_length, output_dim) = (4, 256);
    let ids = tokenizer.encode_ordinary(&load_text());
    let dataset = GptDataset::new(&ids, max_length, max_length);
    let inputs = DataLoader::new(&dataset, 8, false, true, 123)
        .epoch()
        .remove(0)
        .inputs;

    let emb = InputEmbedding::new(50257, max_length, output_dim, 123);
    let token_embeddings: Vec<_> = inputs.iter().map(|row| emb.token.forward(row)).collect();
    let input_embeddings = emb.forward(&inputs);
    println!("embeddings :");
    // Shapes are the thing to notice: [batch, tokens] grows a new axis of size D.
    println!(
        "token ids shape:      ({}, {})",
        inputs.len(),
        inputs[0].len()
    );
    println!(
        "token embeddings:     ({}, {}, {})",
        token_embeddings.len(),
        token_embeddings[0].len(),
        token_embeddings[0][0].len()
    );
    println!(
        "position embeddings:  ({}, {})  <- one vector per position, shared by all rows",
        emb.pos.weight.len(),
        emb.pos.weight[0].len()
    );
    println!(
        "input embeddings:     ({}, {}, {})",
        input_embeddings.len(),
        input_embeddings[0].len(),
        input_embeddings[0][0].len()
    );
    println!("---------------------");
}

fn run_chapter2() {
    let vocab = load_vocab();
    show_vocab_tail(&vocab, 5);
    demo_v1(&vocab);
    demo_v2(&vocab);
    demo_bpe();
    demo_sampling();
    demo_dataloader();
    demo_embeddings();
}

/// `cargo run -- ch03` runs one chapter's demos; with no argument, all of them.
fn main() {
    let mut chapters: Vec<String> = std::env::args().skip(1).collect();
    if chapters.is_empty() {
        chapters = vec!["ch02".into(), "ch03".into()];
    }
    for chapter in &chapters {
        match chapter.as_str() {
            "ch02" => run_chapter2(),
            "ch03" => ch03::run(),
            other => {
                eprintln!("unknown chapter {other:?}; use ch02 or ch03");
                std::process::exit(1);
            }
        }
    }
}

/* Demo runner for the educational tokenizers and data sampling.
 *
 * Run from C-version/:
 *
 *     make run              # all chapters
 *     make run ARGS=ch03    # one chapter
 *
 * Tip: read main() first, then jump into tokenization/ following each demo.
 *
 * Pipeline mirrors python-version/main.py:
 *     raw text -> split_text -> build_vocab -> SimpleTokenizer V1/V2 -> BPE (GPT-2)
 *     -> sliding-window dataset -> data loader -> embeddings
 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#include "ch03.h"
#include "tokenization/bpe.h"
#include "tokenization/dataloader.h"

#define V1_TEXT "\"It's the last he painted, you know,\" Mrs. Gisburn said with pardonable pride."

static void die(const char *msg) {
    fprintf(stderr, "%s\n", msg);
    exit(1);
}

static char *load_text(void) {
    FILE *f = fopen(DATA_PATH, "rb");
    if (!f) die("cannot open " DATA_PATH " (run from C-version/)");
    fseek(f, 0, SEEK_END);
    long size = ftell(f);
    rewind(f);
    char *text = xmalloc((size_t)size + 1);
    size_t got = fread(text, 1, (size_t)size, f);
    text[got] = '\0';
    fclose(f);
    return text;
}

static void print_ids_inline(const int *ids, size_t n) {
    printf("[");
    for (size_t i = 0; i < n; i++) printf(i ? ", %d" : "%d", ids[i]);
    printf("]");
}

static void print_ids(const int *ids, size_t n) {
    print_ids_inline(ids, n);
    printf("\n");
}

static Vocab load_vocab(const char *text) {
    Tokens tokens = split_text(text);
    Vocab v = build_vocab(&tokens);
    tokens_free(&tokens);
    return v;
}

/* Print the last `count` vocabulary entries (the highest IDs). */
static void show_vocab_tail(const Vocab *v, size_t count) {
    for (size_t id = v->size - count; id < v->size; id++) printf("('%s', %zu)\n", v->tokens[id], id);
}

/* encode, print IDs, decode, print text. */
static void roundtrip(const SimpleTokenizer *t, const char *text) {
    char err[ERR_LEN];
    size_t n;
    int *ids = simple_encode(t, text, &n, err);
    if (!ids) die(err);
    print_ids(ids, n);
    char *decoded = simple_decode(t, ids, n, err);
    if (!decoded) die(err);
    printf("%s\n", decoded);
    free(decoded);
    free(ids);
}

/* Next-word prediction pairs: the target is the input shifted by one. */
static void demo_sampling(const Bpe *bpe, const char *text) {
    size_t n;
    int *all = bpe_encode(bpe, text, &n);
    /* Drop the first 50 tokens: gives a slightly more interesting passage. */
    const int *sample = all + 50;
    const size_t context_size = 4;

    printf("data sampling :\n");
    printf("x: ");
    print_ids(sample, context_size);
    printf("y:      ");
    print_ids(sample + 1, context_size);
    for (size_t i = 1; i <= context_size; i++) {
        char *context = bpe_decode(bpe, sample, i);
        char *desired = bpe_decode(bpe, sample + i, 1);
        printf("%s ----> %s\n", context, desired);
        free(context);
        free(desired);
    }

    /* The same idea, packaged: every window of the whole text. */
    GPTDataset ds = gpt_dataset_new(all, n, 4, 4);
    const int *in, *tg;
    gpt_dataset_get(&ds, 0, &in, &tg);
    printf("windows: %zu first pair: ", ds.count);
    print_ids_inline(in, 4);
    printf(" ");
    print_ids(tg, 4);
    /* the same windows, batched */
    DataLoader first = dataloader_new(&ds, 8, false, true, 123);
    dataloader_start_epoch(&first);
    int bi[8 * 4], bt[8 * 4];
    dataloader_batch(&first, 0, bi, bt);
    printf("first batch inputs:\n");
    for (size_t r = 0; r < 8; r++) {
        printf("  ");
        print_ids(bi + r * 4, 4);
    }
    printf("first batch targets:\n");
    for (size_t r = 0; r < 8; r++) {
        printf("  ");
        print_ids(bt + r * 4, 4);
    }
    dataloader_free(&first);
    printf("---------------------\n");
    free(all);
}

/* Batches of windows. stride < max_length overlaps windows. */
static void demo_dataloader(const Bpe *bpe, const char *text) {
    size_t n;
    int *ids = bpe_encode(bpe, text, &n);
    int in[8 * 4], tg[8 * 4];
    printf("data loader :\n");

    GPTDataset overlap = gpt_dataset_new(ids, n, 4, 1);
    DataLoader dl = dataloader_new(&overlap, 1, false, true, 123);
    dataloader_start_epoch(&dl);
    for (size_t b = 0; b < 2; b++) { /* first window, then second: shifted by one */
        dataloader_batch(&dl, b, in, tg);
        print_ids_inline(in, 4);
        printf(" ");
        print_ids(tg, 4);
    }
    dataloader_free(&dl);

    GPTDataset ds = gpt_dataset_new(ids, n, 4, 4);
    dl = dataloader_new(&ds, 8, false, true, 123);
    dataloader_start_epoch(&dl);
    size_t rows = dataloader_batch(&dl, 0, in, tg);
    printf("inputs shape: (%zu, 4)\n", rows);
    dataloader_free(&dl);
    printf("---------------------\n");
    free(ids);
}

/* Token IDs -> vectors, then add position vectors. */
static void demo_embeddings(const Bpe *bpe, const char *text) {
    enum { MAX_LENGTH = 4, OUTPUT_DIM = 256, BATCH = 8 };
    size_t n;
    int *ids = bpe_encode(bpe, text, &n);
    GPTDataset ds = gpt_dataset_new(ids, n, MAX_LENGTH, MAX_LENGTH);
    DataLoader dl = dataloader_new(&ds, BATCH, false, true, 123);
    dataloader_start_epoch(&dl);
    int inputs[BATCH * MAX_LENGTH], targets[BATCH * MAX_LENGTH];
    dataloader_batch(&dl, 0, inputs, targets);

    InputEmbedding emb = input_embedding_new(BPE_ENDOFTEXT_ID + 1, MAX_LENGTH, OUTPUT_DIM, 123);
    float *token_emb = xmalloc(BATCH * MAX_LENGTH * OUTPUT_DIM * sizeof(float));
    float *input_emb = xmalloc(BATCH * MAX_LENGTH * OUTPUT_DIM * sizeof(float));
    embedding_forward(&emb.token, inputs, BATCH * MAX_LENGTH, token_emb); /* token vectors only */
    input_embedding_forward(&emb, inputs, BATCH, MAX_LENGTH, input_emb);   /* token + position */

    /* Shapes are the thing to notice: [batch, tokens] grows a new axis of size D. */
    printf("embeddings :\n");
    printf("token ids shape:      (%d, %d)\n", BATCH, MAX_LENGTH);
    printf("token embeddings:     (%d, %d, %d)\n", BATCH, MAX_LENGTH, OUTPUT_DIM);
    printf("position embeddings:  (%zu, %zu)  <- one vector per position, shared by all rows\n", emb.pos.rows,
           emb.pos.dim);
    printf("input embeddings:     (%d, %d, %d)  <- token + position\n", BATCH, MAX_LENGTH, OUTPUT_DIM);
    printf("---------------------\n");

    free(token_emb);
    free(input_emb);
    input_embedding_free(&emb);
    dataloader_free(&dl);
    free(ids);
}

C-version/tokenization/bpe.h
/* Bonus (book 2.5) - byte pair encoding, the GPT-2 tokenizer.
 *
 * C has no tiktoken, so this is a small from-scratch encoder that reads the
 * GPT-2 vocabulary from assets/gpt2-ranks.txt (one line per token: the token's
 * bytes in hex, then its rank). It gives the same IDs as tiktoken's "gpt2".
 *
 * How it works, for one piece of text:
 *   1. start with one part per byte
 *   2. find the neighbouring pair whose joined bytes have the LOWEST rank
 *   3. merge it, repeat until no pair is a known token
 * Low rank = merged early in training = most frequent. Any byte string can
 * be encoded, so no <|unk|> is needed.
 */
#ifndef BPE_H
#define BPE_H

#include <stddef.h>

#include "util.h"

#define BPE_ENDOFTEXT_ID 50256

typedef struct {
    unsigned char **bytes; /* rank -> token bytes */
    size_t *lens;
    size_t size; /* regular tokens, ranks 0..size-1 */
    int *slots;  /* hash table: bytes -> rank (-1 = empty) */
    size_t n_slots;
} Bpe;

/* Load the ranks file. Returns 0 on success, -1 and fills `err` on failure. */
int bpe_load(Bpe *b, const char *path, char err[ERR_LEN]);
void bpe_free(Bpe *b);

/* Text -> token IDs. "<|endoftext|>" in the text becomes ID 50256. Caller frees. */
int *bpe_encode(const Bpe *b, const char *text, size_t *n);
/* IDs -> text. Caller frees. */
char *bpe_decode(const Bpe *b, const int *ids, size_t n);

#endif
C-version/tokenization/bpe.c
#include "bpe.h"

#include <ctype.h>
#include <stdbool.h>
#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#include "preprocess.h"

/* ---- loading -------------------------------------------------------- */

static uint64_t hash_bytes(const unsigned char *s, size_t n) { /* FNV-1a */
    uint64_t h = 1469598103934665603ULL;
    for (size_t i = 0; i < n; i++) h = (h ^ s[i]) * 1099511628211ULL;
    return h;
}

static int hex_val(int c) { return isdigit(c) ? c - '0' : c - 'a' + 10; }

static int lookup(const Bpe *b, const unsigned char *s, size_t n) {
    size_t slot = hash_bytes(s, n) & (b->n_slots - 1);
    while (b->slots[slot] != -1) {
        int r = b->slots[slot];
        if (b->lens[r] == n && memcmp(b->bytes[r], s, n) == 0) return r;
        slot = (slot + 1) & (b->n_slots - 1);
    }
    return -1;
}

int bpe_load(Bpe *b, const char *path, char err[ERR_LEN]) {
    FILE *f = fopen(path, "r");
    if (!f) {
        snprintf(err, ERR_LEN, "cannot open %s", path);
        return -1;
    }
    size_t cap = 51000;
    *b = (Bpe){0};
    b->bytes = xmalloc(cap * sizeof(char *));
    b->lens = xmalloc(cap * sizeof(size_t));
    char line[512];
    while (fgets(line, sizeof line, f)) {
        char *space = strchr(line, ' ');
        if (!space) continue;
        size_t hex_len = (size_t)(space - line);
        unsigned char *bytes = xmalloc(hex_len / 2 + 1);
        for (size_t i = 0; i < hex_len / 2; i++)
            bytes[i] = (unsigned char)(hex_val(line[2 * i]) * 16 + hex_val(line[2 * i + 1]));
        size_t rank = (size_t)atol(space + 1); /* ranks are 0,1,2,... in file order */
        if (rank >= cap) break;
        b->bytes[rank] = bytes;
        b->lens[rank] = hex_len / 2;
        if (rank + 1 > b->size) b->size = rank + 1;
    }
    fclose(f);

    b->n_slots = 1;
    while (b->n_slots < b->size * 2) b->n_slots <<= 1;
    b->slots = xmalloc(b->n_slots * sizeof(int));
    for (size_t i = 0; i < b->n_slots; i++) b->slots[i] = -1;
    for (size_t r = 0; r < b->size; r++) {
        size_t slot = hash_bytes(b->bytes[r], b->lens[r]) & (b->n_slots - 1);
        while (b->slots[slot] != -1) slot = (slot + 1) & (b->n_slots - 1);
        b->slots[slot] = (int)r;
    }
    return 0;
}

void bpe_free(Bpe *b) {
    for (size_t i = 0; i < b->size; i++) free(b->bytes[i]);
    free(b->bytes);
    free(b->lens);
    free(b->slots);
    *b = (Bpe){0};
}

/* ---- the merge loop ------------------------------------------------- */

/* Encode one piece (a few bytes) by repeatedly merging the best neighbour pair. */
static void encode_piece(const Bpe *b, const unsigned char *s, size_t n, int **out, size_t *len, size_t *cap) {
    /* bounds[i]..bounds[i+1] is part i; begin with one part per byte */
    size_t *bounds = xmalloc((n + 1) * sizeof(size_t));
    size_t parts = n;
    for (size_t i = 0; i <= n; i++) bounds[i] = i;

    for (;;) {
        int best_rank = -1;
        size_t best = 0;
        for (size_t i = 0; i + 1 < parts; i++) { /* rank of parts i and i+1 joined */
            int r = lookup(b, s + bounds[i], bounds[i + 2] - bounds[i]);
            if (r >= 0 && (best_rank < 0 || r < best_rank)) best_rank = r, best = i;
        }
        if (best_rank < 0) break;
        for (size_t i = best + 1; i < parts; i++) bounds[i] = bounds[i + 1]; /* merge */
        parts--;
    }
    for (size_t i = 0; i < parts; i++) {
        if (*len == *cap) {
            *cap *= 2;
            *out = xrealloc(*out, *cap * sizeof(int));
        }
        (*out)[(*len)++] = lookup(b, s + bounds[i], bounds[i + 1] - bounds[i]);
    }
    free(bounds);
}

/* ---- splitting text into pieces (GPT-2's regex, written by hand) ----- */

/* The GPT-2 pattern, in words:
 *   'm 'd 's 't 're 've 'll | optional space + letters | optional space + digits
 *   | optional space + other symbols | whitespace
 * Bytes >= 0x80 (UTF-8) are treated as letters, which is right for most text. */
static bool is_letter(unsigned char c) { return isalpha(c) || c >= 0x80; }
static bool is_digit(unsigned char c) { return isdigit(c); }
static bool is_space(unsigned char c) { return c < 0x80 && isspace(c); }
static bool is_other(unsigned char c) { return !is_space(c) && !is_letter(c) && !is_digit(c); }

static size_t consume(const unsigned char *s, size_t n, size_t i, bool (*pred)(unsigned char)) {
    while (i < n && pred(s[i])) i++;
    return i;
}

/* End of the piece starting at i. */
static size_t next_piece(const unsigned char *s, size_t n, size_t i) {
    if (s[i] == '\'' && i + 1 < n) {
        if (strchr("sdmt", s[i + 1])) return i + 2;
        if (i + 2 < n && ((s[i + 1] == 'l' && s[i + 2] == 'l') || (s[i + 1] == 'v' && s[i + 2] == 'e') ||
                          (s[i + 1] == 'r' && s[i + 2] == 'e')))
            return i + 3;
    }
    size_t k = (s[i] == ' ') ? i + 1 : i; /* optional single leading space */
    if (k < n) {
        if (is_letter(s[k])) return consume(s, n, k, is_letter);
        if (is_digit(s[k])) return consume(s, n, k, is_digit);
        if (is_other(s[k])) return consume(s, n, k, is_other);
    }
    /* whitespace run: leave its last char for the next piece, unless it is the end */
    size_t e = consume(s, n, i, is_space);
    return (e < n && e - i > 1) ? e - 1 : e;
}

int *bpe_encode(const Bpe *b, const char *text, size_t *n_out) {
    const unsigned char *s = (const unsigned char *)text;
    size_t n = strlen(text), len = 0, cap = 1024;
    int *out = xmalloc(cap * sizeof(int));
    const size_t special_len = strlen(END_OF_TEXT);

    size_t i = 0;
    while (i < n) {
        if (strncmp(text + i, END_OF_TEXT, special_len) == 0) { /* special token */
            if (len == cap) out = xrealloc(out, (cap *= 2) * sizeof(int));
            out[len++] = BPE_ENDOFTEXT_ID;
            i += special_len;
            continue;
        }
        size_t end = next_piece(s, n, i);
        /* never let a piece swallow the start of a special token */
        for (size_t j = i + 1; j < end; j++)
            if (strncmp(text + j, END_OF_TEXT, special_len) == 0) {
                end = j;
                break;
            }
        encode_piece(b, s + i, end - i, &out, &len, &cap);
        i = end;
    }
    *n_out = len;
    return out;
}

char *bpe_decode(const Bpe *b, const int *ids, size_t n) {
    size_t total = 1;
    for (size_t i = 0; i < n; i++)
        total += ids[i] == BPE_ENDOFTEXT_ID ? strlen(END_OF_TEXT) : b->lens[ids[i]];
    char *out = xmalloc(total), *p = out;
    for (size_t i = 0; i < n; i++) {
        if (ids[i] == BPE_ENDOFTEXT_ID) {
            memcpy(p, END_OF_TEXT, strlen(END_OF_TEXT));
            p += strlen(END_OF_TEXT);
        } else {
            memcpy(p, b->bytes[ids[i]], b->lens[ids[i]]);
            p += b->lens[ids[i]];
        }
    }
    *p = '\0';
    return out;
}

const Out = *std.Io.Writer;

fn printIdsInline(out: Out, ids: []const u32) !void {
    try out.writeAll("[");
    for (ids, 0..) |id, i| {
        if (i > 0) try out.writeAll(", ");
        try out.print("{d}", .{id});
    }
    try out.writeAll("]");
}

fn printIds(out: Out, ids: []const u32) !void {
    try printIdsInline(out, ids);
    try out.writeAll("\n");
}

/// Print the last `count` vocabulary entries (the highest IDs).
fn showVocabTail(out: Out, vocab: tk.Vocab, count: usize) !void {
    for (vocab.size() - count..vocab.size()) |id| try out.print("('{s}', {d})\n", .{ vocab.tokens[id], id });
}

/// encode, print IDs, decode, print text.
fn roundtrip(gpa: std.mem.Allocator, out: Out, t: tk.SimpleTokenizer, text: []const u8) !void {
    const ids = try t.encode(gpa, text);
    try printIds(out, ids);
    const decoded = try t.decode(gpa, ids);
    try out.print("{s}\n", .{decoded});
}

    var inputs: [8 * 4]u32 = undefined;

    _ = dl.batch(0, &inputs, &targets);
    try out.writeAll("first batch inputs:\n");

    for (0..8) |r| {

        try printIds(out, targets[r * 4 ..][0..4]);

/// Token IDs -> vectors, then add position vectors.
fn demoEmbeddings(gpa: std.mem.Allocator, out: Out, bpe: tk.Bpe, text: []const u8) !void {
    const max_length = 4;
    const output_dim = 256;
    const batch = 8;
    const ids = try bpe.encode(gpa, text);
    const ds = tk.GptDataset.init(ids, max_length, max_length);
    var dl = try tk.DataLoader.init(gpa, &ds, batch, false, true, 123);
    _ = dl.startEpoch();
    var inputs: [batch * max_length]u32 = undefined;
    var targets: [batch * max_length]u32 = undefined;
    _ = dl.batch(0, &inputs, &targets);

    const emb = try tk.InputEmbedding.init(gpa, tk.bpe.ENDOFTEXT_ID + 1, max_length, output_dim, 123);
    const token_emb = try gpa.alloc(f32, batch * max_length * output_dim);
    const input_emb = try gpa.alloc(f32, batch * max_length * output_dim);
    emb.token.forward(&inputs, token_emb); // token vectors only
    emb.forward(&inputs, batch, max_length, input_emb); // token + position

    // Shapes are the thing to notice: [batch, tokens] grows a new axis of size D.
    try out.writeAll("embeddings :\n");
    try out.print("token ids shape:      ({d}, {d})\n", .{ batch, max_length });
    try out.print("token embeddings:     ({d}, {d}, {d})\n", .{ batch, max_length, output_dim });
    try out.print("position embeddings:  ({d}, {d})  <- one vector per position, shared by all rows\n", .{ emb.pos.rows, emb.pos.dim });
    try out.print("input embeddings:     ({d}, {d}, {d})  <- token + position\n", .{ batch, max_length, output_dim });
    try out.writeAll("---------------------\n");
}

pub fn main() !void {
    // Tip: an arena frees everything at once when the program ends, so the
    // demos do not need a `defer free` on every allocation.
    var arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
    defer arena.deinit();
    const gpa = arena.allocator();

    var buf: [4096]u8 = undefined;
    var stdout = std.fs.File.stdout().writer(&buf);
    const out = &stdout.interface;
    defer out.flush() catch {};

    const args = try std.process.argsAlloc(gpa);
    if (args.len == 1) {
        try runChapter2(gpa, out);
        try ch03.run(gpa, out);
        return;
    }
    for (args[1..]) |chapter| {
        if (std.mem.eql(u8, chapter, "ch02")) {
            try runChapter2(gpa, out);
        } else if (std.mem.eql(u8, chapter, "ch03")) {
            try ch03.run(gpa, out);
        } else {
            std.debug.print("unknown chapter '{s}'; use ch02 or ch03\n", .{chapter});
            std.process.exit(1);
        }
    }
}

//! Bonus (book 2.5) - byte pair encoding, the GPT-2 tokenizer.
//!
//! There is no tiktoken for Zig, so this is a small from-scratch encoder that
//! reads the GPT-2 vocabulary from `assets/gpt2-ranks.txt` (one line per token:
//! the token's bytes in hex, then its rank). It gives the same IDs as tiktoken's "gpt2".
//!
//! How it works, for one piece of text:
//!   1. start with one part per byte
//!   2. find the neighbouring pair whose joined bytes have the LOWEST rank
//!   3. merge it, repeat until no pair is a known token
//! Low rank = merged early in training = most frequent. Any byte string can be
//! encoded, so no `<|unk|>` is needed.

const std = @import("std");
const END_OF_TEXT = @import("preprocess.zig").END_OF_TEXT;

pub const ENDOFTEXT_ID: u32 = 50256;

pub const Bpe = struct {
    /// rank -> token bytes.
    tokens: [][]const u8,
    /// token bytes -> rank (keys are the slices in `tokens`).
    ranks: std.StringHashMapUnmanaged(u32),

    /// Parse the ranks file contents. Lines look like `48656c6c6f 1234`.
    pub fn init(gpa: std.mem.Allocator, file_contents: []const u8) !Bpe {
        var tokens: std.ArrayList([]const u8) = .empty;
        errdefer tokens.deinit(gpa);
        var lines = std.mem.splitScalar(u8, file_contents, '\n');
        while (lines.next()) |line| {
            const space = std.mem.indexOfScalar(u8, line, ' ') orelse continue;
            const bytes = try gpa.alloc(u8, space / 2);
            _ = try std.fmt.hexToBytes(bytes, line[0..space]);
            try tokens.append(gpa, bytes); // ranks are 0, 1, 2, ... in file order
        }
        var ranks: std.StringHashMapUnmanaged(u32) = .empty;
        errdefer ranks.deinit(gpa);
        try ranks.ensureTotalCapacity(gpa, @intCast(tokens.items.len));
        for (tokens.items, 0..) |t, rank| ranks.putAssumeCapacity(t, @intCast(rank));
        return .{ .tokens = try tokens.toOwnedSlice(gpa), .ranks = ranks };
    }

    pub fn deinit(self: *Bpe, gpa: std.mem.Allocator) void {
        for (self.tokens) |t| gpa.free(t);
        gpa.free(self.tokens);
        self.ranks.deinit(gpa);
    }

    /// Text -> token IDs. `<|endoftext|>` in the text becomes ID 50256. Caller frees.
    pub fn encode(self: Bpe, gpa: std.mem.Allocator, text: []const u8) ![]u32 {
        var out: std.ArrayList(u32) = .empty;
        errdefer out.deinit(gpa);
        var i: usize = 0;
        while (i < text.len) {
            if (std.mem.startsWith(u8, text[i..], END_OF_TEXT)) { // special token
                try out.append(gpa, ENDOFTEXT_ID);
                i += END_OF_TEXT.len;
                continue;
            }
            var end = nextPiece(text, i);
            // never let a piece swallow the start of a special token
            for (i + 1..end) |j| {
                if (std.mem.startsWith(u8, text[j..], END_OF_TEXT)) {
                    end = j;
                    break;
                }
            }
            try self.encodePiece(gpa, text[i..end], &out);
            i = end;
        }
        return out.toOwnedSlice(gpa);
    }

    /// IDs -> text. Caller frees.
    pub fn decode(self: Bpe, gpa: std.mem.Allocator, ids: []const u32) ![]u8 {
        var out: std.ArrayList(u8) = .empty;
        errdefer out.deinit(gpa);
        for (ids) |id| try out.appendSlice(gpa, if (id == ENDOFTEXT_ID) END_OF_TEXT else self.tokens[id]);
        return out.toOwnedSlice(gpa);
    }

    /// Encode one piece (a few bytes) by repeatedly merging the best neighbour pair.
    fn encodePiece(self: Bpe, gpa: std.mem.Allocator, piece: []const u8, out: *std.ArrayList(u32)) !void {
        // bounds[i]..bounds[i+1] is part i; begin with one part per byte
        var bounds: std.ArrayList(usize) = .empty;
        defer bounds.deinit(gpa);
        for (0..piece.len + 1) |i| try bounds.append(gpa, i);

        while (bounds.items.len > 2) {
            var best_rank: ?u32 = null;
            var best: usize = 0;
            for (0..bounds.items.len - 2) |i| { // rank of parts i and i+1 joined
                const joined = piece[bounds.items[i]..bounds.items[i + 2]];
                if (self.ranks.get(joined)) |r| {
                    if (best_rank == null or r < best_rank.?) {
                        best_rank = r;
                        best = i;
                    }
                }
            }
            if (best_rank == null) break;
            _ = bounds.orderedRemove(best + 1); // merge: drop the boundary between the two parts
        }
        for (0..bounds.items.len - 1) |i| {
            try out.append(gpa, self.ranks.get(piece[bounds.items[i]..bounds.items[i + 1]]).?);
        }
    }
};

// ---- splitting text into pieces (GPT-2's regex, written by hand) ----

// The GPT-2 pattern, in words:
//   'm 'd 's 't 're 've 'll | optional space + letters | optional space + digits
//   | optional space + other symbols | whitespace
// Bytes >= 0x80 (UTF-8) are treated as letters, which is right for most text.
fn isLetter(c: u8) bool {
    return std.ascii.isAlphabetic(c) or c >= 0x80;
}
fn isDigit(c: u8) bool {
    return std.ascii.isDigit(c);
}
fn isSpace(c: u8) bool {
    return c < 0x80 and std.ascii.isWhitespace(c);
}
fn isOther(c: u8) bool {
    return !isSpace(c) and !isLetter(c) and !isDigit(c);
}

fn consume(s: []const u8, start: usize, comptime pred: fn (u8) bool) usize {
    var i = start;
    while (i < s.len and pred(s[i])) i += 1;
    return i;
}

/// End of the piece starting at `i`.
fn nextPiece(s: []const u8, i: usize) usize {
    if (s[i] == '\'' and i + 1 < s.len) {
        if (std.mem.indexOfScalar(u8, "sdmt", s[i + 1]) != null) return i + 2;
        if (i + 2 < s.len) {
            const two = s[i + 1 .. i + 3];
            if (std.mem.eql(u8, two, "ll") or std.mem.eql(u8, two, "ve") or std.mem.eql(u8, two, "re")) return i + 3;
        }
    }
    const k = if (s[i] == ' ') i + 1 else i; // optional single leading space
    if (k < s.len) {
        if (isLetter(s[k])) return consume(s, k, isLetter);
        if (isDigit(s[k])) return consume(s, k, isDigit);
        if (isOther(s[k])) return consume(s, k, isOther);
    }
    // whitespace run: leave its last char for the next piece, unless it is the end
    const e = consume(s, i, isSpace);
    return if (e < s.len and e - i > 1) e - 1 else e;
}

test "pieces follow the GPT-2 pattern" {
    try std.testing.expectEqual(@as(usize, 5), nextPiece("Hello world", 0)); // "Hello"
    try std.testing.expectEqual(@as(usize, 11), nextPiece("Hello world", 5)); // " world"
    try std.testing.expectEqual(@as(usize, 2), nextPiece("'s x", 0)); // "'s"
}

test "bpe merges by lowest rank" {
    const gpa = std.testing.allocator;
    // bytes: a=0 b=1, then "ab"=2
    var bpe = try Bpe.init(gpa, "61 0\n62 1\n6162 2\n");
    defer bpe.deinit(gpa);
    const ids = try bpe.encode(gpa, "abab");
    defer gpa.free(ids);
    try std.testing.expectEqualSlices(u32, &.{ 2, 2 }, ids);
}
