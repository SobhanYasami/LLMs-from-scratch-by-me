What to compare with the book

  • 3.3 and 3.4 match the book exactly: scores, weights, query_2 = [0.4306, 1.4551], context_2 = [0.3061, 0.8210], and the six context vectors of SelfAttention_v1. The 3.4 weights are the book's torch.manual_seed(123) matrices, written out in the code so no language needs torch.
  • 3.5 weights follow from the same matrices. The book's causal example uses its v2 weights, so those rows differ.
  • Dropout and 3.6 use the seeded generator shared by all languages, so they differ from the book's torch output but agree with each other.

"""Chapter 3 demos: attention, one idea at a time (book 3.3 to 3.6).

Run:  python-version/.venv/bin/python python-version/main.py ch03

Tip: read the demos top to bottom; each builds on the one before. The numbers
in 3.3 and 3.4 match the book exactly (3.4 uses the book's seed-123 weights,
hard-coded so every language shares them).
"""

import math

import torch

from attention import CausalAttention, MultiHeadAttention, Rng, SelfAttention, dropout, simple_attention

# "Your journey starts with one step." as 3-dimensional embeddings (book 3.3.1).
INPUTS = torch.tensor(
    [
        [0.43, 0.15, 0.89],  # Your
        [0.55, 0.87, 0.66],  # journey
        [0.57, 0.85, 0.64],  # starts
        [0.22, 0.58, 0.33],  # with
        [0.77, 0.25, 0.10],  # one
        [0.05, 0.80, 0.55],  # step
    ]
)

# The book's torch.manual_seed(123) weights (d_in=3, d_out=2), written out so no language needs torch.
W_QUERY = [[0.29611194, 0.51656228], [0.25167072, 0.68855679], [0.07397246, 0.86652195]]
W_KEY = [[0.13657987, 0.10247904], [0.18405646, 0.72644675], [0.31525391, 0.68710667]]
W_VALUE = [[0.07563531, 0.19663817], [0.31641197, 0.40174013], [0.11856830, 0.82739538]]


def fmt_row(values) -> str:
    return "[" + ", ".join(f"{v:.4f}" for v in values) + "]"


def show(label: str, t: torch.Tensor) -> None:
    t = t.detach()
    if t.dim() == 1:
        print(f"{label}: {fmt_row(t.tolist())}")
    else:
        print(f"{label}:")
        for row in t.tolist():
            print("  " + fmt_row(row))


def demo_simple_attention() -> None:
    """3.3: one query by hand, then all tokens at once."""
    print("== 3.3.1 simple attention for one query (x2, 'journey') ==")
    query = INPUTS[1]
    scores_2 = INPUTS @ query  # dot product of every token with the query
    weights_2 = torch.softmax(scores_2, dim=0)
    context_2 = weights_2 @ INPUTS
    show("scores    ", scores_2)
    show("weights   ", weights_2)
    show("context z2", context_2)

    print("== 3.3.2 all tokens at once ==")
    scores, weights, context = simple_attention(INPUTS)
    show("scores (T x T)", scores)
    show("weights (rows sum to 1)", weights)
    show("row sums", weights.sum(dim=1))
    show("context vectors (T x d)", context)


def demo_self_attention() -> None:
    """3.4: query/key/value with trainable (here: fixed) weights."""
    print("== 3.4 self-attention with trainable weights (d_in=3, d_out=2) ==")
    wq, wk, wv = (torch.tensor(w) for w in (W_QUERY, W_KEY, W_VALUE))
    sa = SelfAttention(wq, wk, wv)
    x2 = INPUTS[1]
    query_2, keys, values = x2 @ wq, INPUTS @ wk, INPUTS @ wv
    show("query_2", query_2)
    print(f"keys shape: {tuple(keys.shape)}, values shape: {tuple(values.shape)}")
    scores_2 = query_2 @ keys.T
    show("scores_2 (q2 . keys)", scores_2)
    weights_2 = torch.softmax(scores_2 / math.sqrt(keys.shape[-1]), dim=-1)
    show("weights_2 (scaled)", weights_2)
    show("context_2", weights_2 @ values)
    show("all context vectors (T x d_out)", sa(INPUTS))


def demo_causal_attention() -> None:
    """3.5: mask the future, then dropout."""
    print("== 3.5 causal attention ==")
    wq, wk, wv = (torch.tensor(w) for w in (W_QUERY, W_KEY, W_VALUE))
    ca = CausalAttention(wq, wk, wv, context_length=6)
    show("causal weights (upper triangle is 0)", ca.attention_weights(INPUTS))
    show("causal context vectors", ca(INPUTS))
    dropped = dropout(torch.ones(6, 6), 0.5, Rng(123))
    show("dropout 0.5 on ones(6, 6), seed 123 (kept values are doubled)", dropped)


def demo_multihead() -> None:
    """3.6: split the work across heads."""
    print("== 3.6 multi-head attention (d_in=3, d_out=2, heads=2) ==")
    mha = MultiHeadAttention.random(d_in=3, d_out=2, num_heads=2, context_length=6, rng=Rng(123))
    batch = torch.stack((INPUTS, INPUTS), dim=0)  # two copies = a batch of 2
    out = mha(batch)
    print(f"input shape: {tuple(batch.shape)}  output shape: {tuple(out.shape)}")
    show("output[0]", out[0])


def run() -> None:
    demo_simple_attention()
    demo_self_attention()
    demo_causal_attention()
    demo_multihead()

package main

// Chapter 3 demos: attention, one idea at a time (book 3.3 to 3.6).
//
// Run from go-version/:  go run ./cmd ch03
//
// Tip: read the demos top to bottom; each builds on the one before. The numbers
// in 3.3 and 3.4 match the book exactly (3.4 uses the book's seed-123 weights,
// hard-coded so every language shares them).

import (
	"fmt"
	"math"
	"strings"

	"llm-from-scratch/attention"
)

// "Your journey starts with one step." as 3-dimensional embeddings (book 3.3.1).
var inputs = attention.FromRows([][]float32{
	{0.43, 0.15, 0.89}, // Your
	{0.55, 0.87, 0.66}, // journey
	{0.57, 0.85, 0.64}, // starts
	{0.22, 0.58, 0.33}, // with
	{0.77, 0.25, 0.10}, // one
	{0.05, 0.80, 0.55}, // step
})

// The book's torch.manual_seed(123) weights (d_in=3, d_out=2), written out so no language needs torch.
var (
	wQuery = attention.FromRows([][]float32{{0.29611194, 0.51656228}, {0.25167072, 0.68855679}, {0.07397246, 0.86652195}})
	wKey   = attention.FromRows([][]float32{{0.13657987, 0.10247904}, {0.18405646, 0.72644675}, {0.31525391, 0.68710667}})
	wValue = attention.FromRows([][]float32{{0.07563531, 0.19663817}, {0.31641197, 0.40174013}, {0.11856830, 0.82739538}})
)

func fmtRow(values []float32) string {
	parts := make([]string, len(values))
	for i, v := range values {
		if math.IsInf(float64(v), -1) {
			parts[i] = "-inf"
		} else {
			parts[i] = fmt.Sprintf("%.4f", v)
		}
	}
	return "[" + strings.Join(parts, ", ") + "]"
}

func showVec(label string, v []float32) { fmt.Printf("%s: %s\n", label, fmtRow(v)) }

func showMatrix(label string, m attention.Matrix) {
	fmt.Printf("%s:\n", label)
	for i := 0; i < m.Rows; i++ {
		fmt.Println("  " + fmtRow(m.Row(i)))
	}
}

// demoSimpleAttention: 3.3, one query by hand, then all tokens at once.
func demoSimpleAttention() {
	fmt.Println("== 3.3.1 simple attention for one query (x2, 'journey') ==")
	query := attention.FromRows([][]float32{inputs.Row(1)})
	scores2 := attention.MatMul(query, attention.Transpose(inputs)) // dot product of every token with the query
	weights2 := attention.SoftmaxRows(scores2)
	context2 := attention.MatMul(weights2, inputs)
	showVec("scores    ", scores2.Row(0))
	showVec("weights   ", weights2.Row(0))
	showVec("context z2", context2.Row(0))

	fmt.Println("== 3.3.2 all tokens at once ==")
	scores, weights, context := attention.SimpleAttention(inputs)
	showMatrix("scores (T x T)", scores)
	showMatrix("weights (rows sum to 1)", weights)
	sums := make([]float32, weights.Rows)
	for i := range sums {
		for _, w := range weights.Row(i) {
			sums[i] += w
		}
	}
	showVec("row sums", sums)
	showMatrix("context vectors (T x d)", context)
}

// demoSelfAttention: 3.4, query/key/value with trainable (here: fixed) weights.
func demoSelfAttention() {
	fmt.Println("== 3.4 self-attention with trainable weights (d_in=3, d_out=2) ==")
	sa := &attention.SelfAttention{WQuery: wQuery, WKey: wKey, WValue: wValue}
	x2 := attention.FromRows([][]float32{inputs.Row(1)})
	query2 := attention.MatMul(x2, wQuery)
	keys, values := attention.MatMul(inputs, wKey), attention.MatMul(inputs, wValue)
	showVec("query_2", query2.Row(0))
	fmt.Printf("keys shape: (%d, %d), values shape: (%d, %d)\n", keys.Rows, keys.Cols, values.Rows, values.Cols)
	scores2 := attention.MatMul(query2, attention.Transpose(keys))
	showVec("scores_2 (q2 . keys)", scores2.Row(0))
	scaled := attention.Scale(scores2, 1/float32(math.Sqrt(float64(keys.Cols))))
	weights2 := attention.SoftmaxRows(scaled)
	showVec("weights_2 (scaled)", weights2.Row(0))
	showVec("context_2", attention.MatMul(weights2, values).Row(0))
	showMatrix("all context vectors (T x d_out)", sa.Forward(inputs))
}

// demoCausalAttention: 3.5, mask the future, then dropout.
func demoCausalAttention() {
	fmt.Println("== 3.5 causal attention ==")
	ca := attention.NewCausalAttention(wQuery, wKey, wValue, 0, nil)
	showMatrix("causal weights (upper triangle is 0)", ca.CausalWeights(inputs, false))
	showMatrix("causal context vectors", ca.CausalForward(inputs, false))
	ones := attention.NewMatrix(6, 6)
	for i := range ones.Data {
		ones.Data[i] = 1
	}
	dropped := attention.Dropout(ones, 0.5, attention.NewRng(123))
	showMatrix("dropout 0.5 on ones(6, 6), seed 123 (kept values are doubled)", dropped)
}

// demoMultiHead: 3.6, split the work across heads.
func demoMultiHead() {
	fmt.Println("== 3.6 multi-head attention (d_in=3, d_out=2, heads=2) ==")
	mha := attention.NewRandomMultiHead(3, 2, 2, attention.NewRng(123))
	// A batch is just the same computation repeated per example; two copies = a batch of 2.
	out := mha.Forward(inputs)
	fmt.Printf("input shape: (2, %d, %d)  output shape: (2, %d, %d)\n", inputs.Rows, inputs.Cols, out.Rows, out.Cols)
	showMatrix("output[0]", out)
}

func runChapter3() {
	demoSimpleAttention()
	demoSelfAttention()
	demoCausalAttention()
	demoMultiHead()
}

//! Chapter 3 demos: attention, one idea at a time (book 3.3 to 3.6).
//!
//! Run from rust-version/:  `cargo run -- ch03`
//!
//! Tip: read the demos top to bottom; each builds on the one before. The numbers
//! in 3.3 and 3.4 match the book exactly (3.4 uses the book's seed-123 weights,
//! hard-coded so every language shares them).

#![allow(clippy::excessive_precision)] // the book weights are written out in full

use llm_from_scratch::attention::{
    CausalAttention, Matrix, MultiHeadAttention, SelfAttention, simple_attention,
};
use llm_from_scratch::tokenization::rng::Rng;

/// "Your journey starts with one step." as 3-dimensional embeddings (book 3.3.1).
fn inputs() -> Matrix {
    Matrix::from_rows(&[
        &[0.43, 0.15, 0.89], // Your
        &[0.55, 0.87, 0.66], // journey
        &[0.57, 0.85, 0.64], // starts
        &[0.22, 0.58, 0.33], // with
        &[0.77, 0.25, 0.10], // one
        &[0.05, 0.80, 0.55], // step
    ])
}

/// The book's torch.manual_seed(123) weights (d_in=3, d_out=2), written out so no language needs torch.
fn book_weights() -> (Matrix, Matrix, Matrix) {
    (
        Matrix::from_rows(&[
            &[0.29611194, 0.51656228],
            &[0.25167072, 0.68855679],
            &[0.07397246, 0.86652195],
        ]),
        Matrix::from_rows(&[
            &[0.13657987, 0.10247904],
            &[0.18405646, 0.72644675],
            &[0.31525391, 0.68710667],
        ]),
        Matrix::from_rows(&[
            &[0.07563531, 0.19663817],
            &[0.31641197, 0.40174013],
            &[0.11856830, 0.82739538],
        ]),
    )
}

fn fmt_row(values: &[f32]) -> String {
    let parts: Vec<String> = values.iter().map(|v| format!("{v:.4}")).collect();
    format!("[{}]", parts.join(", "))
}

fn show_vec(label: &str, v: &[f32]) {
    println!("{label}: {}", fmt_row(v));
}

fn show_matrix(label: &str, m: &Matrix) {
    println!("{label}:");
    for i in 0..m.rows {
        println!("  {}", fmt_row(m.row(i)));
    }
}

/// 3.3: one query by hand, then all tokens at once.
fn demo_simple_attention() {
    let inputs = inputs();
    println!("== 3.3.1 simple attention for one query (x2, 'journey') ==");
    let query = Matrix::from_rows(&[inputs.row(1)]);
    let scores_2 = query.matmul(&inputs.transpose()); // dot product of every token with the query
    let weights_2 = scores_2.softmax_rows();
    let context_2 = weights_2.matmul(&inputs);
    show_vec("scores    ", scores_2.row(0));
    show_vec("weights   ", weights_2.row(0));
    show_vec("context z2", context_2.row(0));

    println!("== 3.3.2 all tokens at once ==");
    let (scores, weights, context) = simple_attention(&inputs);
    show_matrix("scores (T x T)", &scores);
    show_matrix("weights (rows sum to 1)", &weights);
    let sums: Vec<f32> = (0..weights.rows)
        .map(|i| weights.row(i).iter().sum())
        .collect();
    show_vec("row sums", &sums);
    show_matrix("context vectors (T x d)", &context);
}

/// 3.4: query/key/value with trainable (here: fixed) weights.
fn demo_self_attention() {
    let inputs = inputs();
    println!("== 3.4 self-attention with trainable weights (d_in=3, d_out=2) ==");
    let (wq, wk, wv) = book_weights();
    let x2 = Matrix::from_rows(&[inputs.row(1)]);
    let query_2 = x2.matmul(&wq);
    let keys = inputs.matmul(&wk);
    let values = inputs.matmul(&wv);
    show_vec("query_2", query_2.row(0));
    println!(
        "keys shape: ({}, {}), values shape: ({}, {})",
        keys.rows, keys.cols, values.rows, values.cols
    );
    let scores_2 = query_2.matmul(&keys.transpose());
    show_vec("scores_2 (q2 . keys)", scores_2.row(0));
    let weights_2 = scores_2
        .scale(1.0 / (keys.cols as f64).sqrt() as f32)
        .softmax_rows();
    show_vec("weights_2 (scaled)", weights_2.row(0));
    show_vec("context_2", weights_2.matmul(&values).row(0));
    let sa = SelfAttention {
        w_query: wq,
        w_key: wk,
        w_value: wv,
    };
    show_matrix("all context vectors (T x d_out)", &sa.forward(&inputs));
}

/// 3.5: mask the future, then dropout.
fn demo_causal_attention() {
    let inputs = inputs();
    println!("== 3.5 causal attention ==");
    let (wq, wk, wv) = book_weights();
    let ca = CausalAttention::new(wq, wk, wv, 0.0);
    show_matrix(
        "causal weights (upper triangle is 0)",
        &ca.attention_weights(&inputs, None),
    );
    show_matrix("causal context vectors", &ca.forward(&inputs, None));
    let mut ones = Matrix::new(6, 6);
    ones.data.fill(1.0);
    let dropped = ones.dropout(0.5, &mut Rng::new(123));
    show_matrix(
        "dropout 0.5 on ones(6, 6), seed 123 (kept values are doubled)",
        &dropped,
    );
}

/// 3.6: split the work across heads.
fn demo_multihead() {
    let inputs = inputs();
    println!("== 3.6 multi-head attention (d_in=3, d_out=2, heads=2) ==");
    let mha = MultiHeadAttention::random(3, 2, 2, &mut Rng::new(123));
    // A batch is just the same computation repeated per example; two copies = a batch of 2.
    let out = mha.forward(&inputs);
    println!(
        "input shape: (2, {}, {})  output shape: (2, {}, {})",
        inputs.rows, inputs.cols, out.rows, out.cols
    );
    show_matrix("output[0]", &out);
}

pub fn run() {
    demo_simple_attention();
    demo_self_attention();
    demo_causal_attention();
    demo_multihead();
}

/* Chapter 3 demos: attention, one idea at a time (book 3.3 to 3.6).
 *
 * Run from C-version/:  make run ARGS=ch03   (or ./build/main ch03)
 *
 * Tip: read the demos top to bottom; each builds on the one before. The numbers
 * in 3.3 and 3.4 match the book exactly (3.4 uses the book's seed-123 weights,
 * hard-coded so every language shares them).
 */
#include "ch03.h"

#include <math.h>
#include <stdio.h>

#include "attention/causal.h"
#include "attention/matrix.h"
#include "attention/multihead.h"
#include "attention/self_attention.h"
#include "attention/simple.h"

/* "Your journey starts with one step." as 3-dimensional embeddings (book 3.3.1). */
static const float INPUTS[6][3] = {
    {0.43f, 0.15f, 0.89f}, /* Your */
    {0.55f, 0.87f, 0.66f}, /* journey */
    {0.57f, 0.85f, 0.64f}, /* starts */
    {0.22f, 0.58f, 0.33f}, /* with */
    {0.77f, 0.25f, 0.10f}, /* one */
    {0.05f, 0.80f, 0.55f}, /* step */
};

/* The book's torch.manual_seed(123) weights (d_in=3, d_out=2), written out so no language needs torch. */
static const float W_QUERY[3][2] = {{0.29611194f, 0.51656228f}, {0.25167072f, 0.68855679f}, {0.07397246f, 0.86652195f}};
static const float W_KEY[3][2] = {{0.13657987f, 0.10247904f}, {0.18405646f, 0.72644675f}, {0.31525391f, 0.68710667f}};
static const float W_VALUE[3][2] = {{0.07563531f, 0.19663817f}, {0.31641197f, 0.40174013f}, {0.11856830f, 0.82739538f}};

static void print_row(const float *values, size_t n) {
    printf("[");
    for (size_t i = 0; i < n; i++) printf(i ? ", %.4f" : "%.4f", values[i]);
    printf("]");
}

static void show_vec(const char *label, const float *values, size_t n) {
    printf("%s: ", label);
    print_row(values, n);
    printf("\n");
}

static void show_matrix(const char *label, Matrix m) {
    printf("%s:\n", label);
    for (size_t i = 0; i < m.rows; i++) {
        printf("  ");
        print_row(matrix_row(m, i), m.cols);
        printf("\n");
    }
}

/* 3.3: one query by hand, then all tokens at once. */
static void demo_simple_attention(void) {
    Arena a = arena_new(1 << 20);
    Matrix inputs = matrix_from(&a, 6, 3, &INPUTS[0][0]);
    printf("== 3.3.1 simple attention for one query (x2, 'journey') ==\n");
    Matrix query = matrix_from(&a, 1, 3, INPUTS[1]);
    Matrix scores_2 = matmul(&a, query, transpose(&a, inputs)); /* dot product of every token with the query */
    Matrix weights_2 = softmax_rows(&a, scores_2);
    Matrix context_2 = matmul(&a, weights_2, inputs);
    show_vec("scores    ", scores_2.data, 6);
    show_vec("weights   ", weights_2.data, 6);
    show_vec("context z2", context_2.data, 3);

    printf("== 3.3.2 all tokens at once ==\n");
    SimpleAttentionResult r = simple_attention(&a, inputs);
    show_matrix("scores (T x T)", r.scores);
    show_matrix("weights (rows sum to 1)", r.weights);
    float sums[6] = {0};
    for (size_t i = 0; i < 6; i++)
        for (size_t j = 0; j < 6; j++) sums[i] += matrix_at(r.weights, i, j);
    show_vec("row sums", sums, 6);
    show_matrix("context vectors (T x d)", r.context);
    arena_free(&a);
}

/* 3.4: query/key/value with trainable (here: fixed) weights. */
static void demo_self_attention(void) {
    Arena a = arena_new(1 << 20);
    Matrix inputs = matrix_from(&a, 6, 3, &INPUTS[0][0]);
    printf("== 3.4 self-attention with trainable weights (d_in=3, d_out=2) ==\n");
    SelfAttention sa = {matrix_from(&a, 3, 2, &W_QUERY[0][0]), matrix_from(&a, 3, 2, &W_KEY[0][0]),
                        matrix_from(&a, 3, 2, &W_VALUE[0][0])};
    Matrix x2 = matrix_from(&a, 1, 3, INPUTS[1]);
    Matrix query_2 = matmul(&a, x2, sa.w_query);
    Matrix keys = matmul(&a, inputs, sa.w_key), values = matmul(&a, inputs, sa.w_value);
    show_vec("query_2", query_2.data, 2);
    printf("keys shape: (%zu, %zu), values shape: (%zu, %zu)\n", keys.rows, keys.cols, values.rows, values.cols);
    Matrix scores_2 = matmul(&a, query_2, transpose(&a, keys));
    show_vec("scores_2 (q2 . keys)", scores_2.data, 6);
    Matrix weights_2 = softmax_rows(&a, scale_by_sqrt(&a, scores_2, keys.cols));
    show_vec("weights_2 (scaled)", weights_2.data, 6);
    show_vec("context_2", matmul(&a, weights_2, values).data, 2);
    show_matrix("all context vectors (T x d_out)", self_attention_forward(&a, &sa, inputs));
    arena_free(&a);
}

static void demo_causal_attention(void) {
    Arena a = arena_new(1 << 20);
    Matrix inputs = matrix_from(&a, 6, 3, &INPUTS[0][0]);
    printf("== 3.5 causal attention ==\n");
    CausalAttention ca = {{matrix_from(&a, 3, 2, &W_QUERY[0][0]), matrix_from(&a, 3, 2, &W_KEY[0][0]),
                           matrix_from(&a, 3, 2, &W_VALUE[0][0])},
                          0.0f};
    show_matrix("causal weights (upper triangle is 0)", causal_weights(&a, &ca, inputs, NULL));
    show_matrix("causal context vectors", causal_forward(&a, &ca, inputs, NULL));
    Matrix ones = matrix_new(&a, 6, 6);
    for (size_t i = 0; i < 36; i++) ones.data[i] = 1.0f;
    Rng rng = rng_new(123);
    show_matrix("dropout 0.5 on ones(6, 6), seed 123 (kept values are doubled)", dropout(&a, ones, 0.5f, &rng));
    arena_free(&a);
}

//! Chapter 3 demos: attention, one idea at a time (book 3.3 to 3.6).
//!
//! Run from Zig-version/:  `zig build run -- ch03`
//!
//! Tip: read the demos top to bottom; each builds on the one before. The numbers
//! in 3.3 and 3.4 match the book exactly (3.4 uses the book's seed-123 weights,
//! hard-coded so every language shares them).

const std = @import("std");
const att = @import("attention/root.zig");
const Rng = @import("tokenization/rng.zig").Rng;

const Matrix = att.Matrix;
const Out = *std.Io.Writer;

/// "Your journey starts with one step." as 3-dimensional embeddings (book 3.3.1).
const inputs_data = [_]f32{
    0.43, 0.15, 0.89, // Your
    0.55, 0.87, 0.66, // journey
    0.57, 0.85, 0.64, // starts
    0.22, 0.58, 0.33, // with
    0.77, 0.25, 0.10, // one
    0.05, 0.80, 0.55, // step
};

// The book's torch.manual_seed(123) weights (d_in=3, d_out=2), written out so no language needs torch.
const w_query_data = [_]f32{ 0.29611194, 0.51656228, 0.25167072, 0.68855679, 0.07397246, 0.86652195 };
const w_key_data = [_]f32{ 0.13657987, 0.10247904, 0.18405646, 0.72644675, 0.31525391, 0.68710667 };
const w_value_data = [_]f32{ 0.07563531, 0.19663817, 0.31641197, 0.40174013, 0.11856830, 0.82739538 };

fn printRow(out: Out, values: []const f32) !void {
    try out.writeAll("[");
    for (values, 0..) |v, i| {
        if (i > 0) try out.writeAll(", ");
        try out.print("{d:.4}", .{v});
    }
    try out.writeAll("]");
}

fn showVec(out: Out, label: []const u8, values: []const f32) !void {
    try out.print("{s}: ", .{label});
    try printRow(out, values);
    try out.writeAll("\n");
}

fn showMatrix(out: Out, label: []const u8, m: Matrix) !void {
    try out.print("{s}:\n", .{label});
    for (0..m.rows) |i| {
        try out.writeAll("  ");
        try printRow(out, m.row(i));
        try out.writeAll("\n");
    }
}

/// 3.3: one query by hand, then all tokens at once.
fn demoSimpleAttention(gpa: std.mem.Allocator, out: Out) !void {
    const inputs = try Matrix.fromSlice(gpa, 6, 3, &inputs_data);
    try out.writeAll("== 3.3.1 simple attention for one query (x2, 'journey') ==\n");
    const query = try Matrix.fromSlice(gpa, 1, 3, inputs.row(1));
    const scores_2 = try query.matmul(gpa, try inputs.transpose(gpa)); // dot product of every token with the query
    const weights_2 = try scores_2.softmaxRows(gpa);
    const context_2 = try weights_2.matmul(gpa, inputs);
    try showVec(out, "scores    ", scores_2.row(0));
    try showVec(out, "weights   ", weights_2.row(0));
    try showVec(out, "context z2", context_2.row(0));

    try out.writeAll("== 3.3.2 all tokens at once ==\n");
    const r = try att.simpleAttention(gpa, inputs);
    try showMatrix(out, "scores (T x T)", r.scores);
    try showMatrix(out, "weights (rows sum to 1)", r.weights);
    var sums: [6]f32 = undefined;
    for (0..6) |i| {
        var total: f32 = 0;
        for (r.weights.row(i)) |w| total += w;
        sums[i] = total;
    }
    try showVec(out, "row sums", &sums);
    try showMatrix(out, "context vectors (T x d)", r.context);
}

/// 3.4: query/key/value with trainable (here: fixed) weights.
fn demoSelfAttention(gpa: std.mem.Allocator, out: Out) !void {
    const inputs = try Matrix.fromSlice(gpa, 6, 3, &inputs_data);
    try out.writeAll("== 3.4 self-attention with trainable weights (d_in=3, d_out=2) ==\n");
    const sa = att.SelfAttention{
        .w_query = try Matrix.fromSlice(gpa, 3, 2, &w_query_data),
        .w_key = try Matrix.fromSlice(gpa, 3, 2, &w_key_data),
        .w_value = try Matrix.fromSlice(gpa, 3, 2, &w_value_data),
    };
    const x2 = try Matrix.fromSlice(gpa, 1, 3, inputs.row(1));
    const query_2 = try x2.matmul(gpa, sa.w_query);
    const keys = try inputs.matmul(gpa, sa.w_key);
    const values = try inputs.matmul(gpa, sa.w_value);
    try showVec(out, "query_2", query_2.row(0));
    try out.print("keys shape: ({d}, {d}), values shape: ({d}, {d})\n", .{ keys.rows, keys.cols, values.rows, values.cols });
    const scores_2 = try query_2.matmul(gpa, try keys.transpose(gpa));
    try showVec(out, "scores_2 (q2 . keys)", scores_2.row(0));
    const weights_2 = try (try att.self_attention.scaleBySqrt(gpa, scores_2, keys.cols)).softmaxRows(gpa);
    try showVec(out, "weights_2 (scaled)", weights_2.row(0));
    try showVec(out, "context_2", (try weights_2.matmul(gpa, values)).row(0));
    try showMatrix(out, "all context vectors (T x d_out)", try sa.forward(gpa, inputs));
}

/// 3.5: mask the future, then dropout.
fn demoCausalAttention(gpa: std.mem.Allocator, out: Out) !void {
    const inputs = try Matrix.fromSlice(gpa, 6, 3, &inputs_data);
    try out.writeAll("== 3.5 causal attention ==\n");
    const ca = att.CausalAttention{ .base = .{
        .w_query = try Matrix.fromSlice(gpa, 3, 2, &w_query_data),
        .w_key = try Matrix.fromSlice(gpa, 3, 2, &w_key_data),
        .w_value = try Matrix.fromSlice(gpa, 3, 2, &w_value_data),
    } };
    try showMatrix(out, "causal weights (upper triangle is 0)", try ca.attentionWeights(gpa, inputs, null));
    try showMatrix(out, "causal context vectors", try ca.forward(gpa, inputs, null));
    const ones = try Matrix.init(gpa, 6, 6);
    @memset(ones.data, 1);
    var rng = Rng.init(123);
    try showMatrix(out, "dropout 0.5 on ones(6, 6), seed 123 (kept values are doubled)", try ones.dropout(gpa, 0.5, &rng));
}

/// 3.6: split the work across heads.
fn demoMultihead(gpa: std.mem.Allocator, out: Out) !void {
    const inputs = try Matrix.fromSlice(gpa, 6, 3, &inputs_data);
    try out.writeAll("== 3.6 multi-head attention (d_in=3, d_out=2, heads=2) ==\n");
    var rng = Rng.init(123);
    const mha = try att.MultiHeadAttention.random(gpa, 3, 2, 2, &rng);

    const result = try mha.forward(gpa, inputs);

}
