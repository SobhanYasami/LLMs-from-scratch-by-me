2.2: توکن‌سازی متن

بخش کتاب 2.2 · کد به زبان Python, Go, Rust, C, Zig

هدف. مدل متن نمی‌خواند، فقط عدد می‌خواند. پیش از عددگذاری، متن را به تکه‌هایی به نام توکن ببُرید.

توکن اینجا یک کلمه یا یک علامت نگارشی است. یک عبارت منظم جداکننده‌ها (علامت نگارشی، --، فاصله) را پیدا می‌کند و ما جداکننده‌ها را نگه می‌داریم تا علامت نگارشی توکنِ مستقل شود. سپس تکه‌های فاصله دور ریخته می‌شوند.

"Hello, world!"  ->  ['Hello', ',', 'world', '!']

کتاب چگونه به اینجا می‌رسد

  1. فقط روی فاصله تقسیم کن: re.split(r'(\s)', text) علامت نگارشی را به کلمه چسبیده می‌گذارد ('Hello,').
  2. روی , و . هم تقسیم کن: اکنون علامت‌ها جدا شده‌اند اما رشته‌های خالی و فاصله می‌مانند.
  3. با item.strip() فیلتر کن تا تکه‌های فاصله حذف شوند.
  4. علامت‌های ? " ( ) _ ! و -- را اضافه کن. نتیجه روی کل داستان: 4,649 توکن.
در Python، re.split فقط وقتی جداکننده‌ها را نگه می‌دارد که الگو داخل پرانتز (گروه ثبت‌کننده) باشد. Go و Rust چنین قابلیتی ندارند، پس تطابق‌ها را دستی پیمایش می‌کنند. C و Zig اصلاً regex ندارند و بایت‌به‌بایت اسکن می‌کنند. نتیجه یکی است، سازوکار آشکارتر.
حروف را کوچک نمی‌کنیم: بزرگی حروف به مدل کمک می‌کند نام‌ها را از اسم‌های عام تشخیص دهد و ساختار جمله را یاد بگیرد. برای اختصار فاصله‌ها را دور می‌ریزیم؛ برای کد (که تورفتگی مهم است) آن‌ها را نگه می‌دارید.

python-version/tokenization/preprocess.py
"""Step 1 - Split raw text into tokens (and join them back).

A token here is a word, a punctuation mark, or a whitespace-free piece of text.
The regex keeps the delimiters, so punctuation becomes its own token:

    "Hello, world!"  ->  ['"', 'Hello', ',', 'world', '!', '"']   (roughly)

Note: ``re.split`` with a *capturing* group returns the separators too.
Without the parentheses in ``SPLIT_PATTERN`` they would be dropped.
"""

import re

# Tip: start reading here, then vocab -> simple -> dataset -> embedding.
# Each file uses the one before it.

# Captures: , . ? _ ! " ( ) ' , the double dash "--", and any whitespace.
SPLIT_PATTERN = r'([,.?_!"()\']|--|\s)'

# Used when decoding: removes the space that " ".join() put before punctuation.
DETOKENIZE_PATTERN = r'\s+([,.?!"()\'])'

# Special tokens. They are not produced by splitting text; they are added on purpose.
END_OF_TEXT = "<|endoftext|>"
UNK_TOKEN = "<|unk|>"


def split_text(text: str) -> list[str]:
    """Return the tokens of ``text``, with whitespace-only pieces removed.

    >>> split_text('Hello, world!')
    ['Hello', ',', 'world', '!']
    """
    pieces = re.split(SPLIT_PATTERN, text)
    # re.split leaves empty strings and lone spaces behind; drop them.
    return [piece.strip() for piece in pieces if piece.strip()]


def detokenize(tokens: list[str]) -> str:
    """Join tokens back into text, undoing the spaces added before punctuation.

    >>> detokenize(['Hello', ',', 'world', '!'])
    'Hello, world!'

    This is not a perfect inverse: the original spacing is lost, so
    ``detokenize(split_text(t)) == t`` only holds for "simple" text.
    """
    text = " ".join(tokens)
    return re.sub(DETOKENIZE_PATTERN, r"\1", text)

// Package tokenization holds the educational tokenizer building blocks.
//
// Files, in the order you learn them:
//   - preprocess.go: split raw text into tokens and join them back.
//   - vocab.go:      map each unique token to an integer ID.
//   - simple.go:     SimpleTokenizerV1 and SimpleTokenizerV2.
//   - dataset.go:    sliding-window (input, target) pairs for next-token prediction.
//   - dataloader.go: batching and shuffling of those pairs.
//   - embedding.go:  token + positional embeddings.
//
// Step 1 - split raw text into tokens. A token is a word or a punctuation mark.
//
//	"Hello, world!" -> ["Hello", ",", "world", "!"]
//
// Tip: start reading here, then vocab.go -> simple.go -> dataset.go ->
// dataloader.go -> embedding.go. Each file uses the one before it.
package tokenization

import (
	"regexp"
	"strings"
)

// Special tokens. They are not produced by splitting text; they are added on purpose.
const (
	EndOfText = "<|endoftext|>"
	Unknown   = "<|unk|>"
)

// Tip: the regex lists the delimiters. Keeping them (instead of dropping them)
// is what turns punctuation into tokens of its own.
// splitRe mirrors the Python pattern: captured delimiters are kept in the output.
var splitRe = regexp.MustCompile(`([,.?_!"()']|--|\s)`)

// punctSpaceRe removes the space that Detokenize inserts before punctuation.
var punctSpaceRe = regexp.MustCompile(`\s+([,.?!"()'])`)

// splitKeep behaves like Python's re.split with a capturing group.
func splitKeep(re *regexp.Regexp, s string) []string {
	var out []string
	last := 0
	for _, m := range re.FindAllStringIndex(s, -1) {
		out = append(out, s[last:m[0]], s[m[0]:m[1]])
		last = m[1]
	}
	return append(out, s[last:])
}

// SplitText returns the tokens of text, with whitespace-only pieces removed.
func SplitText(text string) []string {
	var out []string
	for _, item := range splitKeep(splitRe, text) {
		if t := strings.TrimSpace(item); t != "" {
			out = append(out, t)
		}
	}
	return out
}

// Detokenize joins tokens back into text, undoing the spaces added before
// punctuation. Not a perfect inverse: the original spacing is lost.
func Detokenize(tokens []string) string {
	return punctSpaceRe.ReplaceAllString(strings.Join(tokens, " "), "${1}")
}
rust-version/src/tokenization/preprocess.rs
//! Step 1 - split raw text into tokens (and join them back).
//!
//! A token is a word or a punctuation mark.
//!
//! Tip: start reading here, then vocab -> simple -> dataset -> dataloader ->
//! embedding. Each file uses the one before it. The pattern keeps its delimiters,
//! so punctuation becomes its own token. Rust's `regex` has no capturing split,
//! so `split_text` walks the matches by hand (Go does the same).

use std::sync::LazyLock;

use regex::Regex;

/// Special tokens. They are not produced by splitting text; they are added on purpose.
pub const END_OF_TEXT: &str = "<|endoftext|>";
pub const UNK_TOKEN: &str = "<|unk|>";

/// Matches: , . ? _ ! " ( ) ' , the double dash "--", and any whitespace.
static SPLIT_RE: LazyLock<Regex> = LazyLock::new(|| Regex::new(r#"[,.?_!"()']|--|\s"#).unwrap());

/// Used when decoding: removes the space that join(" ") put before punctuation.
static DETOKENIZE_RE: LazyLock<Regex> = LazyLock::new(|| Regex::new(r#"\s+([,.?!"()'])"#).unwrap());

/// Return the tokens of `text`, with whitespace-only pieces removed.
///
/// ```
/// use llm_from_scratch::tokenization::split_text;
/// assert_eq!(split_text("Hello, world!"), ["Hello", ",", "world", "!"]);
/// ```
pub fn split_text(text: &str) -> Vec<String> {
    let mut pieces = Vec::new();
    let mut last = 0;
    for m in SPLIT_RE.find_iter(text) {
        pieces.push(&text[last..m.start()]);
        pieces.push(m.as_str());
        last = m.end();
    }
    pieces.push(&text[last..]);
    pieces
        .into_iter()
        .map(str::trim)
        .filter(|p| !p.is_empty())
        .map(String::from)
        .collect()
}

/// Join tokens back into text, undoing the spaces added before punctuation.
///
/// Not a perfect inverse: the original spacing is lost.
pub fn detokenize(tokens: &[String]) -> String {
    DETOKENIZE_RE
        .replace_all(&tokens.join(" "), "$1")
        .into_owned()
}

    size_t len;

C-version/tokenization/preprocess.c
#include "preprocess.h"

#include <ctype.h>
#include <stdbool.h>
#include <stdlib.h>
#include <string.h>

#include "util.h"

/* Delimiters that become tokens of their own. "--" and whitespace are handled
 * in split_text. Same set as the Python regex: , . ? _ ! " ( ) ' */
#define SPLIT_PUNCT ",.?_!\"()'"
/* Punctuation that detokenize glues to the previous word. */
#define DETOKENIZE_PUNCT ",.?!\"()'"

void tokens_push(Tokens *t, char *owned) {
    if (t->len == t->cap) {
        t->cap = t->cap ? t->cap * 2 : 64;
        t->items = xrealloc(t->items, t->cap * sizeof(char *));
    }
    t->items[t->len++] = owned;
}

void tokens_free(Tokens *t) {
    for (size_t i = 0; i < t->len; i++) free(t->items[i]);
    free(t->items);
    *t = (Tokens){0};
}

/* There is no regex library in C, so we scan by hand: walk the characters and
 * cut a token whenever we meet whitespace, "--" or a punctuation mark. */
Tokens split_text(const char *text) {
    Tokens out = {0};
    size_t n = strlen(text), word_start = 0;

#define FLUSH(end)                                                    \
    do {                                                              \
        if ((end) > word_start)                                       \
            tokens_push(&out, xstrndup(text + word_start, (end) - word_start)); \
    } while (0)

    for (size_t i = 0; i < n; i++) {
        unsigned char c = (unsigned char)text[i];
        if (isspace(c)) {
            FLUSH(i);
            word_start = i + 1;
        } else if (c == '-' && text[i + 1] == '-') {
            FLUSH(i);
            tokens_push(&out, xstrdup("--"));
            i++;
            word_start = i + 1;
        } else if (strchr(SPLIT_PUNCT, c)) {
            FLUSH(i);
            tokens_push(&out, xstrndup(text + i, 1));
            word_start = i + 1;
        }
    }
    FLUSH(n);
#undef FLUSH
    return out;
}

char *detokenize(const char *const *tokens, size_t n) {
    size_t total = 1;
    for (size_t i = 0; i < n; i++) total += strlen(tokens[i]) + 1;
    char *out = xmalloc(total), *p = out;
    for (size_t i = 0; i < n; i++) {
        size_t len = strlen(tokens[i]);
        bool glue = len == 1 && strchr(DETOKENIZE_PUNCT, tokens[i][0]);
        if (i > 0 && !glue) *p++ = ' ';
        memcpy(p, tokens[i], len);
        p += len;
    }
    *p = '\0';
    return out;
}
Zig-version/src/tokenization/preprocess.zig
//! Step 1 (book 2.2) - split raw text into tokens, and join them back.
//!
//! ```
//! "Hello, world!"  ->  ["Hello", ",", "world", "!"]
//! ```
//!
//! Tip: start reading here, then vocab -> simple -> dataset -> dataloader ->
//! embedding. Each file uses the one before it. bpe is the bonus (book 2.5).

const std = @import("std");

/// Special tokens. They are not produced by splitting text; they are added on purpose.
pub const END_OF_TEXT = "<|endoftext|>";
pub const UNK_TOKEN = "<|unk|>";

/// Delimiters that become tokens of their own. "--" and whitespace are handled
/// in `splitText`. Same set as the Python regex: , . ? _ ! " ( ) '
const split_punct = ",.?_!\"()'";
/// Punctuation that `detokenize` glues to the previous word.
const detokenize_punct = ",.?!\"()'";

fn flush(gpa: std.mem.Allocator, out: *std.ArrayList([]const u8), text: []const u8, start: usize, end: usize) !void {
    if (end > start) try out.append(gpa, text[start..end]);
}

/// Return the tokens of `text`; whitespace-only pieces are dropped.
///
/// Tip: tokens are *slices into `text`*, nothing is copied. They are only
/// valid while `text` is alive. The returned list is owned by the caller.
///
/// There is no regex library in the standard library, so we scan by hand: walk the
/// bytes and cut a token whenever we meet whitespace, "--" or a punctuation mark.
pub fn splitText(gpa: std.mem.Allocator, text: []const u8) ![]const []const u8 {
    var out: std.ArrayList([]const u8) = .empty;
    errdefer out.deinit(gpa);

    var word_start: usize = 0;
    var i: usize = 0;
    while (i < text.len) : (i += 1) {
        const c = text[i];
        if (std.ascii.isWhitespace(c)) {
            try flush(gpa, &out, text, word_start, i);
            word_start = i + 1;
        } else if (c == '-' and i + 1 < text.len and text[i + 1] == '-') {
            try flush(gpa, &out, text, word_start, i);
            try out.append(gpa, text[i .. i + 2]);
            i += 1;
            word_start = i + 1;
        } else if (std.mem.indexOfScalar(u8, split_punct, c) != null) {
            try flush(gpa, &out, text, word_start, i);
            try out.append(gpa, text[i .. i + 1]);
            word_start = i + 1;
        }
    }
    try flush(gpa, &out, text, word_start, text.len);
    return out.toOwnedSlice(gpa);
}

/// Join tokens with spaces, then remove the space before punctuation.
/// Not a perfect inverse: the original spacing is lost. Caller frees.
pub fn detokenize(gpa: std.mem.Allocator, tokens: []const []const u8) ![]u8 {
    var out: std.ArrayList(u8) = .empty;
    errdefer out.deinit(gpa);
    for (tokens, 0..) |token, i| {
        const glue = token.len == 1 and std.mem.indexOfScalar(u8, detokenize_punct, token[0]) != null;
        if (i > 0 and !glue) try out.append(gpa, ' ');
        try out.appendSlice(gpa, token);
    }
    return out.toOwnedSlice(gpa);
}

test "splitText keeps punctuation as tokens" {
    const gpa = std.testing.allocator;
    const tokens = try splitText(gpa, "Hello, world!");
    defer gpa.free(tokens);
    try std.testing.expectEqual(@as(usize, 4), tokens.len);
    try std.testing.expectEqualStrings("Hello", tokens[0]);
    try std.testing.expectEqualStrings(",", tokens[1]);
    try std.testing.expectEqualStrings("!", tokens[3]);
}

test "double dash is one token" {
    const gpa = std.testing.allocator;
    const tokens = try splitText(gpa, "Is this-- a test?");
    defer gpa.free(tokens);
    try std.testing.expectEqualStrings("--", tokens[2]);
}

test "detokenize removes space before punctuation" {
    const gpa = std.testing.allocator;
    const text = try detokenize(gpa, &.{ "Hello", ",", "world", "!" });
    defer gpa.free(text);
    try std.testing.expectEqualStrings("Hello, world!", text);
}
