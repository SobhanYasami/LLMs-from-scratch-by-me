3.4: خودتوجهی با وزن‌های آموزش‌پذیر

بخش کتاب 3.4 · کد به زبان Python, Go, Rust, C, Zig

هدف. سه ماتریس وزن اضافه کنید تا مدل بتواند یاد بگیرد دنبال چه بگردد، و ببینید چرا امتیازها بر جذر بُعد کلید تقسیم می‌شوند.

این سازوکارِ درون ترنسفورمر اصلی و هر GPT است و توجه ضرب داخلی مقیاس‌شده نام دارد. ساختار همان 3.3 است. تفاوت: بردارهایی که مقایسه می‌کنیم دیگر امبدینگ‌های خام نیستند بلکه تصویرهای آن‌ها هستند.

Query، key، value

نقشمحاسبهچگونه به آن فکر کنیم
queryx @ W_query«دنبال چه می‌گردم؟» (یک پرس‌وجو)
keyx @ W_key«چه چیزی به جست‌وجوها عرضه می‌کنم؟» (یک نمایه)
valuex @ W_value«اگر انتخاب شوم چه چیزی تحویل می‌دهم؟» (محتوای ذخیره‌شده)

نام‌ها از پایگاه‌داده می‌آیند: یک query را با key ها تطبیق می‌دهید، سپس value های متناظر را می‌خوانید. اینجا مدل W_query، W_key و W_value را در آموزش یاد می‌گیرد.

گام‌به‌گام، برای توکن x2

ورودی‌های سه‌بعدی را به d_out = 2 می‌بریم. (در مدل‌های شبیه GPT معمولاً d_in و d_out برابرند؛ اندازه‌های متفاوت فقط دنبال کردن شکل‌ها را آسان‌تر می‌کند.) وزن‌های زیر وزن‌های کتاب‌اند (torch.manual_seed(123)).

query_2 = x2 @ W_query = [0.4306, 1.4551]
keys   shape (6, 2)     values shape (6, 2)     (all tokens, projected)

scores_2 = query_2 @ keys.T
         = [1.2705, 1.8524, 1.8111, 1.0795, 0.5577, 1.5440]

weights_2 = softmax(scores_2 / sqrt(d_k))      d_k = 2
          = [0.1500, 0.2264, 0.2199, 0.1311, 0.0906, 0.1820]

context_2 = weights_2 @ values = [0.3061, 0.8210]

فقط منبع بردارها عوض شد: امتیازها query را با key مقایسه می‌کنند (نه ورودی را با ورودی)، و زمینه value ها را می‌آمیزد (نه ورودی‌های خام را).

چرا بر sqrt(d_k) تقسیم می‌کنیم؟

ضرب داخلی با افزایش تعداد بُعدها بزرگ می‌شود. امتیازهای بزرگ softmax را به توزیع «یک برنده» سخت می‌کشانند که در آن گرادیان‌ها تقریباً از بین می‌روند و یادگیری متوقف می‌شود. تقسیم بر جذر بُعد کلید امتیازها را در بازه‌ای نگه می‌دارد که softmax نرم بماند. مطلب تکمیلی دربارهٔ مقیاس عددهای واقعی را نشان می‌دهد.

پارامترهای وزن (W_query، W_key، W_value: یادگرفته‌شده و پس از آموزش ثابت) را با وزن‌های توجه (خروجی softmax: برای هر ورودی تازه محاسبه می‌شود) اشتباه نگیرید.

"""Step 7 (book 3.4) - self-attention with trainable weights (scaled dot-product).

Three weight matrices turn each input vector into three roles:

    query = x @ W_query    "what am I looking for?"
    key   = x @ W_key      "what do I offer to a search?"
    value = x @ W_value    "what do I hand over if I am chosen?"

    scores  = query @ key.T                 how well does each query match each key?
    weights = softmax(scores / sqrt(d_k))   scale, then normalise each row
    context = weights @ value               mix the values

Why divide by sqrt(d_k)? Big dot products push softmax towards a hard 0/1
choice, which makes gradients tiny. Scaling keeps training healthy.

Tip: do not confuse *weight parameters* (W_*, learned) with *attention weights*
(softmax output, computed fresh for every input).
"""

import math

import torch
from torch import nn

from .rng import Rng


class SelfAttention(nn.Module):
    def __init__(self, w_query: torch.Tensor, w_key: torch.Tensor, w_value: torch.Tensor):
        super().__init__()
        self.W_query = nn.Parameter(w_query)
        self.W_key = nn.Parameter(w_key)
        self.W_value = nn.Parameter(w_value)

    @classmethod
    def random(cls, d_in: int, d_out: int, rng: Rng) -> "SelfAttention":
        """Weights drawn uniform in [0, 1), in query, key, value order."""
        return cls(rng.rand(d_in, d_out), rng.rand(d_in, d_out), rng.rand(d_in, d_out))

    def attention_weights(self, x: torch.Tensor) -> torch.Tensor:
        queries, keys = x @ self.W_query, x @ self.W_key
        scores = queries @ keys.T
        return torch.softmax(scores / math.sqrt(keys.shape[-1]), dim=-1)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        """``x`` is [T, d_in]; returns context vectors [T, d_out]."""
        return self.attention_weights(x) @ (x @ self.W_value)

package attention

import "math"

// Step 7 (book 3.4) - self-attention with trainable weights (scaled dot-product).
//
// Three weight matrices turn each input vector into three roles:
//
//	query = x x W_query    "what am I looking for?"
//	key   = x x W_key      "what do I offer to a search?"
//	value = x x W_value    "what do I hand over if I am chosen?"
//
//	scores  = query x key^T                 how well does each query match each key?
//	weights = softmax(scores / sqrt(d_k))   scale, then normalise each row
//	context = weights x value               mix the values
//
// Why divide by sqrt(d_k)? Big dot products push softmax towards a hard 0/1
// choice, which makes gradients tiny. Scaling keeps training healthy.
//
// Tip: do not confuse weight parameters (W_*, learned) with attention weights
// (softmax output, computed fresh for every input).

type SelfAttention struct {
	WQuery, WKey, WValue Matrix
}

// NewRandomSelfAttention draws weights uniform in [0, 1), in query, key, value order.
func NewRandomSelfAttention(dIn, dOut int, rng *Rng) *SelfAttention {
	return &SelfAttention{rng.Rand(dIn, dOut), rng.Rand(dIn, dOut), rng.Rand(dIn, dOut)}
}

func scaleBySqrt(scores Matrix, dK int) Matrix {
	return Scale(scores, 1/float32(math.Sqrt(float64(dK))))
}

// AttentionWeights takes x [T, d_in] and returns [T, T] weights.
func (a *SelfAttention) AttentionWeights(x Matrix) Matrix {
	queries, keys := MatMul(x, a.WQuery), MatMul(x, a.WKey)
	scores := MatMul(queries, Transpose(keys))
	return SoftmaxRows(scaleBySqrt(scores, keys.Cols))
}

// Forward returns context vectors [T, d_out].
func (a *SelfAttention) Forward(x Matrix) Matrix {
	return MatMul(a.AttentionWeights(x), MatMul(x, a.WValue))
}

//! Step 7 (book 3.4) - self-attention with trainable weights (scaled dot-product).
//!
//! Three weight matrices turn each input vector into three roles:
//!
//! ```text
//! query = x x W_query    "what am I looking for?"
//! key   = x x W_key      "what do I offer to a search?"
//! value = x x W_value    "what do I hand over if I am chosen?"
//!
//! scores  = query x key^T                 how well does each query match each key?
//! weights = softmax(scores / sqrt(d_k))   scale, then normalise each row
//! context = weights x value               mix the values
//! ```
//!
//! Why divide by sqrt(d_k)? Big dot products push softmax towards a hard 0/1
//! choice, which makes gradients tiny. Scaling keeps training healthy.
//!
//! Tip: do not confuse *weight parameters* (W_*, learned) with *attention weights*
//! (softmax output, computed fresh for every input).

use super::Matrix;
use crate::tokenization::rng::Rng;

pub struct SelfAttention {
    pub w_query: Matrix,
    pub w_key: Matrix,
    pub w_value: Matrix,
}

impl SelfAttention {
    /// Weights drawn uniform in [0, 1), in query, key, value order.
    pub fn random(d_in: usize, d_out: usize, rng: &mut Rng) -> Self {
        let w_query = Matrix::rand(d_in, d_out, rng);
        let w_key = Matrix::rand(d_in, d_out, rng);
        let w_value = Matrix::rand(d_in, d_out, rng);
        Self {
            w_query,
            w_key,
            w_value,
        }
    }

    /// `x` is [T, d_in]; returns [T, T] weights.
    pub fn attention_weights(&self, x: &Matrix) -> Matrix {
        let queries = x.matmul(&self.w_query);
        let keys = x.matmul(&self.w_key);
        let scores = queries.matmul(&keys.transpose());
        scale_by_sqrt(&scores, keys.cols).softmax_rows()
    }

    /// Context vectors [T, d_out].
    pub fn forward(&self, x: &Matrix) -> Matrix {
        self.attention_weights(x).matmul(&x.matmul(&self.w_value))
    }
}

/// Divide scores by sqrt(d_k).
pub(super) fn scale_by_sqrt(scores: &Matrix, d_k: usize) -> Matrix {
    scores.scale(1.0 / (d_k as f64).sqrt() as f32)
}

#include "self_attention.h"

#include <math.h>

SelfAttention self_attention_random(Arena *a, size_t d_in, size_t d_out, Rng *rng) {
    SelfAttention sa;
    sa.w_query = matrix_rand(a, d_in, d_out, rng);
    sa.w_key = matrix_rand(a, d_in, d_out, rng);
    sa.w_value = matrix_rand(a, d_in, d_out, rng);
    return sa;
}

Matrix scale_by_sqrt(Arena *a, Matrix scores, size_t d_k) { return scale(a, scores, 1.0f / (float)sqrt((double)d_k)); }

Matrix self_attention_weights(Arena *a, const SelfAttention *sa, Matrix x) {
    Matrix queries = matmul(a, x, sa->w_query), keys = matmul(a, x, sa->w_key);
    Matrix scores = matmul(a, queries, transpose(a, keys));
    return softmax_rows(a, scale_by_sqrt(a, scores, keys.cols));
}

Matrix self_attention_forward(Arena *a, const SelfAttention *sa, Matrix x) {
    return matmul(a, self_attention_weights(a, sa, x), matmul(a, x, sa->w_value));
}
Zig-version/src/attention/self_attention.zig
//! Step 7 (book 3.4) - self-attention with trainable weights (scaled dot-product).
//!
//! Three weight matrices turn each input vector into three roles:
//!
//! ```
//! query = x x W_query    "what am I looking for?"
//! key   = x x W_key      "what do I offer to a search?"
//! value = x x W_value    "what do I hand over if I am chosen?"
//!
//! scores  = query x key^T                 how well does each query match each key?
//! weights = softmax(scores / sqrt(d_k))   scale, then normalise each row
//! context = weights x value               mix the values
//! ```
//!
//! Why divide by sqrt(d_k)? Big dot products push softmax towards a hard 0/1
//! choice, which makes gradients tiny. Scaling keeps training healthy.
//!
//! Tip: do not confuse *weight parameters* (W_*, learned) with *attention weights*
//! (softmax output, computed fresh for every input).

const std = @import("std");
const Matrix = @import("matrix.zig").Matrix;
const Rng = @import("../tokenization/rng.zig").Rng;

pub const SelfAttention = struct {
    w_query: Matrix,
    w_key: Matrix,
    w_value: Matrix,

    /// Weights drawn uniform in [0, 1), in query, key, value order.
    pub fn random(gpa: std.mem.Allocator, d_in: usize, d_out: usize, rng: *Rng) !SelfAttention {
        const w_query = try Matrix.rand(gpa, d_in, d_out, rng);
        const w_key = try Matrix.rand(gpa, d_in, d_out, rng);
        const w_value = try Matrix.rand(gpa, d_in, d_out, rng);
        return .{ .w_query = w_query, .w_key = w_key, .w_value = w_value };
    }

    /// `x` is [T, d_in]; returns [T, T] weights.
    pub fn attentionWeights(self: SelfAttention, gpa: std.mem.Allocator, x: Matrix) !Matrix {
        const queries = try x.matmul(gpa, self.w_query);
        const keys = try x.matmul(gpa, self.w_key);
        const scores = try queries.matmul(gpa, try keys.transpose(gpa));
        return (try scaleBySqrt(gpa, scores, keys.cols)).softmaxRows(gpa);
    }

    /// Context vectors [T, d_out].
    pub fn forward(self: SelfAttention, gpa: std.mem.Allocator, x: Matrix) !Matrix {
        const weights = try self.attentionWeights(gpa, x);
        return weights.matmul(gpa, try x.matmul(gpa, self.w_value));
    }
};

/// scores / sqrt(d_k), shared by the attention variants.
pub fn scaleBySqrt(gpa: std.mem.Allocator, scores: Matrix, d_k: usize) !Matrix {
    const root: f32 = @floatCast(@sqrt(@as(f64, @floatFromInt(d_k))));
    return scores.scale(gpa, 1.0 / root);
}

دو کلاس کتاب

کتاب این را به‌صورت SelfAttention_v1 (با nn.Parameter(torch.rand(...)) خام) و SelfAttention_v2 (با nn.Linear بدون بایاس) کد می‌کند. هر دو یک چیز را حساب می‌کنند؛ v2 فقط مقداردهی اولیهٔ بهتری دارد، به همین دلیل با بذرهای متفاوت عددهای متفاوتی چاپ می‌کنند. SelfAttention ما به سبک v1 است و وزن‌ها را از بیرون می‌گیرد تا همهٔ زبان‌ها بتوانند عددهای یکسان داشته باشند.

تمرین‌ها

  1. تمرین ۳٫۱ کتاب: وزن‌های یک SelfAttention_v2 را در یک v1 کپی کنید تا هر دو خروجی یکسان بدهند. (راهنمایی: nn.Linear ماتریس وزن را ترانهاده ذخیره می‌کند.)
  2. d_out را از 2 به 4 تغییر دهید و شکل‌ها را ببینید. چه چیزی تغییر نمی‌کند؟
پاسخ ۲وزن‌های توجه همان (6 x 6) می‌مانند: به تعداد توکن‌ها بستگی دارند نه به d_out. فقط key ها، value ها و بردارهای زمینه پهن‌تر می‌شوند.
