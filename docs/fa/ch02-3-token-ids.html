2.3: تبدیل توکن‌ها به شناسه

بخش کتاب 2.3 · کد به زبان Python, Go, Rust, C, Zig

هدف. به هر توکن یکتا یک عدد بدهید (واژگان یا vocabulary)، سپس توکن‌سازی با encode (متن به شناسه) و decode (شناسه به متن) بسازید.

واژگان

کل متن آموزشی را توکن‌سازی کن، تکراری‌ها را حذف کن، به ترتیب الفبا مرتب کن و از ۰ شماره بزن. داستان 1,159 توکن یکتا دارد.

('!', 0)  ('"', 1)  ("'", 2)  ...  ('Has', 49)  ('He', 50)
  • مرتب‌سازی می‌کنیم تا یک پیکره همیشه شناسه‌های یکسان بدهد. مجموعه‌ها و نگاشت‌ها در هر سه زبان ترتیب تصادفی دارند.
  • برای رمزگشایی نگاشت معکوس هم لازم است، یعنی شناسه به توکن.

"""Step 2 - Build a vocabulary: a mapping from each token to a unique integer ID.

    vocab = {"!": 0, '"': 1, ..., "<|endoftext|>": 1159, "<|unk|>": 1160}

Computers work with numbers, so every token must get an ID before a model
can use it. The vocabulary is the lookup table between the two worlds.

Tip: a Python ``set`` has no fixed order. Sorting before numbering is what makes
the IDs the same on every run.
"""

from .preprocess import END_OF_TEXT, UNK_TOKEN

SPECIAL_TOKENS = [END_OF_TEXT, UNK_TOKEN]


def build_vocab(tokens: list[str]) -> dict[str, int]:
    """Assign IDs to the unique tokens in sorted order, then append special tokens.

    Sorting makes the IDs deterministic: the same corpus always gives the same vocab.
    Special tokens go last so the regular word IDs stay stable.
    """
    all_tokens = sorted(set(tokens))
    all_tokens.extend(SPECIAL_TOKENS)
    return {token: integer for integer, token in enumerate(all_tokens)}
python-version/tokenization/simple.py
"""Step 3 - Simple tokenizers that combine splitting with the vocabulary.

Both classes expose the same two methods:

    encode(text) -> list[int]   text  -> token IDs
    decode(ids)  -> str         IDs   -> text

V1 only knows the words in its vocabulary and raises ``KeyError`` otherwise.
V2 replaces unknown words with ``<|unk|>`` so encoding never fails.
"""

from .preprocess import UNK_TOKEN, detokenize, split_text


def _invert(vocab: dict[str, int]) -> dict[int, str]:
    """Swap keys and values: ID -> token, needed for decoding."""
    return {integer: token for token, integer in vocab.items()}


class SimpleTokenizerV1:
    """Encode and decode with a vocabulary; unknown tokens are an error."""

    def __init__(self, vocab: dict[str, int]):
        self.str_to_int = vocab              # token -> ID
        self.int_to_str = _invert(vocab)     # ID -> token

    def encode(self, text: str) -> list[int]:
        tokens = split_text(text)
        return [self.str_to_int[token] for token in tokens]  # KeyError if unknown

    def decode(self, ids: list[int]) -> str:
        tokens = [self.int_to_str[i] for i in ids]
        return detokenize(tokens)


class SimpleTokenizerV2(SimpleTokenizerV1):
    """Same as V1, but unknown tokens become ``<|unk|>`` instead of raising."""

    def encode(self, text: str) -> list[int]:
        tokens = split_text(text)
        tokens = [
            token if token in self.str_to_int else UNK_TOKEN
            for token in tokens
        ]
        return [self.str_to_int[token] for token in tokens]
go-version/tokenization/vocab.go
package tokenization

import "sort"

// Step 2 - build a vocabulary: a mapping from each token to a unique integer ID.
//
//	{"!": 0, "\"": 1, ..., "<|endoftext|>": 1159, "<|unk|>": 1160}
//
// Models only understand numbers, so this table is the bridge between text and IDs.
//
// Tip: a Go map has random iteration order. Sorting before numbering is what
// makes the IDs the same on every run.

// SpecialTokens are appended after the regular tokens so word IDs stay stable.
var SpecialTokens = []string{EndOfText, Unknown}

// BuildVocab assigns IDs to the unique tokens in sorted order, then appends the
// special tokens. Sorting makes IDs deterministic for a given corpus.
func BuildVocab(tokens []string) map[string]int {
	set := make(map[string]struct{})
	for _, t := range tokens {
		set[t] = struct{}{}
	}
	all := make([]string, 0, len(set)+len(SpecialTokens))
	for t := range set {
		all = append(all, t)
	}
	sort.Strings(all)
	all = append(all, SpecialTokens...)

	vocab := make(map[string]int, len(all))
	for i, t := range all {
		vocab[t] = i
	}
	return vocab
}

package tokenization

import "fmt"

// Step 3 - simple tokenizers: split text, then look each token up in the vocab.
//
//	encode: "Hello, world" -> ["Hello", ",", "world"] -> [1131, 5, 987]
//	decode: [1131, 5, 987] -> ["Hello", ",", "world"] -> "Hello, world"
//
// Tip: compare the two types. V1 fails on a word it never saw; V2 swaps it for
// <|unk|> and keeps going. That one difference is why real models use BPE.

// SimpleTokenizerV1 maps known tokens to IDs and fails on unknown tokens.
type SimpleTokenizerV1 struct {
	strToInt map[string]int // token -> ID, used by Encode
	intToStr map[int]string // ID -> token, used by Decode (the vocab, inverted)
}

func NewSimpleTokenizerV1(vocab map[string]int) *SimpleTokenizerV1 {
	intToStr := make(map[int]string, len(vocab))
	for s, i := range vocab {
		intToStr[i] = s
	}
	return &SimpleTokenizerV1{strToInt: vocab, intToStr: intToStr}
}

// encode is shared by V1 and V2. The only difference is what happens to a
// token that is not in the vocab: error (V1) or <|unk|> (V2).
func (t *SimpleTokenizerV1) encode(text string, unknownToUnk bool) ([]int, error) {
	var ids []int
	for _, token := range SplitText(text) {
		id, ok := t.strToInt[token]
		if !ok && unknownToUnk {
			id, ok = t.strToInt[Unknown]
		}
		if !ok {
			return nil, fmt.Errorf("token not in vocab: %q", token)
		}
		ids = append(ids, id)
	}
	return ids, nil
}

func (t *SimpleTokenizerV1) Encode(text string) ([]int, error) { return t.encode(text, false) }

func (t *SimpleTokenizerV1) Decode(ids []int) (string, error) {
	tokens := make([]string, 0, len(ids))
	for _, id := range ids {
		token, ok := t.intToStr[id]
		if !ok {
			return "", fmt.Errorf("id not in vocab: %d", id)
		}
		tokens = append(tokens, token)
	}
	return Detokenize(tokens), nil
}

// SimpleTokenizerV2 is V1, except unknown tokens become <|unk|>.
// It embeds V1, so Decode is inherited unchanged.
type SimpleTokenizerV2 struct {
	SimpleTokenizerV1
}

func NewSimpleTokenizerV2(vocab map[string]int) *SimpleTokenizerV2 {
	return &SimpleTokenizerV2{SimpleTokenizerV1: *NewSimpleTokenizerV1(vocab)}
}

func (t *SimpleTokenizerV2) Encode(text string) ([]int, error) { return t.encode(text, true) }

//! Step 2 - build a vocabulary: a mapping from each token to a unique integer ID.
//!
//! ```text
//! {"!": 0, "\"": 1, ..., "<|endoftext|>": 1159, "<|unk|>": 1160}
//! ```
//!
//! Models only understand numbers, so this table is the bridge between text and IDs.
//!
//! Tip: a `HashMap` has random iteration order. Collecting into a sorted
//! `BTreeSet` before numbering is what makes IDs the same on every run.

use std::collections::{BTreeSet, HashMap};

use super::preprocess::{END_OF_TEXT, UNK_TOKEN};

pub const SPECIAL_TOKENS: [&str; 2] = [END_OF_TEXT, UNK_TOKEN];

/// Assign IDs to the unique tokens in sorted order, then append special tokens.
///
/// Sorting makes IDs deterministic for a given corpus. Special tokens go last so
/// regular word IDs stay stable.
pub fn build_vocab(tokens: &[String]) -> HashMap<String, usize> {
    let unique: BTreeSet<&str> = tokens.iter().map(String::as_str).collect();
    unique
        .into_iter()
        .chain(SPECIAL_TOKENS)
        .enumerate()
        .map(|(i, t)| (t.to_string(), i))
        .collect()
}

//! Step 3 - simple tokenizers: split text, then look each token up in the vocab.
//!
//! ```text
//! encode: "Hello, world" -> ["Hello", ",", "world"] -> [1131, 5, 987]
//! decode: [1131, 5, 987] -> ["Hello", ",", "world"] -> "Hello, world"
//! ```
//!
//! Tip: compare the two types. V1 fails on a word it never saw; V2 swaps it for
//! `<|unk|>` and keeps going. That one difference is why real models use BPE.

use std::collections::HashMap;

use super::preprocess::{UNK_TOKEN, detokenize, split_text};

/// Encode and decode with a vocabulary; unknown tokens are an error.
pub struct SimpleTokenizerV1 {
    str_to_int: HashMap<String, usize>, // token -> ID, used by encode
    int_to_str: HashMap<usize, String>, // ID -> token, used by decode (the vocab, inverted)
}

impl SimpleTokenizerV1 {
    pub fn new(vocab: HashMap<String, usize>) -> Self {
        let int_to_str = vocab.iter().map(|(s, &i)| (i, s.clone())).collect();
        Self {
            str_to_int: vocab,
            int_to_str,
        }
    }

    /// Shared by V1 and V2. The only difference is what happens to a token that
    /// is not in the vocab: error (V1) or `<|unk|>` (V2).
    fn encode_with(&self, text: &str, unknown_to_unk: bool) -> Result<Vec<usize>, String> {
        split_text(text)
            .iter()
            .map(|token| {
                let known = self.str_to_int.get(token);
                let id = match (known, unknown_to_unk) {
                    (None, true) => self.str_to_int.get(UNK_TOKEN),
                    _ => known,
                };
                id.copied()
                    .ok_or_else(|| format!("token not in vocab: {token:?}"))
            })
            .collect()
    }

    pub fn encode(&self, text: &str) -> Result<Vec<usize>, String> {
        self.encode_with(text, false)
    }

    pub fn decode(&self, ids: &[usize]) -> Result<String, String> {
        let tokens = ids
            .iter()
            .map(|id| {
                self.int_to_str
                    .get(id)
                    .cloned()
                    .ok_or_else(|| format!("id not in vocab: {id}"))
            })
            .collect::<Result<Vec<_>, _>>()?;
        Ok(detokenize(&tokens))
    }
}

/// V1, except unknown tokens become `<|unk|>`. Wraps V1, so decode is reused.
pub struct SimpleTokenizerV2(SimpleTokenizerV1);

impl SimpleTokenizerV2 {
    pub fn new(vocab: HashMap<String, usize>) -> Self {
        Self(SimpleTokenizerV1::new(vocab))
    }

    pub fn encode(&self, text: &str) -> Result<Vec<usize>, String> {
        self.0.encode_with(text, true)
    }

    pub fn decode(&self, ids: &[usize]) -> Result<String, String> {
        self.0.decode(ids)
    }
}

/* Step 2 (book 2.3) - build a vocabulary: token -> unique integer ID.
 *
 *   {"!": 0, "\"": 1, ..., "<|endoftext|>": 1159, "<|unk|>": 1160}
 *
 * Models only understand numbers, so this table is the bridge between text
 * and IDs. Tip: sorting before numbering makes IDs identical on every run.
 */
#ifndef VOCAB_H
#define VOCAB_H

#include <stddef.h>

#include "preprocess.h"

typedef struct {
    char **tokens;   /* ID -> token, used by decode */
    size_t size;
    int *slots;      /* hash table of IDs (-1 = empty), used by encode */
    size_t n_slots;
} Vocab;

Vocab build_vocab(const Tokens *tokens);

C-version/tokenization/vocab.c
#include "vocab.h"

#include <stdint.h>
#include <stdlib.h>
#include <string.h>

#include "util.h"

static uint64_t hash_str(const char *s) { /* FNV-1a */
    uint64_t h = 1469598103934665603ULL;
    for (; *s; s++) h = (h ^ (unsigned char)*s) * 1099511628211ULL;
    return h;
}

static int cmp_str(const void *a, const void *b) {
    return strcmp(*(const char *const *)a, *(const char *const *)b);
}

Vocab build_vocab(const Tokens *tokens) {
    /* sort a copy of the pointers, then skip duplicates */
    char **sorted = xmalloc(tokens->len * sizeof(char *));
    memcpy(sorted, tokens->items, tokens->len * sizeof(char *));
    qsort(sorted, tokens->len, sizeof(char *), cmp_str);

    Vocab v = {0};
    v.tokens = xmalloc((tokens->len + 2) * sizeof(char *));
    for (size_t i = 0; i < tokens->len; i++)
        if (i == 0 || strcmp(sorted[i], sorted[i - 1]) != 0)
            v.tokens[v.size++] = xstrdup(sorted[i]);
    free(sorted);
    /* special tokens go last so regular word IDs stay stable */
    v.tokens[v.size++] = xstrdup(END_OF_TEXT);
    v.tokens[v.size++] = xstrdup(UNK_TOKEN);

    v.n_slots = 1;
    while (v.n_slots < v.size * 2) v.n_slots <<= 1;
    v.slots = xmalloc(v.n_slots * sizeof(int));
    for (size_t i = 0; i < v.n_slots; i++) v.slots[i] = -1;
    for (size_t id = 0; id < v.size; id++) {
        size_t s = hash_str(v.tokens[id]) & (v.n_slots - 1);
        while (v.slots[s] != -1) s = (s + 1) & (v.n_slots - 1);
        v.slots[s] = (int)id;
    }
    return v;
}

void vocab_free(Vocab *v) {
    for (size_t i = 0; i < v->size; i++) free(v->tokens[i]);
    free(v->tokens);
    free(v->slots);
    *v = (Vocab){0};
}

int vocab_id(const Vocab *v, const char *token) {
    size_t s = hash_str(token) & (v->n_slots - 1);
    while (v->slots[s] != -1) {
        if (strcmp(v->tokens[v->slots[s]], token) == 0) return v->slots[s];
        s = (s + 1) & (v->n_slots - 1);
    }
    return -1;
}

const char *vocab_token(const Vocab *v, int id) {
    return (id >= 0 && (size_t)id < v->size) ? v->tokens[id] : NULL;
}

#include "simple.h"

#include <stdio.h>
#include <stdlib.h>

#include "preprocess.h"

SimpleTokenizer simple_tokenizer_v1(const Vocab *vocab) { return (SimpleTokenizer){vocab, false}; }
SimpleTokenizer simple_tokenizer_v2(const Vocab *vocab) { return (SimpleTokenizer){vocab, true}; }

/* V1 and V2 share this code. The only difference is what happens to a token
 * that is not in the vocab: error (V1) or <|unk|> (V2). */
int *simple_encode(const SimpleTokenizer *t, const char *text, size_t *n, char err[ERR_LEN]) {
    Tokens tokens = split_text(text);
    int *ids = xmalloc(tokens.len * sizeof(int));
    for (size_t i = 0; i < tokens.len; i++) {
        int id = vocab_id(t->vocab, tokens.items[i]);
        if (id < 0 && t->unknown_to_unk) id = vocab_id(t->vocab, UNK_TOKEN);
        if (id < 0) {
            snprintf(err, ERR_LEN, "token not in vocab: \"%s\"", tokens.items[i]);
            free(ids);
            tokens_free(&tokens);
            return NULL;
        }
        ids[i] = id;
    }
    *n = tokens.len;
    tokens_free(&tokens);
    return ids;
}

char *simple_decode(const SimpleTokenizer *t, const int *ids, size_t n, char err[ERR_LEN]) {
    const char **tokens = xmalloc(n * sizeof(char *));
    for (size_t i = 0; i < n; i++) {
        tokens[i] = vocab_token(t->vocab, ids[i]);
        if (!tokens[i]) {
            snprintf(err, ERR_LEN, "id not in vocab: %d", ids[i]);
            free(tokens);
            return NULL;
        }
    }
    char *text = detokenize(tokens, n);
    free(tokens);
    return text;
}
Zig-version/src/tokenization/vocab.zig
//! Step 2 (book 2.3) - build a vocabulary: token -> unique integer ID.
//!
//! ```
//! {"!": 0, "\"": 1, ..., "<|endoftext|>": 1159, "<|unk|>": 1160}
//! ```
//!
//! Models only understand numbers, so this table is the bridge between text and IDs.
//!
//! Tip: a hash map has no fixed order. Sorting before numbering is what makes
//! IDs the same on every run.

const std = @import("std");
const preprocess = @import("preprocess.zig");

pub const Vocab = struct {
    /// ID -> token, used by decode.
    tokens: []const []const u8,
    /// token -> ID, used by encode.
    ids: std.StringHashMapUnmanaged(u32),

    /// Unique tokens in sorted order, then `<|endoftext|>` and `<|unk|>`
    /// (last, so regular word IDs stay stable). The vocab borrows the token
    /// strings: keep the text they came from alive.
    pub fn build(gpa: std.mem.Allocator, tokens: []const []const u8) !Vocab {
        const sorted = try gpa.dupe([]const u8, tokens);
        defer gpa.free(sorted);
        std.mem.sort([]const u8, sorted, {}, lessThan);

        var list: std.ArrayList([]const u8) = .empty;
        errdefer list.deinit(gpa);
        for (sorted, 0..) |token, i| {
            if (i == 0 or !std.mem.eql(u8, token, sorted[i - 1])) try list.append(gpa, token);
        }
        try list.append(gpa, preprocess.END_OF_TEXT);
        try list.append(gpa, preprocess.UNK_TOKEN);
        const all = try list.toOwnedSlice(gpa);
        errdefer gpa.free(all);

        var ids: std.StringHashMapUnmanaged(u32) = .empty;
        errdefer ids.deinit(gpa);
        try ids.ensureTotalCapacity(gpa, @intCast(all.len));
        for (all, 0..) |token, id| ids.putAssumeCapacity(token, @intCast(id));
        return .{ .tokens = all, .ids = ids };
    }

    pub fn deinit(self: *Vocab, gpa: std.mem.Allocator) void {
        gpa.free(self.tokens);
        self.ids.deinit(gpa);
    }

    pub fn size(self: Vocab) usize {
        return self.tokens.len;
    }

    /// token -> ID, or null if the token is not in the vocabulary.
    pub fn idOf(self: Vocab, text: []const u8) ?u32 {
        return self.ids.get(text);
    }

    /// ID -> token, or null if out of range.
    pub fn tokenOf(self: Vocab, id: u32) ?[]const u8 {
        return if (id < self.tokens.len) self.tokens[id] else null;
    }
};

fn lessThan(_: void, a: []const u8, b: []const u8) bool {
    return std.mem.order(u8, a, b) == .lt;
}

test "vocab is sorted with special tokens last" {
    const gpa = std.testing.allocator;
    var v = try Vocab.build(gpa, &.{ "b", "a", "b", "c" });
    defer v.deinit(gpa);
    try std.testing.expectEqual(@as(usize, 5), v.size());
    try std.testing.expectEqual(@as(?u32, 0), v.idOf("a"));
    try std.testing.expectEqual(@as(?u32, 3), v.idOf(preprocess.END_OF_TEXT));
    try std.testing.expectEqual(@as(?u32, 4), v.idOf(preprocess.UNK_TOKEN));

//! Step 3 (book 2.3-2.4) - simple tokenizers: split text, look tokens up.
//!
//! ```
//! encode: "Hello, world" -> ["Hello", ",", "world"] -> [1131, 5, 987]
//! decode: [1131, 5, 987] -> ["Hello", ",", "world"] -> "Hello, world"
//! ```
//!
//! Tip: compare V1 and V2. V1 fails on a word it never saw; V2 swaps it for
//! `<|unk|>` and keeps going. That one difference is why real models use BPE.

const std = @import("std");
const preprocess = @import("preprocess.zig");
const Vocab = @import("vocab.zig").Vocab;

pub const SimpleTokenizer = struct {
    vocab: *const Vocab,
    /// false = V1 (error), true = V2 (`<|unk|>`).
    unknown_to_unk: bool,

    pub fn v1(vocab: *const Vocab) SimpleTokenizer {
        return .{ .vocab = vocab, .unknown_to_unk = false };
    }

    pub fn v2(vocab: *const Vocab) SimpleTokenizer {
        return .{ .vocab = vocab, .unknown_to_unk = true };
    }

    /// text -> IDs. V1 fails with `error.UnknownToken` on a word outside the
    /// vocabulary; V2 uses the `<|unk|>` ID instead. Caller frees.
    pub fn encode(self: SimpleTokenizer, gpa: std.mem.Allocator, text: []const u8) ![]u32 {
        const tokens = try preprocess.splitText(gpa, text);
        defer gpa.free(tokens);
        const ids = try gpa.alloc(u32, tokens.len);
        errdefer gpa.free(ids);
        for (tokens, ids) |token, *id| {
            id.* = self.vocab.idOf(token) orelse
                (if (self.unknown_to_unk) self.vocab.idOf(preprocess.UNK_TOKEN) else null) orelse
                return error.UnknownToken;
        }
        return ids;
    }

    /// IDs -> text. Fails with `error.UnknownId` for an ID outside the vocabulary. Caller frees.
    pub fn decode(self: SimpleTokenizer, gpa: std.mem.Allocator, ids: []const u32) ![]u8 {
        const tokens = try gpa.alloc([]const u8, ids.len);
        defer gpa.free(tokens);
        for (ids, tokens) |id, *token| token.* = self.vocab.tokenOf(id) orelse return error.UnknownId;
        return preprocess.detokenize(gpa, tokens);
    }
};

fn testVocab(gpa: std.mem.Allocator) !Vocab {
    return Vocab.build(gpa, &.{ "Hello", ",", "world" });
}

test "V1 fails on unknown words, V2 uses <|unk|>" {
    const gpa = std.testing.allocator;
    var vocab = try testVocab(gpa);
    defer vocab.deinit(gpa);

    try std.testing.expectError(error.UnknownToken, SimpleTokenizer.v1(&vocab).encode(gpa, "Hello there"));

    const ids = try SimpleTokenizer.v2(&vocab).encode(gpa, "Hello there");
    defer gpa.free(ids);
    try std.testing.expectEqual(vocab.idOf(preprocess.UNK_TOKEN).?, ids[1]);
}

test "encode then decode round-trips simple text" {
    const gpa = std.testing.allocator;
    var vocab = try testVocab(gpa);
    defer vocab.deinit(gpa);
    const t = SimpleTokenizer.v1(&vocab);
    const ids = try t.encode(gpa, "Hello, world");
    defer gpa.free(ids);
    const text = try t.decode(gpa, ids);
    defer gpa.free(text);
    try std.testing.expectEqualStrings("Hello, world", text);
}
