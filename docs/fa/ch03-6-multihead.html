3.6: گسترش به توجه چندسَری

بخش کتاب 3.6 · کد به زبان Python, Go, Rust, C, Zig

هدف. چند سَر توجه را کنار هم اجرا کنید تا مدل بتواند هم‌زمان دنبال الگوهای مختلف بگردد، و ببینید چگونه یک ماتریس بزرگ همهٔ آن‌ها را انجام می‌دهد.

یک سَر توجه یک شیوهٔ مرتبط کردن توکن‌ها را یاد می‌گیرد. توجه چندسَری چند سَر را موازی اجرا می‌کند، هرکدام با query، key و value خودش، تا یک سَر دستور زبان را دنبال کند و دیگری موضوع را. نتایجشان به هم چسبانده و با یک لایهٔ خطی نهایی آمیخته می‌شود.

نسخهٔ ساده: روی هم گذاشتن سَرها

کتاب نخست پوششی می‌سازد که چند ماژول CausalAttention را نگه می‌دارد و خروجی‌شان را به هم می‌چسباند. با دو سَر و d_out = 2 برای هرکدام، نتیجه ۴ ستون پهنا دارد. کار می‌کند اما سَرها را یکی‌یکی حساب می‌کند.

نسخهٔ کارآمد: یک ماتریس بزرگ، تقسیم‌شده به سَرها

به‌جای آن، یک W_query، W_key و W_value با پهنای کامل d_out نگه دارید و نتیجه را به num_heads برش با head_dim = d_out / num_heads ستون ببُرید.

d_out = 4, num_heads = 2  ->  head_dim = 2

queries [T, 4] = [ head 0 (cols 0-1) | head 1 (cols 2-3) ]

per head:  scores = q_h x k_h^T,  causal mask,  softmax(scores / sqrt(head_dim)),  weights x v_h
concat heads  ->  [T, 4]
out_proj      ->  [T, 4]   a linear layer that mixes the heads together

یک ضرب ماتریسی همهٔ سَرها را یک‌جا می‌سازد که بسیار سریع‌تر از حلقه روی ماژول‌هاست. در torch با view و transpose انجام می‌شود تا همهٔ سَرها به‌صورت یک matmul دسته‌ای اجرا شوند؛ در زبان‌های دیگر ما یک حلقهٔ کوچک روی برش ستون‌ها همان محاسبه را می‌کند. نتیجه یکی است.

توجه کنید مقیاس‌گذاری با head_dim انجام می‌شود، یعنی اندازه‌ای که هر سَر روی آن کار می‌کند، نه d_out.

شکل‌ها، با یک دسته

x                [batch, T, d_in]           (2, 6, 3)
x @ W_query      [batch, T, d_out]          (2, 6, 2)
split into heads [batch, heads, T, head_dim] (2, 2, 6, 1)
scores           [batch, heads, T, T]       (2, 2, 6, 6)
context          [batch, heads, T, head_dim], heads glued back -> [batch, T, d_out]
out_proj         [batch, T, d_out]          (2, 6, 2)

دموی ما

با وزن‌های تصادفی با بذر ثابت (d_in = 3، d_out = 2، دو سَر)، خروجی دنبالهٔ اول در دسته‌ای از دو نسخهٔ جمله:

"""Step 9 (book 3.6) - multi-head attention.

One attention head learns one way of relating tokens. Several heads, each
with its own smaller query/key/value, can look for different patterns in
parallel (one for grammar, one for topic, ...). Their outputs are joined
and mixed by a final linear layer.

    d_out = 4, num_heads = 2  ->  each head works on head_dim = 2 columns

    W_query columns: [ head 0 | head 1 ]     (same for W_key and W_value)
    per head: causal attention on its own slice -> [T, head_dim]
    concatenate heads -> [T, d_out] -> out_proj

Tip: one big weight matrix cut into column slices is the same as separate
small matrices per head, just faster (one matmul instead of many).
"""

import math

import torch
from torch import nn

from .rng import Rng


class MultiHeadAttention(nn.Module):
    def __init__(self, w_query, w_key, w_value, out_weight, out_bias, num_heads: int, context_length: int):
        super().__init__()
        d_out = w_query.shape[1]
        assert d_out % num_heads == 0, "d_out must be divisible by num_heads"
        self.num_heads = num_heads
        self.head_dim = d_out // num_heads
        self.W_query, self.W_key, self.W_value = (nn.Parameter(w) for w in (w_query, w_key, w_value))
        self.out_weight = nn.Parameter(out_weight)
        self.out_bias = nn.Parameter(out_bias)
        self.register_buffer("mask", torch.triu(torch.ones(context_length, context_length), diagonal=1).bool())

    @classmethod
    def random(cls, d_in: int, d_out: int, num_heads: int, context_length: int, rng: Rng) -> "MultiHeadAttention":
        """Uniform [0, 1) weights (query, key, value, out_proj); zero bias."""
        wq, wk, wv = rng.rand(d_in, d_out), rng.rand(d_in, d_out), rng.rand(d_in, d_out)
        return cls(wq, wk, wv, rng.rand(d_out, d_out), torch.zeros(d_out), num_heads, context_length)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        """``x`` is [batch, T, d_in]; returns [batch, T, d_out]."""
        b, t, _ = x.shape
        # [b, T, d_out] -> [b, T, heads, head_dim] -> [b, heads, T, head_dim]
        def split(m):
            return (x @ m).view(b, t, self.num_heads, self.head_dim).transpose(1, 2)

        q, k, v = split(self.W_query), split(self.W_key), split(self.W_value)
        scores = (q @ k.transpose(-2, -1)).masked_fill(self.mask[:t, :t], -math.inf)
        weights = torch.softmax(scores / math.sqrt(self.head_dim), dim=-1)
        context = (weights @ v).transpose(1, 2).reshape(b, t, -1)  # glue the heads back together
        return context @ self.out_weight + self.out_bias

package attention

// Step 9 (book 3.6) - multi-head attention.
//
// One attention head learns one way of relating tokens. Several heads, each
// with its own smaller query/key/value, can look for different patterns in
// parallel (one for grammar, one for topic, ...). Their outputs are joined
// and mixed by a final linear layer.
//
//	d_out = 4, num_heads = 2  ->  each head works on head_dim = 2 columns
//
//	W_query columns: [ head 0 | head 1 ]     (same for W_key and W_value)
//	per head: causal attention on its own slice -> [T, head_dim]
//	concatenate heads -> [T, d_out] -> out_proj
//
// Tip: one big weight matrix cut into column slices is the same as separate
// small matrices per head, just faster (one matmul instead of many).

type MultiHeadAttention struct {
	WQuery, WKey, WValue Matrix
	OutWeight            Matrix
	OutBias              []float32
	NumHeads, HeadDim    int
}

// NewRandomMultiHead draws uniform [0, 1) weights (query, key, value,
// out_proj) and a zero bias. dOut must divide evenly by numHeads.
func NewRandomMultiHead(dIn, dOut, numHeads int, rng *Rng) *MultiHeadAttention {
	wq, wk, wv := rng.Rand(dIn, dOut), rng.Rand(dIn, dOut), rng.Rand(dIn, dOut)
	return &MultiHeadAttention{wq, wk, wv, rng.Rand(dOut, dOut), make([]float32, dOut), numHeads, dOut / numHeads}
}

// Forward takes x [T, d_in] and returns [T, d_out].
func (m *MultiHeadAttention) Forward(x Matrix) Matrix {
	q, k, v := MatMul(x, m.WQuery), MatMul(x, m.WKey), MatMul(x, m.WValue)
	context := NewMatrix(x.Rows, m.NumHeads*m.HeadDim)
	for h := 0; h < m.NumHeads; h++ {
		lo, hi := h*m.HeadDim, (h+1)*m.HeadDim // this head's column slice
		qh, kh, vh := SliceCols(q, lo, hi), SliceCols(k, lo, hi), SliceCols(v, lo, hi)
		scores := MaskFuture(MatMul(qh, Transpose(kh)))
		weights := SoftmaxRows(scaleBySqrt(scores, m.HeadDim))
		head := MatMul(weights, vh)
		for i := 0; i < head.Rows; i++ { // glue the heads back together
			copy(context.Row(i)[lo:hi], head.Row(i))
		}
	}
	out := MatMul(context, m.OutWeight)
	for i := 0; i < out.Rows; i++ {
		for j := 0; j < out.Cols; j++ {
			out.Set(i, j, out.At(i, j)+m.OutBias[j])
		}
	}
	return out
}

//! Step 9 (book 3.6) - multi-head attention.
//!
//! One attention head learns one way of relating tokens. Several heads, each
//! with its own smaller query/key/value, can look for different patterns in
//! parallel (one for grammar, one for topic, ...). Their outputs are joined
//! and mixed by a final linear layer.
//!
//! ```text
//! d_out = 4, num_heads = 2  ->  each head works on head_dim = 2 columns
//!
//! W_query columns: [ head 0 | head 1 ]     (same for W_key and W_value)
//! per head: causal attention on its own slice -> [T, head_dim]
//! concatenate heads -> [T, d_out] -> out_proj
//! ```
//!
//! Tip: one big weight matrix cut into column slices is the same as separate
//! small matrices per head, just faster (one matmul instead of many).

use super::Matrix;
use super::self_attention::scale_by_sqrt;
use crate::tokenization::rng::Rng;

pub struct MultiHeadAttention {
    pub w_query: Matrix,
    pub w_key: Matrix,
    pub w_value: Matrix,
    pub out_weight: Matrix,
    pub out_bias: Vec<f32>,
    pub num_heads: usize,
    pub head_dim: usize,
}

impl MultiHeadAttention {
    /// Uniform [0, 1) weights (query, key, value, out_proj) and a zero bias.
    /// `d_out` must divide evenly by `num_heads`.
    pub fn random(d_in: usize, d_out: usize, num_heads: usize, rng: &mut Rng) -> Self {
        let w_query = Matrix::rand(d_in, d_out, rng);
        let w_key = Matrix::rand(d_in, d_out, rng);
        let w_value = Matrix::rand(d_in, d_out, rng);
        let out_weight = Matrix::rand(d_out, d_out, rng);
        Self {
            w_query,
            w_key,
            w_value,
            out_weight,
            out_bias: vec![0.0; d_out],
            num_heads,
            head_dim: d_out / num_heads,
        }
    }

    /// `x` is [T, d_in]; returns [T, d_out].
    pub fn forward(&self, x: &Matrix) -> Matrix {
        let (q, k, v) = (
            x.matmul(&self.w_query),
            x.matmul(&self.w_key),
            x.matmul(&self.w_value),
        );
        let mut context = Matrix::new(x.rows, self.num_heads * self.head_dim);
        for h in 0..self.num_heads {
            let (lo, hi) = (h * self.head_dim, (h + 1) * self.head_dim); // this head's column slice
            let (qh, kh, vh) = (
                q.slice_cols(lo, hi),
                k.slice_cols(lo, hi),
                v.slice_cols(lo, hi),
            );
            let scores = qh.matmul(&kh.transpose()).mask_future();
            let head = scale_by_sqrt(&scores, self.head_dim)
                .softmax_rows()
                .matmul(&vh);
            for i in 0..head.rows {
                // glue the heads back together
                context.data[i * context.cols + lo..i * context.cols + hi]
                    .copy_from_slice(head.row(i));
            }
        }
        let mut out = context.matmul(&self.out_weight);
        for i in 0..out.rows {
            for j in 0..out.cols {
                out.data[i * out.cols + j] += self.out_bias[j];
            }
        }
        out
    }
}

#include "multihead.h"

#include <string.h>

#include "self_attention.h"

MultiHeadAttention multihead_random(Arena *a, size_t d_in, size_t d_out, size_t num_heads, Rng *rng) {
    MultiHeadAttention m;
    m.w_query = matrix_rand(a, d_in, d_out, rng);
    m.w_key = matrix_rand(a, d_in, d_out, rng);
    m.w_value = matrix_rand(a, d_in, d_out, rng);
    m.out_weight = matrix_rand(a, d_out, d_out, rng);
    m.out_bias = matrix_new(a, 1, d_out);
    m.num_heads = num_heads;
    m.head_dim = d_out / num_heads;
    return m;
}

Matrix multihead_forward(Arena *a, const MultiHeadAttention *m, Matrix x) {
    Matrix q = matmul(a, x, m->w_query), k = matmul(a, x, m->w_key), v = matmul(a, x, m->w_value);
    Matrix context = matrix_new(a, x.rows, m->num_heads * m->head_dim);
    for (size_t h = 0; h < m->num_heads; h++) {
        size_t lo = h * m->head_dim, hi = lo + m->head_dim; /* this head's column slice */
        Matrix qh = slice_cols(a, q, lo, hi), kh = slice_cols(a, k, lo, hi), vh = slice_cols(a, v, lo, hi);
        Matrix scores = mask_future(a, matmul(a, qh, transpose(a, kh)));
        Matrix head = matmul(a, softmax_rows(a, scale_by_sqrt(a, scores, m->head_dim)), vh);
        for (size_t i = 0; i < head.rows; i++) /* glue the heads back together */
            memcpy(matrix_row(context, i) + lo, matrix_row(head, i), m->head_dim * sizeof(float));
    }
    Matrix out = matmul(a, context, m->out_weight);
    for (size_t i = 0; i < out.rows; i++)
        for (size_t j = 0; j < out.cols; j++) out.data[i * out.cols + j] += m->out_bias.data[j];
    return out;
}

//! Step 9 (book 3.6) - multi-head attention.
//!
//! One attention head learns one way of relating tokens. Several heads, each
//! with its own smaller query/key/value, can look for different patterns in
//! parallel (one for grammar, one for topic, ...). Their outputs are joined
//! and mixed by a final linear layer.
//!
//! ```
//! d_out = 4, num_heads = 2  ->  each head works on head_dim = 2 columns
//!
//! W_query columns: [ head 0 | head 1 ]     (same for W_key and W_value)
//! per head: causal attention on its own slice -> [T, head_dim]
//! concatenate heads -> [T, d_out] -> out_proj
//! ```
//!
//! Tip: one big weight matrix cut into column slices is the same as separate
//! small matrices per head, just faster (one matmul instead of many).

const std = @import("std");
const Matrix = @import("matrix.zig").Matrix;
const sa = @import("self_attention.zig");
const Rng = @import("../tokenization/rng.zig").Rng;

pub const MultiHeadAttention = struct {
    w_query: Matrix,
    w_key: Matrix,
    w_value: Matrix,
    out_weight: Matrix,
    out_bias: Matrix, // 1 x d_out
    num_heads: usize,
    head_dim: usize,

    /// Uniform [0, 1) weights (query, key, value, out_proj) and a zero bias.
    /// `d_out` must divide evenly by `num_heads`.
    pub fn random(gpa: std.mem.Allocator, d_in: usize, d_out: usize, num_heads: usize, rng: *Rng) !MultiHeadAttention {
        const w_query = try Matrix.rand(gpa, d_in, d_out, rng);
        const w_key = try Matrix.rand(gpa, d_in, d_out, rng);
        const w_value = try Matrix.rand(gpa, d_in, d_out, rng);
        const out_weight = try Matrix.rand(gpa, d_out, d_out, rng);
        return .{
            .w_query = w_query,
            .w_key = w_key,
            .w_value = w_value,
            .out_weight = out_weight,
            .out_bias = try Matrix.init(gpa, 1, d_out),
            .num_heads = num_heads,
            .head_dim = d_out / num_heads,
        };
    }

    /// `x` is [T, d_in]; returns [T, d_out].
    pub fn forward(self: MultiHeadAttention, gpa: std.mem.Allocator, x: Matrix) !Matrix {
        const q = try x.matmul(gpa, self.w_query);
        const k = try x.matmul(gpa, self.w_key);
        const v = try x.matmul(gpa, self.w_value);
        const context = try Matrix.init(gpa, x.rows, self.num_heads * self.head_dim);
        for (0..self.num_heads) |h| {
            const lo = h * self.head_dim; // this head's column slice
            const hi = lo + self.head_dim;
            const qh = try q.sliceCols(gpa, lo, hi);
            const kh = try k.sliceCols(gpa, lo, hi);
            const vh = try v.sliceCols(gpa, lo, hi);
            const scores = try (try qh.matmul(gpa, try kh.transpose(gpa))).maskFuture(gpa);
            const weights = try (try sa.scaleBySqrt(gpa, scores, self.head_dim)).softmaxRows(gpa);
            const head = try weights.matmul(gpa, vh);
            for (0..head.rows) |i| @memcpy(context.row(i)[lo..hi], head.row(i)); // glue the heads back together
        }
        const out = try context.matmul(gpa, self.out_weight);
        for (0..out.rows) |i| {
            for (0..out.cols) |j| out.data[i * out.cols + j] += self.out_bias.data[j];
        }
        return out;
    }
};
