3.5: پنهان کردن کلمات آینده با توجه علّی

بخش کتاب 3.5 · کد به زبان Python, Go, Rust, C, Zig

هدف. جلوی دیدن آینده را توسط توکن‌ها بگیرید تا مدل بتواند پیش‌بینی توکن بعدی را یاد بگیرد، و dropout را برای مقابله با بیش‌برازش اضافه کنید.

GPT برای پیش‌بینی توکن بعدی آموزش می‌بیند، پس وقتی توکن i را پردازش می‌کند فقط باید از توکن‌های 0 تا i استفاده کند. اگر بتواند به توکن i+1 نگاه کند، پاسخ را کپی می‌کند. توجه علّی (یا ماسک‌شده) همین را تضمین می‌کند: وزن‌های توجه بالای قطر حذف می‌شوند.

دو راه برای ماسک کردن

ماسک پس از softmax

همه‌چیز بالای قطر را صفر کن، سپس هر سطر را بر مجموع جدیدش تقسیم کن تا دوباره به 1 برسد. درست است اما یک گام اضافه می‌خواهد.

ماسک پیش از softmax (مورد استفاده)

امتیازهای آینده را با -inf بازنویسی کن، سپس softmax بگیر. exp(-inf) = 0 و مجموع سطر باز هم 1 است. یک گام، همان نتیجه.
scores                      after the -inf mask
[ s00  s01  s02 ]           [ s00  -inf  -inf ]
[ s10  s11  s12 ]   ---->   [ s10  s11   -inf ]
[ s20  s21  s22 ]           [ s20  s21   s22  ]

با عددهای واقعی بررسی کنید. برای «journey» وزن‌های بدون ماسک 0.1500 برای «Your» و 0.2264 برای خودش بود. فقط همین دو را نگه دارید و دوباره نرمال کنید: 0.1500 / (0.1500 + 0.2264) = 0.3985. سطر توجهٔ ماسک‌شده واقعاً 0.3986 را نشان می‌دهد. ماسک کردن پیش یا پس از softmax جواب یکسان می‌دهد (تا حد گرد کردن).

  [1.0000, 0.0000, 0.0000, 0.0000, 0.0000, 0.0000]
  [0.3986, 0.6014, 0.0000, 0.0000, 0.0000, 0.0000]
  [0.2526, 0.3791, 0.3683, 0.0000, 0.0000, 0.0000]

  [0.1557, 0.2092, 0.2048, 0.1419, 0.1089, 0.1794]

توکن اول فقط می‌تواند به خودش توجه کند (وزن 1.0). توکن آخر همه‌چیز را می‌بیند، پس سطرش صفر ندارد.

Dropout

Dropout در آموزش به‌صورت تصادفی بعضی مقادیر را صفر می‌کند تا مدل به هیچ اتصال تنهایی تکیه نکند. آن را بعد از softmax روی وزن‌های توجه اعمال می‌کنیم. دو نکته:

  • مقادیر باقی‌مانده 1 / (1 - p) برابر بزرگ می‌شوند. با p = 0.5 مقادیر نگه‌داشته‌شده دو برابر می‌شوند. این میانگین را ثابت نگه می‌دارد، پس هنگام استنتاج که dropout خاموش است نیازی به بازمقیاس نیست.
  • فقط در آموزش است. کتاب برای نمایش 50% را به کار برده؛ آموزش واقعی GPT از 0.1 تا 0.2 استفاده می‌کند.
dropout 0.5 on a 6 x 6 matrix of ones (our seeded generator):
  [2, 2, 2, 2, 2, 2]
  [2, 0, 2, 0, 2, 2]
  [0, 0, 2, 0, 0, 0]
  ...
الگوی دقیق به مولد تصادفی بستگی دارد. خروجی torch کتاب با ما فرق دارد؛ خروجی ما از چند خط splitmix64 می‌آید تا هر پنج زبان با هم یکی باشند.

دسته‌ها

CausalAttention کتاب یک دسته از دنباله‌ها را هم مدیریت می‌کند (شکل [batch, tokens, d_in])، چون دیتالودر فصل ۲ دسته تولید می‌کند. نسخه‌های Go، Rust، C و Zig ما هر بار یک دنباله را پردازش می‌کنند و دمو روی دسته حلقه می‌زند؛ ریاضی هر دنباله یکسان است. ماسک در torch به‌صورت بافر ذخیره می‌شود (register_buffer): همراه مدل ذخیره و به GPU منتقل می‌شود اما پارامتر یادگرفته‌شده نیست.

"""Step 8 (book 3.5) - causal attention: hide the future.

A GPT writes text left to right, so token i may only look at tokens 0..i.
We hide the later tokens with a mask. Set their scores to -inf *before*
softmax: exp(-inf) = 0, and the remaining weights in each row still sum to 1.

    scores (T x T)                       after the mask
    [ s00 s01 s02 ]                      [ s00 -inf -inf ]
    [ s10 s11 s12 ]      -------->       [ s10  s11 -inf ]
    [ s20 s21 s22 ]                      [ s20  s21  s22 ]

Dropout (training only) randomly zeroes some attention weights so the model
does not lean on any single connection.
"""

import math

import torch
from torch import nn

from .rng import Rng, dropout
from .self_attention import SelfAttention


class CausalAttention(SelfAttention):
    def __init__(self, w_query, w_key, w_value, context_length: int, dropout_p: float = 0.0, rng: Rng | None = None):
        super().__init__(w_query, w_key, w_value)
        self.dropout_p = dropout_p
        self.rng = rng
        # True above the diagonal = "future" positions to hide. A buffer is saved
        # with the model but is not a learned parameter.
        self.register_buffer("mask", torch.triu(torch.ones(context_length, context_length), diagonal=1).bool())

    def attention_weights(self, x: torch.Tensor, train: bool = False) -> torch.Tensor:
        """``x`` is [T, d_in] or [batch, T, d_in]."""
        queries, keys = x @ self.W_query, x @ self.W_key
        scores = queries @ keys.transpose(-2, -1)
        t = x.shape[-2]
        scores = scores.masked_fill(self.mask[:t, :t], -math.inf)
        weights = torch.softmax(scores / math.sqrt(keys.shape[-1]), dim=-1)
        if train and self.dropout_p > 0:
            weights = dropout(weights, self.dropout_p, self.rng)
        return weights

    def forward(self, x: torch.Tensor, train: bool = False) -> torch.Tensor:
        return self.attention_weights(x, train) @ (x @ self.W_value)

"""Tiny seeded random generator (splitmix64), so every language gives the same numbers.

Used for random weights and for dropout masks. Same seed, same numbers, on
every run and in every language of this repo (Go, Rust, C and Zig have the
same few lines).
"""

import torch

MASK = (1 << 64) - 1


class Rng:
    def __init__(self, seed: int):
        self.state = seed & MASK

    def next_u64(self) -> int:
        self.state = (self.state + 0x9E3779B97F4A7C15) & MASK
        z = self.state
        z = ((z ^ (z >> 30)) * 0xBF58476D1CE4E5B9) & MASK
        z = ((z ^ (z >> 27)) * 0x94D049BB133111EB) & MASK
        return z ^ (z >> 31)

    def next_f64(self) -> float:
        """Uniform in [0, 1)."""
        return (self.next_u64() >> 11) / 9007199254740992.0  # 2**53

    def rand(self, rows: int, cols: int) -> torch.Tensor:
        """A [rows, cols] float32 matrix, uniform in [0, 1), like ``torch.rand``."""
        values = [self.next_f64() for _ in range(rows * cols)]
        return torch.tensor(values, dtype=torch.float32).reshape(rows, cols)


def dropout(x: torch.Tensor, p: float, rng: Rng) -> torch.Tensor:
    """Zero each element with probability ``p`` and scale the rest by 1/(1-p).

    The scaling keeps the average value unchanged, so nothing needs rescaling
    at inference time (when dropout is simply switched off).
    """
    keep = [1.0 if rng.next_f64() >= p else 0.0 for _ in range(x.numel())]
    mask = torch.tensor(keep, dtype=x.dtype).reshape(x.shape)
    return x * mask / (1.0 - p)

// Step 8 (book 3.5) - causal attention: hide the future.
//
// A GPT writes text left to right, so token i may only look at tokens 0..i.
// We hide the later tokens with a mask. Set their scores to -Inf before
// softmax: exp(-Inf) = 0, and the remaining weights in each row still sum to 1.
//
//	scores (T x T)                       after the mask
//	[ s00 s01 s02 ]                      [ s00 -inf -inf ]
//	[ s10 s11 s12 ]      -------->       [ s10  s11 -inf ]
//	[ s20 s21 s22 ]                      [ s20  s21  s22 ]
//
// Dropout (training only) randomly zeroes some attention weights so the model
// does not lean on any single connection.

type CausalAttention struct {
	SelfAttention
	DropoutP float32
	Rng      *Rng
}

func NewCausalAttention(wq, wk, wv Matrix, dropoutP float32, rng *Rng) *CausalAttention {
	return &CausalAttention{SelfAttention{wq, wk, wv}, dropoutP, rng}
}

// CausalWeights takes x [T, d_in]. With train set, dropout is applied.
func (a *CausalAttention) CausalWeights(x Matrix, train bool) Matrix {
	queries, keys := MatMul(x, a.WQuery), MatMul(x, a.WKey)
	scores := MaskFuture(MatMul(queries, Transpose(keys)))
	weights := SoftmaxRows(scaleBySqrt(scores, keys.Cols))
	if train && a.DropoutP > 0 {
		weights = Dropout(weights, a.DropoutP, a.Rng)
	}
	return weights
}

func (a *CausalAttention) CausalForward(x Matrix, train bool) Matrix {
	return MatMul(a.CausalWeights(x, train), MatMul(x, a.WValue))
}

package attention

// Tiny seeded random generator (splitmix64), so every language gives the same
// numbers. Used for random weights and for dropout masks.
type Rng struct{ state uint64 }

func NewRng(seed uint64) *Rng { return &Rng{seed} }

func (r *Rng) nextU64() uint64 {
	r.state += 0x9E3779B97F4A7C15
	z := r.state
	z = (z ^ (z >> 30)) * 0xBF58476D1CE4E5B9
	z = (z ^ (z >> 27)) * 0x94D049BB133111EB
	return z ^ (z >> 31)
}

// NextF64 is uniform in [0, 1).
func (r *Rng) NextF64() float64 { return float64(r.nextU64()>>11) / 9007199254740992.0 }

// Rand returns a rows x cols matrix, uniform in [0, 1), like torch.rand.
func (r *Rng) Rand(rows, cols int) Matrix {
	m := NewMatrix(rows, cols)
	for i := range m.Data {
		m.Data[i] = float32(r.NextF64())
	}
	return m
}

// Dropout zeroes each element with probability p and scales the rest by 1/(1-p).
// The scaling keeps the average unchanged, so nothing needs rescaling at
// inference time (when dropout is simply switched off).
func Dropout(m Matrix, p float32, r *Rng) Matrix {
	out := NewMatrix(m.Rows, m.Cols)
	for i, v := range m.Data {
		if r.NextF64() >= float64(p) {
			out.Data[i] = v / (1 - p)
		}
	}
	return out
}

//! Step 8 (book 3.5) - causal attention: hide the future.
//!
//! A GPT writes text left to right, so token i may only look at tokens 0..i.
//! We hide the later tokens with a mask. Set their scores to -inf *before*
//! softmax: exp(-inf) = 0, and the remaining weights in each row still sum to 1.
//!
//! ```text
//! scores (T x T)                       after the mask
//! [ s00 s01 s02 ]                      [ s00 -inf -inf ]
//! [ s10 s11 s12 ]      -------->       [ s10  s11 -inf ]
//! [ s20 s21 s22 ]                      [ s20  s21  s22 ]
//! ```
//!
//! Dropout (training only) randomly zeroes some attention weights so the model
//! does not lean on any single connection.

use super::self_attention::scale_by_sqrt;
use super::{Matrix, SelfAttention};
use crate::tokenization::rng::Rng;

pub struct CausalAttention {
    pub base: SelfAttention,
    pub dropout_p: f32,
}

impl CausalAttention {
    pub fn new(w_query: Matrix, w_key: Matrix, w_value: Matrix, dropout_p: f32) -> Self {
        Self {
            base: SelfAttention {
                w_query,
                w_key,
                w_value,
            },
            dropout_p,
        }
    }

    /// `x` is [T, d_in]. Pass an `rng` and `train = true` to apply dropout.
    pub fn attention_weights(&self, x: &Matrix, rng: Option<&mut Rng>) -> Matrix {
        let queries = x.matmul(&self.base.w_query);
        let keys = x.matmul(&self.base.w_key);
        let scores = queries.matmul(&keys.transpose()).mask_future();
        let weights = scale_by_sqrt(&scores, keys.cols).softmax_rows();
        match rng {
            Some(rng) if self.dropout_p > 0.0 => weights.dropout(self.dropout_p, rng),
            _ => weights,
        }
    }

    pub fn forward(&self, x: &Matrix, rng: Option<&mut Rng>) -> Matrix {
        self.attention_weights(x, rng)
            .matmul(&x.matmul(&self.base.w_value))
    }
}

//! Tiny seeded random generator (splitmix64), so the crate needs no `rand`.
//!
//! Used for shuffling batches and for random embedding weights. Same seed,
//! same numbers, on every run.

pub struct Rng(u64);

impl Rng {
    pub fn new(seed: u64) -> Self {
        Self(seed)
    }

    fn next_u64(&mut self) -> u64 {
        self.0 = self.0.wrapping_add(0x9E37_79B9_7F4A_7C15);
        let mut z = self.0;
        z = (z ^ (z >> 30)).wrapping_mul(0xBF58_476D_1CE4_E5B9);
        z = (z ^ (z >> 27)).wrapping_mul(0x94D0_49BB_1331_11EB);
        z ^ (z >> 31)
    }

    /// Uniform in [0, 1).
    pub fn next_f64(&mut self) -> f64 {
        (self.next_u64() >> 11) as f64 / (1u64 << 53) as f64
    }

    /// Uniform integer in [0, n).
    pub fn below(&mut self, n: usize) -> usize {
        (self.next_u64() % n as u64) as usize
    }

    /// Normal(0, 1) via the Box-Muller transform.
    pub fn normal(&mut self) -> f32 {
        let u1 = 1.0 - self.next_f64(); // avoid ln(0)
        let u2 = self.next_f64();
        ((-2.0 * u1.ln()).sqrt() * (2.0 * std::f64::consts::PI * u2).cos()) as f32
    }

    /// Fisher-Yates shuffle.
    pub fn shuffle<T>(&mut self, items: &mut [T]) {
        for i in (1..items.len()).rev() {
            items.swap(i, self.below(i + 1));
        }
    }
}

#include "causal.h"

Matrix causal_weights(Arena *a, const CausalAttention *ca, Matrix x, Rng *rng) {
    Matrix queries = matmul(a, x, ca->base.w_query), keys = matmul(a, x, ca->base.w_key);
    Matrix scores = mask_future(a, matmul(a, queries, transpose(a, keys)));
    Matrix weights = softmax_rows(a, scale_by_sqrt(a, scores, keys.cols));
    if (rng && ca->dropout_p > 0) weights = dropout(a, weights, ca->dropout_p, rng);
    return weights;
}

Matrix causal_forward(Arena *a, const CausalAttention *ca, Matrix x, Rng *rng) {
    return matmul(a, causal_weights(a, ca, x, rng), matmul(a, x, ca->base.w_value));
}

C-version/tokenization/rng.c
#include "rng.h"

#include <math.h>

Rng rng_new(uint64_t seed) { return (Rng){seed}; }

static uint64_t next_u64(Rng *r) {
    r->state += 0x9E3779B97F4A7C15ULL;
    uint64_t z = r->state;
    z = (z ^ (z >> 30)) * 0xBF58476D1CE4E5B9ULL;
    z = (z ^ (z >> 27)) * 0x94D049BB133111EBULL;
    return z ^ (z >> 31);
}

double rng_f64(Rng *r) { return (double)(next_u64(r) >> 11) / 9007199254740992.0; /* 2^53 */ }

size_t rng_below(Rng *r, size_t n) { return (size_t)(next_u64(r) % n); }

float rng_normal(Rng *r) {
    double u1 = 1.0 - rng_f64(r); /* avoid log(0) */
    double u2 = rng_f64(r);
    return (float)(sqrt(-2.0 * log(u1)) * cos(2.0 * M_PI * u2));
}

void rng_shuffle(Rng *r, size_t *items, size_t n) {
    for (size_t i = n; i-- > 1;) {
        size_t j = rng_below(r, i + 1);
        size_t tmp = items[i];
        items[i] = items[j];
        items[j] = tmp;
    }
}

//! Step 8 (book 3.5) - causal attention: hide the future.
//!
//! A GPT writes text left to right, so token i may only look at tokens 0..i.
//! We hide the later tokens with a mask. Set their scores to -inf *before*
//! softmax: exp(-inf) = 0, and the remaining weights in each row still sum to 1.
//!
//! ```
//! scores (T x T)                       after the mask
//! [ s00 s01 s02 ]                      [ s00 -inf -inf ]
//! [ s10 s11 s12 ]      -------->       [ s10  s11 -inf ]
//! [ s20 s21 s22 ]                      [ s20  s21  s22 ]
//! ```
//!
//! Dropout (training only) randomly zeroes some attention weights so the model
//! does not lean on any single connection.

const std = @import("std");
const Matrix = @import("matrix.zig").Matrix;
const sa = @import("self_attention.zig");
const Rng = @import("../tokenization/rng.zig").Rng;

pub const CausalAttention = struct {
    base: sa.SelfAttention,
    dropout_p: f32 = 0,

    /// `x` is [T, d_in]. Pass an `rng` to apply dropout (training), or null (inference).
    pub fn attentionWeights(self: CausalAttention, gpa: std.mem.Allocator, x: Matrix, rng: ?*Rng) !Matrix {
        const queries = try x.matmul(gpa, self.base.w_query);
        const keys = try x.matmul(gpa, self.base.w_key);
        const scores = try (try queries.matmul(gpa, try keys.transpose(gpa))).maskFuture(gpa);
        const weights = try (try sa.scaleBySqrt(gpa, scores, keys.cols)).softmaxRows(gpa);
        if (rng) |r| {
            if (self.dropout_p > 0) return weights.dropout(gpa, self.dropout_p, r);
        }
        return weights;
    }

    pub fn forward(self: CausalAttention, gpa: std.mem.Allocator, x: Matrix, rng: ?*Rng) !Matrix {
        const weights = try self.attentionWeights(gpa, x, rng);
        return weights.matmul(gpa, try x.matmul(gpa, self.base.w_value));
    }
};

//! Tiny seeded random generator (splitmix64), no library needed.
//!
//! Used for shuffling batches and for random embedding weights. Same seed,
//! same numbers, on every run. (Same algorithm as the Rust and C versions.)

const std = @import("std");

pub const Rng = struct {
    state: u64,

    pub fn init(seed: u64) Rng {
        return .{ .state = seed };
    }

    fn nextU64(self: *Rng) u64 {
        self.state +%= 0x9E37_79B9_7F4A_7C15;
        var z = self.state;
        z = (z ^ (z >> 30)) *% 0xBF58_476D_1CE4_E5B9;
        z = (z ^ (z >> 27)) *% 0x94D0_49BB_1331_11EB;
        return z ^ (z >> 31);
    }

    /// Uniform in [0, 1).
    pub fn nextF64(self: *Rng) f64 {
        return @as(f64, @floatFromInt(self.nextU64() >> 11)) / 9007199254740992.0; // 2^53
    }

    /// Uniform integer in [0, n).
    pub fn below(self: *Rng, n: usize) usize {
        return @intCast(self.nextU64() % n);
    }

    /// Normal(0, 1) via the Box-Muller transform.
    pub fn normal(self: *Rng) f32 {
        const a = 1.0 - self.nextF64(); // avoid log(0)
        const b = self.nextF64();
        return @floatCast(@sqrt(-2.0 * @log(a)) * @cos(2.0 * std.math.pi * b));
    }

    /// Fisher-Yates shuffle.
    pub fn shuffle(self: *Rng, comptime T: type, items: []T) void {
        var i = items.len;
        while (i > 1) {
            i -= 1;
            std.mem.swap(T, &items[i], &items[self.below(i + 1)]);
        }
    }
};
