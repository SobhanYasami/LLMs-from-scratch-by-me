2.6: Data sampling with a sliding window

Aim. Turn one long list of token IDs into training examples for next-token prediction.

BPE-encode the whole story: 5,145 tokens. Each example is a window of IDs (the input) and the same window shifted by one (the target). Every position in a window is a free question: "given these tokens, what comes next?"

x: [290, 4920, 2241, 287]
y:      [4920, 2241, 287, 257]

[290] ----> 4920
[290, 4920] ----> 2241
[290, 4920, 2241] ----> 287
[290, 4920, 2241, 287] ----> 257

as text:
 and ---->  established
 and established ---->  himself
 and established himself ---->  in
 and established himself in ---->  a

Left of the arrow is what the model sees, right is what it must predict. (The book first drops the first 50 tokens for a more interesting passage.)

Dataset and DataLoader

  • Dataset (GPTDatasetV1): holds every (input, target) window as tensors. Python extends torch's Dataset; Go and Rust use plain slices.
  • max_length: window size (the context). Tiny here (4); real training uses 256 or more.
  • stride: how far the window slides. stride=1 gives heavily overlapping windows; stride == max_length gives none.
  • DataLoader: groups windows into batches, optionally shuffled, optionally dropping a short last batch.
stride=1, batch_size=1:
[tensor([[  40,  367, 2885, 1464]]), tensor([[ 367, 2885, 1464, 1807]])]   first window
[tensor([[ 367, 2885, 1464, 1807]]), tensor([[2885, 1464, 1807, 3619]])]   second: shifted by one

stride=4, batch_size=8: inputs shape (8, 4), no overlap, no word skipped
The book uses stride = max_length for real batches: the whole text is used but nothing overlaps, since more overlap risks overfitting. Batch size is a trade-off: small uses less memory but gives noisier updates.

Code

python-version/tokenization/dataset.py
"""Step 4 - Sliding-window data sampling for next-token prediction.

Given a long list of token IDs, cut it into overlapping windows:

    ids     = [10, 11, 12, 13, 14, 15]
    window 0: input [10, 11, 12, 13]  target [11, 12, 13, 14]
    window 1: input [12, 13, 14, 15]  target [13, 14, 15, 16]   (if it fits)

Why shifted by one? The model's job is: given the tokens so far, guess the
next one, so every position in a window is a free training example.
The target is the input shifted by one position. ``stride`` is how far the
window moves each step; ``stride == max_length`` means no overlap.
"""

import torch
from torch.utils.data import DataLoader, Dataset


class GPTDatasetV1(Dataset):
    """Holds every (input, target) window of one tokenized text."""

    def __init__(self, txt, tokenizer, max_length, stride):
        self.input_ids = []
        self.target_ids = []
        token_ids = tokenizer.encode(txt)
        # Stop at len - max_length: the target reaches one token past the input.
        for i in range(0, len(token_ids) - max_length, stride):
            input_chunk = token_ids[i : i + max_length]
            target_chunk = token_ids[i + 1 : i + max_length + 1]
            self.input_ids.append(torch.tensor(input_chunk))
            self.target_ids.append(torch.tensor(target_chunk))

    def __len__(self):
        return len(self.input_ids)

    def __getitem__(self, idx):
        return self.input_ids[idx], self.target_ids[idx]


def create_dataloader_v1(txt, tokenizer, batch_size=4, max_length=256, stride=128, shuffle=True, drop_last=True):
    """Wrap ``GPTDatasetV1`` in a batching ``DataLoader``.

    Why batch? The GPU processes many windows at once. Why shuffle? Random order
    smooths training. Why ``drop_last``? A short final batch can upset training.
    """
    dataset = GPTDatasetV1(txt, tokenizer, max_length, stride)
    return DataLoader(dataset, batch_size=batch_size, shuffle=shuffle, drop_last=drop_last)
go-version/tokenization/dataset.go
package tokenization

// Step 4 - sliding-window data sampling for next-token prediction.
//
//	ids      = [10 11 12 13 14 15]
//	window 0: input [10 11 12 13]  target [11 12 13 14]
//
// Why shifted by one? The model's job is: given the tokens so far, guess the
// next one. Every position in a window is a free training example.
//
// The target is the input shifted by one position. Stride is how far the window
// moves each step; stride == maxLength means no overlap.

// GPTDataset holds every (input, target) window of one tokenized text.
type GPTDataset struct {
	inputIDs  [][]int
	targetIDs [][]int
}

// NewGPTDataset cuts tokenIDs into windows, like Python's GPTDatasetV1.
func NewGPTDataset(tokenIDs []int, maxLength, stride int) *GPTDataset {
	d := &GPTDataset{}
	// Stop at len-maxLength: the target reaches one token past the input.
	for i := 0; i < len(tokenIDs)-maxLength; i += stride {
		d.inputIDs = append(d.inputIDs, tokenIDs[i:i+maxLength])
		d.targetIDs = append(d.targetIDs, tokenIDs[i+1:i+maxLength+1])
	}
	return d
}

func (d *GPTDataset) Len() int { return len(d.inputIDs) }

// Get returns the input and target of window idx.
func (d *GPTDataset) Get(idx int) (input, target []int) {
	return d.inputIDs[idx], d.targetIDs[idx]
}

// Batches groups consecutive windows, dropping a final short batch.
func (d *GPTDataset) Batches(batchSize int) (inputs, targets [][][]int) {
	for i := 0; i+batchSize <= d.Len(); i += batchSize {
		inputs = append(inputs, d.inputIDs[i:i+batchSize])
		targets = append(targets, d.targetIDs[i:i+batchSize])
	}
	return inputs, targets
}
go-version/tokenization/dataloader.go
package tokenization

import "math/rand"

// Step 4b - batching. Mirrors torch's DataLoader.
//
// Why batch? The GPU processes many windows at once, which is much faster.
// Why shuffle? Windows in book order make training swing; random order smooths it.
// Why drop the last short batch? A different size can upset training.

// Batch is one group of windows: inputs[b] and targets[b] are one (input, target) pair.
type Batch struct {
	Inputs  [][]int
	Targets [][]int
}

// DataLoader serves a GPTDataset in batches.
type DataLoader struct {
	ds        *GPTDataset
	batchSize int
	shuffle   bool
	dropLast  bool
	rng       *rand.Rand
}

// NewDataLoader seeds the shuffle so runs are repeatable.
func NewDataLoader(ds *GPTDataset, batchSize int, shuffle, dropLast bool, seed int64) *DataLoader {
	return &DataLoader{ds, batchSize, shuffle, dropLast, rand.New(rand.NewSource(seed))}
}

// Epoch returns every batch once. With shuffle, each call uses a new order.
func (l *DataLoader) Epoch() []Batch {
	order := make([]int, l.ds.Len())
	for i := range order {
		order[i] = i
	}
	if l.shuffle {
		l.rng.Shuffle(len(order), func(i, j int) { order[i], order[j] = order[j], order[i] })
	}
	var batches []Batch
	for i := 0; i < len(order); i += l.batchSize {
		end := min(i+l.batchSize, len(order))
		if end-i < l.batchSize && l.dropLast {
			break
		}
		var b Batch
		for _, idx := range order[i:end] {
			in, tg := l.ds.Get(idx)
			b.Inputs = append(b.Inputs, in)
			b.Targets = append(b.Targets, tg)
		}
		batches = append(batches, b)
	}
	return batches
}
rust-version/src/tokenization/dataset.rs
//! Step 4 - sliding-window data sampling for next-token prediction.
//!
//! ```text
//! ids      = [10 11 12 13 14 15]
//! window 0: input [10 11 12 13]  target [11 12 13 14]
//! ```
//!
//! Why shifted by one? The model's job is: given the tokens so far, guess the
//! next one. Every position in a window is a free training example.
//!
//! The target is the input shifted by one position. `stride` is how far the
//! window moves each step; `stride == max_length` means no overlap.

/// Holds every (input, target) window of one tokenized text.
pub struct GptDataset {
    input_ids: Vec<Vec<u32>>,
    target_ids: Vec<Vec<u32>>,
}

impl GptDataset {
    /// Cut `token_ids` into windows, like Python's `GPTDatasetV1`.
    pub fn new(token_ids: &[u32], max_length: usize, stride: usize) -> Self {
        let mut input_ids = Vec::new();
        let mut target_ids = Vec::new();
        // Stop at len - max_length: the target reaches one token past the input.
        for i in (0..token_ids.len().saturating_sub(max_length)).step_by(stride) {
            input_ids.push(token_ids[i..i + max_length].to_vec());
            target_ids.push(token_ids[i + 1..i + max_length + 1].to_vec());
        }
        Self {
            input_ids,
            target_ids,
        }
    }

    pub fn len(&self) -> usize {
        self.input_ids.len()
    }

    pub fn is_empty(&self) -> bool {
        self.input_ids.is_empty()
    }

    /// The input and target of window `idx`.
    pub fn get(&self, idx: usize) -> (&[u32], &[u32]) {
        (&self.input_ids[idx], &self.target_ids[idx])
    }

    /// Group consecutive windows, dropping a final short batch.
    pub fn batches(&self, batch_size: usize) -> impl Iterator<Item = (&[Vec<u32>], &[Vec<u32>])> {
        self.input_ids
            .chunks_exact(batch_size)
            .zip(self.target_ids.chunks_exact(batch_size))
    }
}
rust-version/src/tokenization/dataloader.rs
//! Step 4b - batching. Mirrors torch's `DataLoader`.
//!
//! Why batch? The GPU processes many windows at once, which is much faster.
//! Why shuffle? Windows in book order make training swing; random order smooths it.
//! Why drop the last short batch? A different size can upset training.

use super::dataset::GptDataset;
use super::rng::Rng;

/// One group of windows: `inputs[b]` and `targets[b]` are one (input, target) pair.
pub struct Batch {
    pub inputs: Vec<Vec<u32>>,
    pub targets: Vec<Vec<u32>>,
}

/// Serves a `GptDataset` in batches.
pub struct DataLoader<'a> {
    dataset: &'a GptDataset,
    batch_size: usize,
    shuffle: bool,
    drop_last: bool,
    rng: Rng,
}

impl<'a> DataLoader<'a> {
    /// `seed` makes the shuffle repeatable.
    pub fn new(
        dataset: &'a GptDataset,
        batch_size: usize,
        shuffle: bool,
        drop_last: bool,
        seed: u64,
    ) -> Self {
        Self {
            dataset,
            batch_size,
            shuffle,
            drop_last,
            rng: Rng::new(seed),
        }
    }

    /// Every batch once. With shuffle, each call uses a new order.
    pub fn epoch(&mut self) -> Vec<Batch> {
        let mut order: Vec<usize> = (0..self.dataset.len()).collect();
        if self.shuffle {
            self.rng.shuffle(&mut order);
        }
        order
            .chunks(self.batch_size)
            .filter(|c| !(self.drop_last && c.len() < self.batch_size))
            .map(|chunk| {
                let (inputs, targets) = chunk
                    .iter()
                    .map(|&i| {
                        let (x, y) = self.dataset.get(i);
                        (x.to_vec(), y.to_vec())
                    })
                    .unzip();
                Batch { inputs, targets }
            })
            .collect()
    }
}
The loop stops at len - max_length because the target reaches one token past the input.
