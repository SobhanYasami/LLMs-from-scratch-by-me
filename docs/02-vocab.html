Step 2: Build a vocabulary

Aim. Give every unique token a number. The table token -> ID is the bridge between text and the model.
{'!': 0, '"': 1, ..., '<|endoftext|>': 1159, '<|unk|>': 1160}
  • Sorted so the same corpus always gives the same IDs. Sets and maps have random order in all three languages.
  • Special tokens go last so word IDs do not move when you add one: <|endoftext|> marks the boundary between documents, <|unk|> stands for any unseen word.

Code

python-version/tokenization/vocab.py
"""Step 2 - Build a vocabulary: a mapping from each token to a unique integer ID.

    vocab = {"!": 0, '"': 1, ..., "<|endoftext|>": 1159, "<|unk|>": 1160}

Computers work with numbers, so every token must get an ID before a model
can use it. The vocabulary is the lookup table between the two worlds.

Tip: a Python ``set`` has no fixed order. Sorting before numbering is what makes
the IDs the same on every run.
"""

from .preprocess import END_OF_TEXT, UNK_TOKEN

SPECIAL_TOKENS = [END_OF_TEXT, UNK_TOKEN]


def build_vocab(tokens: list[str]) -> dict[str, int]:
    """Assign IDs to the unique tokens in sorted order, then append special tokens.

    Sorting makes the IDs deterministic: the same corpus always gives the same vocab.
    Special tokens go last so the regular word IDs stay stable.
    """
    all_tokens = sorted(set(tokens))
    all_tokens.extend(SPECIAL_TOKENS)
    return {token: integer for integer, token in enumerate(all_tokens)}
go-version/tokenization/vocab.go
package tokenization

import "sort"

// Step 2 - build a vocabulary: a mapping from each token to a unique integer ID.
//
//	{"!": 0, "\"": 1, ..., "<|endoftext|>": 1159, "<|unk|>": 1160}
//
// Models only understand numbers, so this table is the bridge between text and IDs.
//
// Tip: a Go map has random iteration order. Sorting before numbering is what
// makes the IDs the same on every run.

// SpecialTokens are appended after the regular tokens so word IDs stay stable.
var SpecialTokens = []string{EndOfText, Unknown}

// BuildVocab assigns IDs to the unique tokens in sorted order, then appends the
// special tokens. Sorting makes IDs deterministic for a given corpus.
func BuildVocab(tokens []string) map[string]int {
	set := make(map[string]struct{})
	for _, t := range tokens {
		set[t] = struct{}{}
	}
	all := make([]string, 0, len(set)+len(SpecialTokens))
	for t := range set {
		all = append(all, t)
	}
	sort.Strings(all)
	all = append(all, SpecialTokens...)

	vocab := make(map[string]int, len(all))
	for i, t := range all {
		vocab[t] = i
	}
	return vocab
}
rust-version/src/tokenization/vocab.rs
//! Step 2 - build a vocabulary: a mapping from each token to a unique integer ID.
//!
//! ```text
//! {"!": 0, "\"": 1, ..., "<|endoftext|>": 1159, "<|unk|>": 1160}
//! ```
//!
//! Models only understand numbers, so this table is the bridge between text and IDs.
//!
//! Tip: a `HashMap` has random iteration order. Collecting into a sorted
//! `BTreeSet` before numbering is what makes IDs the same on every run.

use std::collections::{BTreeSet, HashMap};

use super::preprocess::{END_OF_TEXT, UNK_TOKEN};

pub const SPECIAL_TOKENS: [&str; 2] = [END_OF_TEXT, UNK_TOKEN];

/// Assign IDs to the unique tokens in sorted order, then append special tokens.
///
/// Sorting makes IDs deterministic for a given corpus. Special tokens go last so
/// regular word IDs stay stable.
pub fn build_vocab(tokens: &[String]) -> HashMap<String, usize> {
    let unique: BTreeSet<&str> = tokens.iter().map(String::as_str).collect();
    unique
        .into_iter()
        .chain(SPECIAL_TOKENS)
        .enumerate()
        .map(|(i, t)| (t.to_string(), i))
        .collect()
}
