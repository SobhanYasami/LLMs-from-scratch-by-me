2.7-2.8: Token embeddings and word positions

Book section 2.7, 2.8 · code in Python, Go, Rust, C, Zig

Aim. Turn integer IDs into vectors the model can learn from, and put word order into those vectors.

2.7 Token embeddings

An embedding layer is a weight matrix with one row per vocabulary entry and one column per dimension. Converting an ID is a row lookup.

vocab_size = 6, output_dim = 3
weight matrix (random at start):
 id 0  [ 0.3374, -0.1778, -0.1690]
 id 1  [ 0.9178,  1.5810,  1.3010]
 id 2  [ 1.2753, -0.2010, -0.1606]
 id 3  [-0.4015,  0.9666, -1.1481]   <- embedding of token 3
 id 4  [-1.1589,  0.3255, -0.6315]
 id 5  [-2.8400, -0.7849, -1.4096]

input_ids [2, 3, 5, 1]  ->  rows 2, 3, 5, 1  (a 4 x 3 matrix)
  • Weights start random and are tuned by backpropagation during training (chapter 5). That is why the layer is trainable and why vectors become meaningful.
  • An embedding layer is just a faster version of one-hot encoding times a matrix multiply (see Raschka's bonus folder 03).
  • With the real BPE vocabulary: 50,257 rows, and the book uses 256 dimensions (GPT-3 uses 12,288).

2.8 Encoding word positions

The same ID always maps to the same vector, wherever it sits. Self-attention is also blind to order. So we add position information.

  • Absolute position embeddings: one vector per position (first, second, ...). GPT uses these, learned during training.
  • Relative ones encode the distance between tokens and generalise better to unseen sequence lengths.
token ids          (8, 4)         batch of 8 windows, 4 tokens each
token embeddings   (8, 4, 256)    every ID became a 256-d vector
position embeddings (4, 256)      one vector per position 0..3
input embeddings   (8, 4, 256)    token + position  <- what the model reads

The position table is looked up with 0, 1, ..., context_length - 1 and added to every window in the batch. Text longer than the context length must be truncated.

python-version/tokenization/embedding.py
"""Step 5 - Token and positional embeddings.

A token ID is just an integer. An embedding layer is a lookup table with one
trainable vector per ID, so every token becomes a vector the model can learn from:

    ids [8, 4]  --token_embedding-->  [8, 4, D]
    positions 0..3 --pos_embedding--> [4, D]  (broadcast over the batch)
    input_embeddings = token_embeddings + pos_embeddings   -> [8, 4, D]

Position vectors are needed because attention (later chapters) does not know
word order by itself. GPT-style models learn them as absolute positions.
"""

import torch
from torch import nn


class InputEmbedding(nn.Module):
    """Token embedding plus absolute positional embedding."""

    def __init__(self, vocab_size: int, context_length: int, output_dim: int):
        super().__init__()
        self.token_embedding = nn.Embedding(vocab_size, output_dim)
        self.pos_embedding = nn.Embedding(context_length, output_dim)

    def forward(self, ids: torch.Tensor) -> torch.Tensor:
        """``ids`` has shape [batch, seq_len]; returns [batch, seq_len, output_dim]."""
        token_embeddings = self.token_embedding(ids)
        positions = torch.arange(ids.shape[1])
        return token_embeddings + self.pos_embedding(positions)
go-version/tokenization/embedding.go
package tokenization

import "math/rand"

// Step 5 - token and positional embeddings.
//
// A token ID is just an integer. An embedding layer is a lookup table with one
// vector per ID, so every token becomes a vector the model can learn from:
//
//	ids [B][T]  --token embedding-->  [B][T][D]
//	positions 0..T-1 --pos embedding-> [T][D]  (added to every batch row)
//	input embeddings = token + position  -> [B][T][D]
//
// Why add positions? Attention (later chapters) treats its input as a bag of
// vectors, so word order must be put into the vectors themselves.
//
// There is no tensor library here: tensors are nested slices, and weights are
// random (normal, mean 0, std 1) and untrained, like torch's nn.Embedding at init.

// Embedding is a lookup table of numRows vectors of size dim.
type Embedding struct {
	Weight [][]float32
}

func NewEmbedding(numRows, dim int, rng *rand.Rand) *Embedding {
	w := make([][]float32, numRows)
	for i := range w {
		w[i] = make([]float32, dim)
		for j := range w[i] {
			w[i][j] = float32(rng.NormFloat64())
		}
	}
	return &Embedding{Weight: w}
}

// Forward looks up one vector per ID.
func (e *Embedding) Forward(ids []int) [][]float32 {
	out := make([][]float32, len(ids))
	for i, id := range ids {
		out[i] = e.Weight[id]
	}
	return out
}

// InputEmbedding is a token embedding plus an absolute positional embedding.
type InputEmbedding struct {
	Token *Embedding
	Pos   *Embedding
}

func NewInputEmbedding(vocabSize, contextLength, dim int, seed int64) *InputEmbedding {
	rng := rand.New(rand.NewSource(seed))
	return &InputEmbedding{
		Token: NewEmbedding(vocabSize, dim, rng),
		Pos:   NewEmbedding(contextLength, dim, rng),
	}
}

// Forward takes ids of shape [batch][seqLen] and returns [batch][seqLen][dim].
func (m *InputEmbedding) Forward(ids [][]int) [][][]float32 {
	out := make([][][]float32, len(ids))
	for b, row := range ids {
		tok := m.Token.Forward(row)
		sum := make([][]float32, len(row))
		for t := range row {
			sum[t] = make([]float32, len(tok[t]))
			for d := range sum[t] {
				sum[t][d] = tok[t][d] + m.Pos.Weight[t][d]
			}
		}
		out[b] = sum
	}
	return out
}
rust-version/src/tokenization/embedding.rs
//! Step 5 - token and positional embeddings.
//!
//! A token ID is just an integer. An embedding layer is a lookup table with one
//! vector per ID, so every token becomes a vector the model can learn from:
//!
//! ```text
//! ids [B][T]  --token embedding-->  [B][T][D]
//! positions 0..T-1 --pos embedding-> [T][D]  (added to every batch row)
//! input embeddings = token + position  -> [B][T][D]
//! ```
//!
//! Why add positions? Attention (later chapters) treats its input as a bag of
//! vectors, so word order must be put into the vectors themselves.
//!
//! No tensor library: tensors are nested `Vec`s, and weights are random
//! (normal, mean 0, std 1) and untrained, like torch's `nn.Embedding` at init.

use super::rng::Rng;

/// A lookup table of `num_rows` vectors of size `dim`.
pub struct Embedding {
    pub weight: Vec<Vec<f32>>,
}

impl Embedding {
    pub fn new(num_rows: usize, dim: usize, rng: &mut Rng) -> Self {
        let weight = (0..num_rows)
            .map(|_| (0..dim).map(|_| rng.normal()).collect())
            .collect();
        Self { weight }
    }

    /// Look up one vector per ID.
    pub fn forward(&self, ids: &[u32]) -> Vec<Vec<f32>> {
        ids.iter()
            .map(|&id| self.weight[id as usize].clone())
            .collect()
    }
}

/// Token embedding plus absolute positional embedding.
pub struct InputEmbedding {
    pub token: Embedding,
    pub pos: Embedding,
}

impl InputEmbedding {
    pub fn new(vocab_size: usize, context_length: usize, dim: usize, seed: u64) -> Self {
        let mut rng = Rng::new(seed);
        let token = Embedding::new(vocab_size, dim, &mut rng);
        let pos = Embedding::new(context_length, dim, &mut rng);
        Self { token, pos }
    }

    /// `ids` has shape [batch][seq_len]; returns [batch][seq_len][dim].
    pub fn forward(&self, ids: &[Vec<u32>]) -> Vec<Vec<Vec<f32>>> {
        ids.iter()
            .map(|row| {
                self.token
                    .forward(row)
                    .into_iter()
                    .zip(&self.pos.weight)
                    .map(|(tok, pos)| tok.iter().zip(pos).map(|(a, b)| a + b).collect())
                    .collect()
            })
            .collect()
    }
}
rust-version/src/tokenization/rng.rs
//! Tiny seeded random generator (splitmix64), so the crate needs no `rand`.
//!
//! Used for shuffling batches and for random embedding weights. Same seed,
//! same numbers, on every run.

pub struct Rng(u64);

impl Rng {
    pub fn new(seed: u64) -> Self {
        Self(seed)
    }

    fn next_u64(&mut self) -> u64 {
        self.0 = self.0.wrapping_add(0x9E37_79B9_7F4A_7C15);
        let mut z = self.0;
        z = (z ^ (z >> 30)).wrapping_mul(0xBF58_476D_1CE4_E5B9);
        z = (z ^ (z >> 27)).wrapping_mul(0x94D0_49BB_1331_11EB);
        z ^ (z >> 31)
    }

    /// Uniform in [0, 1).
    pub fn next_f64(&mut self) -> f64 {
        (self.next_u64() >> 11) as f64 / (1u64 << 53) as f64
    }

    /// Uniform integer in [0, n).
    pub fn below(&mut self, n: usize) -> usize {
        (self.next_u64() % n as u64) as usize
    }

    /// Normal(0, 1) via the Box-Muller transform.
    pub fn normal(&mut self) -> f32 {
        let u1 = 1.0 - self.next_f64(); // avoid ln(0)
        let u2 = self.next_f64();
        ((-2.0 * u1.ln()).sqrt() * (2.0 * std::f64::consts::PI * u2).cos()) as f32
    }

    /// Fisher-Yates shuffle.
    pub fn shuffle<T>(&mut self, items: &mut [T]) {
        for i in (1..items.len()).rev() {
            items.swap(i, self.below(i + 1));
        }
    }
}

/* Step 5 (book 2.7-2.8) - token and positional embeddings.
 *
 * A token ID is just an integer. An embedding layer is a lookup table with
 * one vector per ID, so every token becomes a vector the model can learn from:
 *
 *   ids [B][T]  --token embedding-->  [B][T][D]
 *   positions 0..T-1 --pos embedding-> [T][D]  (added to every batch row)
 *   input embeddings = token + position  -> [B][T][D]
 *
 * Why add positions? Attention (later chapters) treats its input as a bag of
 * vectors, so word order must be put into the vectors themselves.
 *
 * No tensor library: a tensor is one flat float array, indexed [b][t][d] as
 * (b * T + t) * D + d. Weights are random (normal, mean 0, std 1) and untrained.
 */

#define EMBEDDING_H

#include <stdint.h>

    size_t rows; /* vocabulary size (or context length) */

C-version/tokenization/embedding.c
#include "embedding.h"

#include <stdlib.h>
#include <string.h>

#include "util.h"

Embedding embedding_new(size_t rows, size_t dim, Rng *rng) {
    Embedding e = {rows, dim, xmalloc(rows * dim * sizeof(float))};
    for (size_t i = 0; i < rows * dim; i++) e.weight[i] = rng_normal(rng);
    return e;
}

void embedding_free(Embedding *e) {
    free(e->weight);
    *e = (Embedding){0};
}

void embedding_forward(const Embedding *e, const int *ids, size_t n, float *out) {
    for (size_t i = 0; i < n; i++) /* a lookup: copy row `ids[i]` */
        memcpy(out + i * e->dim, e->weight + (size_t)ids[i] * e->dim, e->dim * sizeof(float));
}

InputEmbedding input_embedding_new(size_t vocab_size, size_t context_length, size_t dim, uint64_t seed) {
    Rng rng = rng_new(seed);
    InputEmbedding m;
    m.token = embedding_new(vocab_size, dim, &rng);
    m.pos = embedding_new(context_length, dim, &rng);
    return m;
}

void input_embedding_free(InputEmbedding *m) {
    embedding_free(&m->token);
    embedding_free(&m->pos);
}

void input_embedding_forward(const InputEmbedding *m, const int *ids, size_t batch, size_t seq_len, float *out) {
    size_t dim = m->token.dim;
    for (size_t b = 0; b < batch; b++) {
        float *row = out + b * seq_len * dim;
        embedding_forward(&m->token, ids + b * seq_len, seq_len, row);
        for (size_t t = 0; t < seq_len; t++)         /* token + position */
            for (size_t d = 0; d < dim; d++) row[t * dim + d] += m->pos.weight[t * dim + d];
    }
}

C-version/tokenization/rng.c
#include "rng.h"

#include <math.h>

Rng rng_new(uint64_t seed) { return (Rng){seed}; }

static uint64_t next_u64(Rng *r) {
    r->state += 0x9E3779B97F4A7C15ULL;
    uint64_t z = r->state;
    z = (z ^ (z >> 30)) * 0xBF58476D1CE4E5B9ULL;
    z = (z ^ (z >> 27)) * 0x94D049BB133111EBULL;
    return z ^ (z >> 31);
}

double rng_f64(Rng *r) { return (double)(next_u64(r) >> 11) / 9007199254740992.0; /* 2^53 */ }

size_t rng_below(Rng *r, size_t n) { return (size_t)(next_u64(r) % n); }

float rng_normal(Rng *r) {
    double u1 = 1.0 - rng_f64(r); /* avoid log(0) */
    double u2 = rng_f64(r);
    return (float)(sqrt(-2.0 * log(u1)) * cos(2.0 * M_PI * u2));
}

void rng_shuffle(Rng *r, size_t *items, size_t n) {
    for (size_t i = n; i-- > 1;) {
        size_t j = rng_below(r, i + 1);
        size_t tmp = items[i];
        items[i] = items[j];
        items[j] = tmp;
    }
}
Zig-version/src/tokenization/embedding.zig
//! Step 5 (book 2.7-2.8) - token and positional embeddings.
//!
//! A token ID is just an integer. An embedding layer is a lookup table with one
//! vector per ID, so every token becomes a vector the model can learn from:
//!
//! ```
//! ids [B][T]  --token embedding-->  [B][T][D]
//! positions 0..T-1 --pos embedding-> [T][D]  (added to every batch row)
//! input embeddings = token + position  -> [B][T][D]
//! ```
//!
//! Why add positions? Attention (later chapters) treats its input as a bag of
//! vectors, so word order must be put into the vectors themselves.
//!
//! No tensor library: a tensor is one flat `[]f32`, indexed [b][t][d] as
//! `(b * T + t) * D + d`. Weights are random (normal, mean 0, std 1) and untrained,
//! like torch's `nn.Embedding` at init.

const std = @import("std");
const Rng = @import("rng.zig").Rng;

/// A lookup table of `rows` vectors of size `dim`.
pub const Embedding = struct {
    rows: usize,
    dim: usize,
    weight: []f32, // rows x dim

    pub fn init(gpa: std.mem.Allocator, rows: usize, dim: usize, rng: *Rng) !Embedding {
        const weight = try gpa.alloc(f32, rows * dim);
        for (weight) |*w| w.* = rng.normal();
        return .{ .rows = rows, .dim = dim, .weight = weight };
    }

    pub fn deinit(self: *Embedding, gpa: std.mem.Allocator) void {
        gpa.free(self.weight);
    }

    /// Look up one vector per ID; `out` holds `ids.len * dim` floats.
    pub fn forward(self: Embedding, ids: []const u32, out: []f32) void {
        for (ids, 0..) |id, i| { // a lookup: copy row `id`
            @memcpy(out[i * self.dim ..][0..self.dim], self.weight[@as(usize, id) * self.dim ..][0..self.dim]);
        }
    }
};

/// Token embedding plus absolute positional embedding.
pub const InputEmbedding = struct {
    token: Embedding,
    pos: Embedding,

    pub fn init(gpa: std.mem.Allocator, vocab_size: usize, context_length: usize, dim: usize, seed: u64) !InputEmbedding {
        var rng = Rng.init(seed);
        var token = try Embedding.init(gpa, vocab_size, dim, &rng);
        errdefer token.deinit(gpa);
        const pos = try Embedding.init(gpa, context_length, dim, &rng);
        return .{ .token = token, .pos = pos };
    }

    pub fn deinit(self: *InputEmbedding, gpa: std.mem.Allocator) void {
        self.token.deinit(gpa);
        self.pos.deinit(gpa);
    }

    /// `ids` is [batch][seq_len] flat; `out` is [batch][seq_len][dim] flat.
    pub fn forward(self: InputEmbedding, ids: []const u32, batch: usize, seq_len: usize, out: []f32) void {
        const dim = self.token.dim;
        for (0..batch) |b| {
            const row = out[b * seq_len * dim ..][0 .. seq_len * dim];
            self.token.forward(ids[b * seq_len ..][0..seq_len], row);
            for (0..seq_len) |t| { // token + position
                for (0..dim) |d| row[t * dim + d] += self.pos.weight[t * dim + d];
            }
        }
    }
};

//! Tiny seeded random generator (splitmix64), no library needed.
//!
//! Used for shuffling batches and for random embedding weights. Same seed,
//! same numbers, on every run. (Same algorithm as the Rust and C versions.)

const std = @import("std");

pub const Rng = struct {
    state: u64,

    pub fn init(seed: u64) Rng {
        return .{ .state = seed };
    }

    fn nextU64(self: *Rng) u64 {
        self.state +%= 0x9E37_79B9_7F4A_7C15;
        var z = self.state;
        z = (z ^ (z >> 30)) *% 0xBF58_476D_1CE4_E5B9;
        z = (z ^ (z >> 27)) *% 0x94D0_49BB_1331_11EB;
        return z ^ (z >> 31);
    }

    /// Uniform in [0, 1).
    pub fn nextF64(self: *Rng) f64 {
        return @as(f64, @floatFromInt(self.nextU64() >> 11)) / 9007199254740992.0; // 2^53
    }

    /// Uniform integer in [0, n).
    pub fn below(self: *Rng, n: usize) usize {
        return @intCast(self.nextU64() % n);
    }

    /// Normal(0, 1) via the Box-Muller transform.
    pub fn normal(self: *Rng) f32 {
        const a = 1.0 - self.nextF64(); // avoid log(0)
        const b = self.nextF64();
        return @floatCast(@sqrt(-2.0 * @log(a)) * @cos(2.0 * std.math.pi * b));
    }

    /// Fisher-Yates shuffle.
    pub fn shuffle(self: *Rng, comptime T: type, items: []T) void {
        var i = items.len;
        while (i > 1) {
            i -= 1;
            std.mem.swap(T, &items[i], &items[self.below(i + 1)]);
        }
    }
};
