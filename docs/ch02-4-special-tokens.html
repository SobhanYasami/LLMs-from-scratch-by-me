2.4: Adding special context tokens

Book section 2.4 · code in Python, Go, Rust, C, Zig

Aim. Handle unknown words and document boundaries by adding two special tokens to the vocabulary.
TokenRoleID in our vocab
<|unk|>stands for any word not in the vocabulary1160
<|endoftext|>separates unrelated texts when concatenated for training, so the model does not treat two books as one story1159

They are appended last so existing word IDs do not shift. The vocabulary grows from 1,159 to 1,161.

SimpleTokenizerV2

Same as V1, but before lookup any token not in the vocabulary is replaced by <|unk|>, so encoding never fails.

text    = "Hello, do you like tea? <|endoftext|> In the sunlit terraces of the palace."
ids     = [1160, 5, 362, 1155, 642, 1000, 10, 1159, 57, 1013, 981, 1009, 738, 1013, 1160, 7]
decoded = "<|unk|>, do you like tea? <|endoftext|> In the sunlit terraces of the <|unk|>."

Hello and palace were never in the story, so both became <|unk|> (ID 1160) and their meaning is lost. The <|endoftext|> separator became 1159.

Other special tokens you will meet

  • [BOS] beginning of sequence.
  • [EOS] end of sequence, like <|endoftext|>.
  • [PAD] pads shorter texts in a batch to equal length.
GPT uses only <|endoftext|>, for separating and padding (padded positions are masked out later, so which token pads does not matter). GPT also has no <|unk|>: byte pair encoding on the next page replaces it.

python-version/tokenization/simple.py
"""Step 3 - Simple tokenizers that combine splitting with the vocabulary.

Both classes expose the same two methods:

    encode(text) -> list[int]   text  -> token IDs
    decode(ids)  -> str         IDs   -> text

V1 only knows the words in its vocabulary and raises ``KeyError`` otherwise.
V2 replaces unknown words with ``<|unk|>`` so encoding never fails.
"""

from .preprocess import UNK_TOKEN, detokenize, split_text


def _invert(vocab: dict[str, int]) -> dict[int, str]:
    """Swap keys and values: ID -> token, needed for decoding."""
    return {integer: token for token, integer in vocab.items()}


class SimpleTokenizerV1:
    """Encode and decode with a vocabulary; unknown tokens are an error."""

    def __init__(self, vocab: dict[str, int]):
        self.str_to_int = vocab              # token -> ID
        self.int_to_str = _invert(vocab)     # ID -> token

    def encode(self, text: str) -> list[int]:
        tokens = split_text(text)
        return [self.str_to_int[token] for token in tokens]  # KeyError if unknown

    def decode(self, ids: list[int]) -> str:
        tokens = [self.int_to_str[i] for i in ids]
        return detokenize(tokens)


class SimpleTokenizerV2(SimpleTokenizerV1):
    """Same as V1, but unknown tokens become ``<|unk|>`` instead of raising."""

    def encode(self, text: str) -> list[int]:
        tokens = split_text(text)
        tokens = [
            token if token in self.str_to_int else UNK_TOKEN
            for token in tokens
        ]
        return [self.str_to_int[token] for token in tokens]
go-version/tokenization/simple.go
package tokenization

import "fmt"

// Step 3 - simple tokenizers: split text, then look each token up in the vocab.
//
//	encode: "Hello, world" -> ["Hello", ",", "world"] -> [1131, 5, 987]
//	decode: [1131, 5, 987] -> ["Hello", ",", "world"] -> "Hello, world"
//
// Tip: compare the two types. V1 fails on a word it never saw; V2 swaps it for
// <|unk|> and keeps going. That one difference is why real models use BPE.

// SimpleTokenizerV1 maps known tokens to IDs and fails on unknown tokens.
type SimpleTokenizerV1 struct {
	strToInt map[string]int // token -> ID, used by Encode
	intToStr map[int]string // ID -> token, used by Decode (the vocab, inverted)
}

func NewSimpleTokenizerV1(vocab map[string]int) *SimpleTokenizerV1 {
	intToStr := make(map[int]string, len(vocab))
	for s, i := range vocab {
		intToStr[i] = s
	}
	return &SimpleTokenizerV1{strToInt: vocab, intToStr: intToStr}
}

// encode is shared by V1 and V2. The only difference is what happens to a
// token that is not in the vocab: error (V1) or <|unk|> (V2).
func (t *SimpleTokenizerV1) encode(text string, unknownToUnk bool) ([]int, error) {
	var ids []int
	for _, token := range SplitText(text) {
		id, ok := t.strToInt[token]
		if !ok && unknownToUnk {
			id, ok = t.strToInt[Unknown]
		}
		if !ok {
			return nil, fmt.Errorf("token not in vocab: %q", token)
		}
		ids = append(ids, id)
	}
	return ids, nil
}

func (t *SimpleTokenizerV1) Encode(text string) ([]int, error) { return t.encode(text, false) }

func (t *SimpleTokenizerV1) Decode(ids []int) (string, error) {
	tokens := make([]string, 0, len(ids))
	for _, id := range ids {
		token, ok := t.intToStr[id]
		if !ok {
			return "", fmt.Errorf("id not in vocab: %d", id)
		}
		tokens = append(tokens, token)
	}
	return Detokenize(tokens), nil
}

// SimpleTokenizerV2 is V1, except unknown tokens become <|unk|>.
// It embeds V1, so Decode is inherited unchanged.
type SimpleTokenizerV2 struct {
	SimpleTokenizerV1
}

func NewSimpleTokenizerV2(vocab map[string]int) *SimpleTokenizerV2 {
	return &SimpleTokenizerV2{SimpleTokenizerV1: *NewSimpleTokenizerV1(vocab)}
}

func (t *SimpleTokenizerV2) Encode(text string) ([]int, error) { return t.encode(text, true) }
rust-version/src/tokenization/simple.rs
//! Step 3 - simple tokenizers: split text, then look each token up in the vocab.
//!
//! ```text
//! encode: "Hello, world" -> ["Hello", ",", "world"] -> [1131, 5, 987]
//! decode: [1131, 5, 987] -> ["Hello", ",", "world"] -> "Hello, world"
//! ```
//!
//! Tip: compare the two types. V1 fails on a word it never saw; V2 swaps it for
//! `<|unk|>` and keeps going. That one difference is why real models use BPE.

use std::collections::HashMap;

use super::preprocess::{UNK_TOKEN, detokenize, split_text};

/// Encode and decode with a vocabulary; unknown tokens are an error.
pub struct SimpleTokenizerV1 {
    str_to_int: HashMap<String, usize>, // token -> ID, used by encode
    int_to_str: HashMap<usize, String>, // ID -> token, used by decode (the vocab, inverted)
}

impl SimpleTokenizerV1 {
    pub fn new(vocab: HashMap<String, usize>) -> Self {
        let int_to_str = vocab.iter().map(|(s, &i)| (i, s.clone())).collect();
        Self {
            str_to_int: vocab,
            int_to_str,
        }
    }

    /// Shared by V1 and V2. The only difference is what happens to a token that
    /// is not in the vocab: error (V1) or `<|unk|>` (V2).
    fn encode_with(&self, text: &str, unknown_to_unk: bool) -> Result<Vec<usize>, String> {
        split_text(text)
            .iter()
            .map(|token| {
                let known = self.str_to_int.get(token);
                let id = match (known, unknown_to_unk) {
                    (None, true) => self.str_to_int.get(UNK_TOKEN),
                    _ => known,
                };
                id.copied()
                    .ok_or_else(|| format!("token not in vocab: {token:?}"))
            })
            .collect()
    }

    pub fn encode(&self, text: &str) -> Result<Vec<usize>, String> {
        self.encode_with(text, false)
    }

    pub fn decode(&self, ids: &[usize]) -> Result<String, String> {
        let tokens = ids
            .iter()
            .map(|id| {
                self.int_to_str
                    .get(id)
                    .cloned()
                    .ok_or_else(|| format!("id not in vocab: {id}"))
            })
            .collect::<Result<Vec<_>, _>>()?;
        Ok(detokenize(&tokens))
    }
}

/// V1, except unknown tokens become `<|unk|>`. Wraps V1, so decode is reused.
pub struct SimpleTokenizerV2(SimpleTokenizerV1);

impl SimpleTokenizerV2 {
    pub fn new(vocab: HashMap<String, usize>) -> Self {
        Self(SimpleTokenizerV1::new(vocab))
    }

    pub fn encode(&self, text: &str) -> Result<Vec<usize>, String> {
        self.0.encode_with(text, true)
    }

    pub fn decode(&self, ids: &[usize]) -> Result<String, String> {
        self.0.decode(ids)
    }
}

/* Step 3 (book 2.3-2.4) - simple tokenizers: split text, look tokens up.
 *
 *   encode: "Hello, world" -> ["Hello", ",", "world"] -> [1131, 5, 987]
 *   decode: [1131, 5, 987] -> ["Hello", ",", "world"] -> "Hello, world"
 *
 * Tip: compare V1 and V2. V1 fails on a word it never saw; V2 swaps it for
 * <|unk|> and keeps going. That one difference is why real models use BPE.
 */
#ifndef SIMPLE_H
#define SIMPLE_H

#include <stdbool.h>
#include <stddef.h>

#include "util.h"
#include "vocab.h"

typedef struct {
    const Vocab *vocab;
    bool unknown_to_unk; /* false = V1 (error), true = V2 (<|unk|>) */
} SimpleTokenizer;

SimpleTokenizer simple_tokenizer_v1(const Vocab *vocab);
SimpleTokenizer simple_tokenizer_v2(const Vocab *vocab);

/* text -> IDs. Returns NULL and fills `err` on an unknown token (V1). Caller frees. */
int *simple_encode(const SimpleTokenizer *t, const char *text, size_t *n, char err[ERR_LEN]);
/* IDs -> text. Returns NULL and fills `err` on an unknown ID. Caller frees. */
char *simple_decode(const SimpleTokenizer *t, const int *ids, size_t n, char err[ERR_LEN]);

#endif
C-version/tokenization/simple.c
#include "simple.h"

#include <stdio.h>
#include <stdlib.h>

#include "preprocess.h"

SimpleTokenizer simple_tokenizer_v1(const Vocab *vocab) { return (SimpleTokenizer){vocab, false}; }
SimpleTokenizer simple_tokenizer_v2(const Vocab *vocab) { return (SimpleTokenizer){vocab, true}; }

/* V1 and V2 share this code. The only difference is what happens to a token
 * that is not in the vocab: error (V1) or <|unk|> (V2). */
int *simple_encode(const SimpleTokenizer *t, const char *text, size_t *n, char err[ERR_LEN]) {
    Tokens tokens = split_text(text);
    int *ids = xmalloc(tokens.len * sizeof(int));
    for (size_t i = 0; i < tokens.len; i++) {
        int id = vocab_id(t->vocab, tokens.items[i]);
        if (id < 0 && t->unknown_to_unk) id = vocab_id(t->vocab, UNK_TOKEN);
        if (id < 0) {
            snprintf(err, ERR_LEN, "token not in vocab: \"%s\"", tokens.items[i]);
            free(ids);
            tokens_free(&tokens);
            return NULL;
        }
        ids[i] = id;
    }
    *n = tokens.len;
    tokens_free(&tokens);
    return ids;
}

char *simple_decode(const SimpleTokenizer *t, const int *ids, size_t n, char err[ERR_LEN]) {
    const char **tokens = xmalloc(n * sizeof(char *));
    for (size_t i = 0; i < n; i++) {
        tokens[i] = vocab_token(t->vocab, ids[i]);
        if (!tokens[i]) {
            snprintf(err, ERR_LEN, "id not in vocab: %d", ids[i]);
            free(tokens);
            return NULL;
        }
    }
    char *text = detokenize(tokens, n);
    free(tokens);
    return text;
}
Zig-version/src/tokenization/simple.zig
//! Step 3 (book 2.3-2.4) - simple tokenizers: split text, look tokens up.
//!
//! ```
//! encode: "Hello, world" -> ["Hello", ",", "world"] -> [1131, 5, 987]
//! decode: [1131, 5, 987] -> ["Hello", ",", "world"] -> "Hello, world"
//! ```
//!
//! Tip: compare V1 and V2. V1 fails on a word it never saw; V2 swaps it for
//! `<|unk|>` and keeps going. That one difference is why real models use BPE.

const std = @import("std");
const preprocess = @import("preprocess.zig");
const Vocab = @import("vocab.zig").Vocab;

pub const SimpleTokenizer = struct {
    vocab: *const Vocab,
    /// false = V1 (error), true = V2 (`<|unk|>`).
    unknown_to_unk: bool,

    pub fn v1(vocab: *const Vocab) SimpleTokenizer {
        return .{ .vocab = vocab, .unknown_to_unk = false };
    }

    pub fn v2(vocab: *const Vocab) SimpleTokenizer {
        return .{ .vocab = vocab, .unknown_to_unk = true };
    }

    /// text -> IDs. V1 fails with `error.UnknownToken` on a word outside the
    /// vocabulary; V2 uses the `<|unk|>` ID instead. Caller frees.
    pub fn encode(self: SimpleTokenizer, gpa: std.mem.Allocator, text: []const u8) ![]u32 {
        const tokens = try preprocess.splitText(gpa, text);
        defer gpa.free(tokens);
        const ids = try gpa.alloc(u32, tokens.len);
        errdefer gpa.free(ids);
        for (tokens, ids) |token, *id| {
            id.* = self.vocab.idOf(token) orelse
                (if (self.unknown_to_unk) self.vocab.idOf(preprocess.UNK_TOKEN) else null) orelse
                return error.UnknownToken;
        }
        return ids;
    }

    /// IDs -> text. Fails with `error.UnknownId` for an ID outside the vocabulary. Caller frees.
    pub fn decode(self: SimpleTokenizer, gpa: std.mem.Allocator, ids: []const u32) ![]u8 {
        const tokens = try gpa.alloc([]const u8, ids.len);
        defer gpa.free(tokens);
        for (ids, tokens) |id, *token| token.* = self.vocab.tokenOf(id) orelse return error.UnknownId;
        return preprocess.detokenize(gpa, tokens);
    }
};

fn testVocab(gpa: std.mem.Allocator) !Vocab {
    return Vocab.build(gpa, &.{ "Hello", ",", "world" });
}

test "V1 fails on unknown words, V2 uses <|unk|>" {
    const gpa = std.testing.allocator;
    var vocab = try testVocab(gpa);
    defer vocab.deinit(gpa);

    try std.testing.expectError(error.UnknownToken, SimpleTokenizer.v1(&vocab).encode(gpa, "Hello there"));

    const ids = try SimpleTokenizer.v2(&vocab).encode(gpa, "Hello there");
    defer gpa.free(ids);
    try std.testing.expectEqual(vocab.idOf(preprocess.UNK_TOKEN).?, ids[1]);
}

test "encode then decode round-trips simple text" {
    const gpa = std.testing.allocator;
    var vocab = try testVocab(gpa);
    defer vocab.deinit(gpa);
    const t = SimpleTokenizer.v1(&vocab);
    const ids = try t.encode(gpa, "Hello, world");
    defer gpa.free(ids);
    const text = try t.decode(gpa, ids);
    defer gpa.free(text);
    try std.testing.expectEqualStrings("Hello, world", text);
}
